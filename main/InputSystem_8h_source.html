25// Per-frame input state machine: held/pressed/released sets for keys + mouse buttons, accumulated
26// mouse wheel delta, action map (named action → list of keys), and the script-installed
27// callback vectors that fire on each edge. EventBus subscriptions feed OnKeyInput / OnMouseInput
28// / OnMouseWheel; the Lua `input.*` surface (is_key_*, on_key_*, bind, etc.) lives in
29// `Lua/Bindings/InputSystemLuaBinding.cpp` — InputSystem just exposes the public accessors +
30// mutators that binding's lambdas call into.

46 static std::string MakeKey(const std::string& key) {
47 std::string lower = key;
48 std::transform(lower.begin(), lower.end(), lower.begin(),
49 [](const unsigned char c) { return static_cast<char>(std::tolower(c)); });
50 return lower;
51 }

53 void SubscribeToEvents(const std::unique_ptr<EventBus>& eventBus, Registry* registry) {
54 registry_ = registry;
55 eventBus_ = eventBus.get();
56 subscriptions_.clear();
57 subscriptions_.push_back(eventBus->SubscribeEvent<InputSystem, KeyInputEvent>(this, &InputSystem::OnKeyInput));
58 subscriptions_.push_back(eventBus->SubscribeEvent<InputSystem, MouseInputEvent>(this, &InputSystem::OnMouseInput));
59 subscriptions_.push_back(eventBus->SubscribeEvent<InputSystem, MouseWheelEvent>(this, &InputSystem::OnMouseWheel));
60 subscriptions_.push_back(
61 eventBus->SubscribeEvent<InputSystem, GamepadButtonEvent>(this, &InputSystem::OnGamepadButton));
62 }

189 template <typename... Args>
190 static void Dispatch(const std::vector<sol::protected_function>& fns, const Args&... args) {
191 for (const auto& fn : fns) {
192 if (fn == sol::lua_nil) continue;
193 auto result = fn(args...);
194 if (!result.valid()) {
195 const sol::error err = result;
196 Logger::ErrorLua(std::string(err.what()));
197 }
198 }
199 }
