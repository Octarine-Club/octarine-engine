15// Lock-free, multi-producer render queue. Pre-sized to a fixed capacity; parallel
16// systems atomically claim a slot via fetch_add and write directly. No per-system
17// channel + playback merge — that intermediate copy was the bulk of the overhead in
18// the previous CommandBuffer-driven design.
19//
20// Sort() uses an LSB radix sort on the precomputed RenderKey::sortKey. With ~32K+ keys
21// this is dramatically faster than std::sort on the full 80-byte RenderKey because the
22// radix passes only touch 16-byte (key, srcIdx) entries. The full 80-byte permutation is
23// applied via a gather into a scratch buffer at the end — sequential writes are
24// prefetch-friendly and the source data fits in L3 cache at typical entry counts.
25//
26// Capacity must comfortably exceed the max keys-per-frame for the scene; an overflow
27// asserts in debug. Bump Constants::kInitialRenderQueueCapacity if you hit it.

135 for (size_t i = 0; i < clip_rects_.size(); ++i) {
136 const auto& existing = clip_rects_[i];
137 if (existing.x == rect.x && existing.y == rect.y && existing.w == rect.w && existing.h == rect.h) {

159 void Sort() {
160 const size_t n = std::min(count_.load(std::memory_order_relaxed), render_keys_.size());
161 if (n <= 1) return;
162
163 // Build (sortKey, srcIdx) entries — 16 bytes each. Sorting these instead of the
164 // 80-byte RenderKeys is the main speedup; we only gather the heavy keys once at
165 // the end via a sequential-write scatter into a scratch buffer.
166 sort_entries_.resize(n);
167 for (size_t i = 0; i < n; ++i) {
168 sort_entries_[i].key = render_keys_[i].sortKey;
169 sort_entries_[i].srcIdx = static_cast<std::uint32_t>(i);
170 }
171
172 RadixSort();
173
174 // Gather: write the sorted RenderKeys sequentially into permute_scratch_ by
175 // reading from random source positions. Sequential writes are prefetch-friendly;
176 // the random reads hit ~2.5 MB of source data which fits comfortably in L3 cache.
177 // The scratch vector is kept alive across frames so this doesn't allocate after
178 // the first frame.
179 // Resize to full capacity — after the swap this becomes render_keys_, and ClaimSlot()
180 // indexes into it up to the pre-sized capacity.
181 permute_scratch_.resize(render_keys_.size());
182 const auto* perm = sort_entries_.data();
183 const auto* src = render_keys_.data();
184 auto* dst = permute_scratch_.data();
185 for (size_t i = 0; i < n; ++i) {
186 dst[i] = src[perm[i].srcIdx];
187 }
188 std::swap(render_keys_, permute_scratch_);
189 }

218 // LSB radix sort on the 64-bit key with two optimizations over the naive 8-pass version:
219 //
220 // 1. Single-pass histogram: build all 8 byte-histograms in one scan of the data instead
221 // of re-reading the array for each pass. At 32k entries × 16 bytes this avoids ~3.5 MB
222 // of redundant reads and the associated cache-miss overhead.
223 //
224 // 2. Skip-uniform-byte: if every entry has the same value for a given byte (common — e.g.
225 // all sprites on layer 0 share the top 2 bytes), that pass cannot reorder anything.
226 // Skipping it saves both the scatter pass AND its prefix-sum. On a typical stress scene
227 // this cuts 8 passes to ~4.
228 //
229 // Pass-parity bookkeeping ensures the final sorted result lives in sort_entries_ regardless
230 // of how many passes actually executed.
231 static bool IsUniformHistogram(const std::array<size_t, 256>& hist, const size_t n) noexcept {
232 for (size_t bucket = 0; bucket < 256; ++bucket) {
233 if (hist[bucket] == n) {
234 return true;
235 }
236 }
237 return false;
238 }
239
240 void RadixSort() {
241 const size_t n = sort_entries_.size();
242 if (n <= 1) return;
243 radix_scratch_.resize(n);
244
245 // --- Phase 1: build all 8 histograms in one scan ---
246 std::array<std::array<size_t, 256>, 8> histograms{};
247 for (size_t i = 0; i < n; ++i) {
248 const std::uint64_t k = sort_entries_[i].key;
249 histograms[0][k & 0xFFu]++;
250 histograms[1][(k >> 8) & 0xFFu]++;
251 histograms[2][(k >> 16) & 0xFFu]++;
252 histograms[3][(k >> 24) & 0xFFu]++;
253 histograms[4][(k >> 32) & 0xFFu]++;
254 histograms[5][(k >> 40) & 0xFFu]++;
255 histograms[6][(k >> 48) & 0xFFu]++;
256 histograms[7][(k >> 56) & 0xFFu]++;
257 }
258
259 // --- Phase 2: determine which passes are needed ---
260 // A pass is skippable when exactly one bucket holds all n entries (all bytes identical).
261 std::array<bool, 8> skip{};
262 for (size_t pass = 0; pass < 8; ++pass) {
263 skip[pass] = IsUniformHistogram(histograms[pass], n);
264 }
265
266 // --- Phase 3: scatter passes (LSB → MSB), skipping uniform bytes ---
267 std::vector<SortEntry>* in = &sort_entries_;
268 std::vector<SortEntry>* out = &radix_scratch_;
269
270 for (size_t pass = 0; pass < 8; ++pass) {
271 if (skip[pass]) continue;
272
273 const size_t shift = pass * 8;
274
275 // Convert the histogram to exclusive prefix sums (offsets) for scatter.
276 auto& counts = histograms[pass];
277 size_t sum = 0;
278 for (auto& c : counts) {
279 const size_t saved = c;
280 c = sum;
281 sum += saved;
282 }
283
284 // Scatter
285 for (size_t i = 0; i < n; ++i) {
286 const auto bucket = static_cast<std::uint8_t>(((*in)[i].key >> shift) & 0xFFu);
287 (*out)[counts[bucket]++] = (*in)[i];
288 }
289 std::swap(in, out);
290 }
291
292 // Ensure the sorted result lives in sort_entries_.
293 if (in != &sort_entries_) {
294 std::swap(sort_entries_, radix_scratch_);
295 }
296 }
