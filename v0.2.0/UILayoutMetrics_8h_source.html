37inline float ParseFontPixelSize(const std::string& fontId) {
38 static constexpr float kFallbackFontPixelSize = 16.0f;
39 const auto dash = fontId.rfind('-');
40 if (dash == std::string::npos || dash + 1 >= fontId.size()) return kFallbackFontPixelSize;
41 char* endPtr = nullptr;
42 const float parsed = std::strtof(fontId.c_str() + dash + 1, &endPtr);
43 return (endPtr != fontId.c_str() + dash + 1 && parsed > 0.0f) ? parsed : kFallbackFontPixelSize;
44}

46inline bool IsUtf8ContinuationByte(const char c) {
47 static constexpr unsigned char kUtf8ContMask = 0xC0u;
48 static constexpr unsigned char kUtf8ContTag = 0x80u;
49 return (static_cast<unsigned char>(c) & kUtf8ContMask) == kUtf8ContTag;

52inline void FlushWordToWrappedLine(const float wordW, const float spaceW, const float maxW, float& currentLineW,
53 float& maxLineW, int& lines) {
54 if (wordW <= 0.0f) return;
55 const float neededSpace = (currentLineW > 0.0f) ? spaceW : 0.0f;
56 if (maxW > 0.0f && currentLineW > 0.0f && (currentLineW + neededSpace + wordW) > maxW) {
57 maxLineW = std::max(maxLineW, currentLineW);
58 currentLineW = wordW;
59 ++lines;
60 } else {
61 currentLineW += neededSpace + wordW;
62 }
63}

65inline glm::vec2 MeasureFallbackText(const std::string& text, const float fontSize, const int wrapWidth = 0) {
66 static constexpr float kCharWidthFactor = 0.55f;
67 const float charW = fontSize * kCharWidthFactor;
68 const float maxW = wrapWidth > 0 ? static_cast<float>(wrapWidth) : 0.0f;
69 float maxLineW = 0.0f;
70 float currentLineW = 0.0f;
71 float wordW = 0.0f;
72 int lines = 1;
73 for (const char c : text) {
74 if (IsUtf8ContinuationByte(c)) continue;
75 if (c == '\n') {
76 FlushWordToWrappedLine(wordW, charW, maxW, currentLineW, maxLineW, lines);
77 wordW = 0.0f;
78 maxLineW = std::max(maxLineW, currentLineW);
79 currentLineW = 0.0f;
80 ++lines;
81 } else if (c == ' ' || c == '\t') {
82 FlushWordToWrappedLine(wordW, charW, maxW, currentLineW, maxLineW, lines);
83 wordW = 0.0f;
84 } else {
85 wordW += charW;
86 }
87 }
88 FlushWordToWrappedLine(wordW, charW, maxW, currentLineW, maxLineW, lines);
89 maxLineW = std::max(maxLineW, currentLineW);
90 if (maxW > 0.0f) maxLineW = std::min(maxLineW, maxW);
91 return {std::ceil(maxLineW), std::ceil(static_cast<float>(lines) * fontSize)};
92}

94inline glm::vec2 MeasureFromAtlas(const GlyphAtlas& atlas, const std::string& text) {
95 float totalW = 0.0f;
96 for (const char c : text) {
97 const auto cp = static_cast<std::uint32_t>(static_cast<unsigned char>(c));
98 const auto* g = atlas.Find(cp);
99 if (g == nullptr) return {-1.0f, -1.0f};
100 totalW += g->advance;
101 }
102 return {std::ceil(totalW), static_cast<float>(std::max(1, atlas.LineSkip()))};
103}

105inline glm::vec2 MeasureTtfFontText(TTF_Font* font, const std::uint64_t fontGen, const TextLabelComponent& label,
106 const bool hasNewline, const int effWrapWidth) {
107 struct CacheKey {
108 TTF_Font* font = nullptr;
109 std::uint64_t fontGen = 0;
110 int wrapLen = 0;
111 bool wrapped = false;
112 std::string text{};
113 bool operator==(const CacheKey& o) const {
114 return font == o.font && fontGen == o.fontGen && wrapLen == o.wrapLen && wrapped == o.wrapped && text == o.text;
115 }
116 };
117 struct CacheKeyHash {
118 std::size_t operator()(const CacheKey& k) const {
119 std::size_t h = std::hash<const void*>{}(k.font);
120 h ^= std::hash<std::uint64_t>{}(k.fontGen) + 0x9e3779b9u + (h << 6) + (h >> 2);
121 h ^= std::hash<int>{}(k.wrapLen) + 0x9e3779b9u + (h << 6) + (h >> 2);
122 h ^= std::hash<bool>{}(k.wrapped) + 0x9e3779b9u + (h << 6) + (h >> 2);
123 h ^= std::hash<std::string>{}(k.text) + 0x9e3779b9u + (h << 6) + (h >> 2);
124 return h;
125 }
126 };
127 static constexpr std::size_t kMaxCacheEntries = 512;
128 static std::unordered_map<CacheKey, glm::vec2, CacheKeyHash> s_cache;
129
130 const bool wrapped = effWrapWidth > 0 || hasNewline;
131 const int wrapLen = effWrapWidth > 0 ? effWrapWidth : 0;
132 const CacheKey key{font, fontGen, wrapLen, wrapped, label.text};
133 if (const auto it = s_cache.find(key); it != s_cache.end()) {
134 return it->second;
135 }
136
137 int w = 0;
138 int h = 0;
139 const bool ok = wrapped ? TTF_GetStringSizeWrapped(font, label.text.c_str(), 0, wrapLen, &w, &h)
140 : TTF_GetStringSize(font, label.text.c_str(), 0, &w, &h);
141 if (!ok) return {-1.0f, -1.0f};
142 const glm::vec2 result{static_cast<float>(w), static_cast<float>(h)};
143 if (s_cache.size() >= kMaxCacheEntries) {
144 s_cache.clear();
145 }
146 s_cache.emplace(key, result);
147 return result;
148}

150inline glm::vec2 MeasureTextLabel(Registry* registry, const TextLabelComponent& label,
151 const int wrapWidthOverride = 0) {
152 if (label.text.empty()) return {0.0f, 0.0f};
153 const int effWrap = label.wrapWidth > 0 ? label.wrapWidth : wrapWidthOverride;
154 const auto* assetMgr = registry->TryGet<AssetManager>();
155 if (assetMgr == nullptr) {
156 return MeasureFallbackText(label.text, ParseFontPixelSize(label.fontId), effWrap);
157 }
158 const bool hasNewline = label.text.find('\n') != std::string::npos;
159 if (effWrap <= 0 && !hasNewline) {
160 const GlyphAtlas* atlas = assetMgr->GetGlyphAtlas(label.fontId);
161 if (atlas != nullptr && atlas->IsLoaded()) {
162 const glm::vec2 atlasSize = MeasureFromAtlas(*atlas, label.text);
163 if (atlasSize.x >= 0.0f) return atlasSize;
164 }
165 }
166 if (TTF_Font* font = assetMgr->GetFont(label.fontId)) {
167 const glm::vec2 ttfSize = MeasureTtfFontText(font, assetMgr->FontGeneration(), label, hasNewline, effWrap);
168 if (ttfSize.x >= 0.0f) return ttfSize;
169 }
170 return MeasureFallbackText(label.text, ParseFontPixelSize(label.fontId), effWrap);
171}

173inline SDL_Rect IntersectRects(const SDL_Rect& a, const SDL_Rect& b) {
174 const int left = std::max(a.x, b.x);
175 const int top = std::max(a.y, b.y);
176 const int right = std::min(a.x + a.w, b.x + b.w);
177 const int bottom = std::min(a.y + a.h, b.y + b.h);
178 return {left, top, std::max(0, right - left), std::max(0, bottom - top)};
179}

208inline InteractiveEntityFlags ReadInteractiveFlags(Registry* registry, const Entity entity, ResolvedVisualColors& out) {
209 InteractiveEntityFlags flags{};
210 if (registry->HasComponent<UIButtonComponent>(entity)) {
211 const auto& btn = registry->GetComponent<UIButtonComponent>(entity);
212 flags.isDisabled = !btn.isActive;
213 if (btn.isActive) {
214 flags.isHovered = btn.isHovered;
215 flags.isPressed = btn.isPressed;
216 if (flags.isHovered && btn.hasHoverColor) out.bg = btn.hoverColor;
217 if (flags.isPressed && btn.hasPressColor) out.bg = btn.pressColor;
218 }
219 }
220 if (registry->HasComponent<UIFocusComponent>(entity)) {
221 const auto& foc = registry->GetComponent<UIFocusComponent>(entity);
222 flags.isFocused = foc.isFocused;
223 flags.isChecked = foc.isChecked;
224 if (foc.isInert) {
225 flags.isDisabled = foc.isDisabled;
226 } else if (foc.isDisabled) {
227 flags.isDisabled = true;
228 }
229 }

233inline ResolvedVisualColors ResolveVisualState(Registry* registry, const Entity entity, const octarine::Color& baseBg,
234 const octarine::Color& baseBorder, const octarine::Color& baseText) {
235 ResolvedVisualColors out{baseBg, baseBorder, baseText};
236 if (!registry || !registry->IsAlive(entity)) return out;
237
238 const InteractiveEntityFlags flags = ReadInteractiveFlags(registry, entity, out);
239 std::uint8_t opacity = 255;
240 if (registry->HasComponent<UIStyleStateComponent>(entity)) {
241 const auto& st = registry->GetComponent<UIStyleStateComponent>(entity);
242 if (flags.isChecked) ApplyStylePatch(st.checked, out, opacity);
243 if (flags.isFocused) ApplyStylePatch(st.focus, out, opacity);
244 if (flags.isHovered) ApplyStylePatch(st.hover, out, opacity);
245 if (flags.isPressed) ApplyStylePatch(st.press, out, opacity);
246 if (flags.isDisabled) ApplyStylePatch(st.disabled, out, opacity);
247 }
248 out.bg = ApplyOpacityMod(out.bg, opacity);
249 out.border = ApplyOpacityMod(out.border, opacity);
250 out.text = ApplyOpacityMod(out.text, opacity);
251 return out;
252}
