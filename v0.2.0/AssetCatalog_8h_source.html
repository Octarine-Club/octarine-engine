36// One resolved, loadable asset. Produced by AssetCatalog::Build from a discovered file plus its
37// (optional) sidecar metadata. Holds no SDL/MIX handle — actually loading the asset is
38// AssetManager's job (it consults these entries via Acquire).
39struct CatalogEntry {
40 AssetType type{};
41 std::string id; // resolved unique id (filename stem, or meta.id; fonts append "-<size>")
42 std::string fullPath; // absolute path to the source file on disk
43 // Texture
44 std::optional<ScaleMode> scaleMode;
45 bool hasExplicitScaleMode{false}; // true when sidecar explicitly set `scale_mode`
46 std::optional<std::string> atlas; // sidecar `meta.atlas = <group>` (raw group name)
47 bool noAtlas{false}; // sidecar `meta.no_atlas = true` (escape hatch, see AssetMetadata.h)
48 // Resolved atlas membership — populated by TextureAtlasBaker after the scan. When set, this entry
49 // is rendered as a slice of `atlasId`'s texture instead of its own. Both fields move
50 // through asset_manifest.lua so shipped builds skip the bake.
51 std::optional<std::string> atlasId; // catalog id of the packed atlas texture (e.g. "__atlas_main")
52 std::optional<SDL_FRect> atlasSlice; // pixel-rect of this asset inside the atlas
53 // Font — one entry per size
54 float fontSize{0.0F};
55 // Audio
56 bool stream{false};
57 bool normalize{false};
58};

66 // Populate the id -> entry map for the project at `basePath`. When `allowManifest` is true and a
67 // baked `asset_manifest.lua` is present at the root, load it and skip the filesystem walk — assets
68 // resolve with no scan, which is what makes read-only APK/.app bundles work. Otherwise (dev build,
69 // or no manifest) recursively scan the convention folders and parse `<file>.meta` sidecars via the
70 // borrowed `lua` state. `allowManifest` is the dev-vs-shipped gate (OCTARINE_SHIPPED / the
71 // `--use-manifest` dev override): when false the manifest probe is skipped entirely, so a stray
72 // manifest beside dev assets can never hijack the scan. Returns false if any id collision (scan)
73 // or malformed entry (manifest) was found. The GameConfig overload pulls the project default scale
74 // mode from config; the explicit overload takes it directly (used by tests, which avoids
75 // GameConfig's SDL-backed load path).
76 bool Build(const std::string& basePath, sol::state& lua, const GameConfig& gameConfig, bool allowManifest = true,
77 const class AssetPak* pak = nullptr);
78 bool Build(const std::string& basePath, sol::state& lua, std::optional<ScaleMode> defaultScaleMode,
79 bool allowManifest = true, const class AssetPak* pak = nullptr);

81 // Force a live filesystem scan, ignoring any baked manifest at the root. `Build` delegates here
