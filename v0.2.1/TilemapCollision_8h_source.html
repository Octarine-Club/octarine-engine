26[[nodiscard]] constexpr bool MatchesCollisionMask(const uint8_t tileValue, const uint8_t mask) noexcept {
27 if (tileValue == kTileEmpty || tileValue >= 8u) {
28 return false;
29 }
30 return (mask & static_cast<uint8_t>(1u << tileValue)) != 0u;
31}

46inline DdaState InitDda(const float x0, const float y0, const float dirX, const float dirY, const int32_t tileSize) {
47 DdaState s;
48 const auto ts = static_cast<float>(tileSize);
49 s.tileX = WorldToGlobalTile(x0, tileSize);
50 s.tileY = WorldToGlobalTile(y0, tileSize);
51
52 constexpr float kInf = std::numeric_limits<float>::max();
53 if (std::abs(dirX) > 1e-6f) {
54 s.stepX = (dirX > 0.0f) ? 1 : -1;
55 const float nextBoundX = static_cast<float>(s.tileX + (dirX > 0.0f ? 1 : 0)) * ts;
56 s.tMaxX = (nextBoundX - x0) / dirX;
57 s.tDeltaX = ts / std::abs(dirX);
58 } else {
59 s.tMaxX = kInf;
60 s.tDeltaX = kInf;
61 }
62
63 if (std::abs(dirY) > 1e-6f) {
64 s.stepY = (dirY > 0.0f) ? 1 : -1;
65 const float nextBoundY = static_cast<float>(s.tileY + (dirY > 0.0f ? 1 : 0)) * ts;
66 s.tMaxY = (nextBoundY - y0) / dirY;
67 s.tDeltaY = ts / std::abs(dirY);
68 } else {
69 s.tMaxY = kInf;
70 s.tDeltaY = kInf;
71 }
72 return s;

75inline void ResolveCircleSingleTile(float& cx, float& cy, const float radius, const int32_t tx, const int32_t ty,
76 const float ts) noexcept {
77 const float minX = static_cast<float>(tx) * ts;
78 const float minY = static_cast<float>(ty) * ts;
79 const float maxX = minX + ts;
80 const float maxY = minY + ts;
81
82 const float closestX = std::clamp(cx, minX, maxX);
83 const float closestY = std::clamp(cy, minY, maxY);
84 const float dx = cx - closestX;
85 const float dy = cy - closestY;
86 const float distSq = dx * dx + dy * dy;
87
88 if (distSq >= radius * radius) {
89 return;
90 }
91
92 if (distSq > 1e-6f) {
93 const float dist = std::sqrt(distSq);
94 const float push = radius - dist;
95 cx += (dx / dist) * push;
96 cy += (dy / dist) * push;
97 return;
98 }
99
100 // Circle center is inside the solid tile AABB: push out along shortest axis to tile edge.
101 const float penLeft = (cx - minX) + radius;
102 const float penRight = (maxX - cx) + radius;
103 const float penTop = (cy - minY) + radius;
104 const float penBottom = (maxY - cy) + radius;
105 const float minPen = std::min({penLeft, penRight, penTop, penBottom});
106 if (minPen == penLeft) {
107 cx = minX - radius;
108 } else if (minPen == penRight) {
109 cx = maxX + radius;
110 } else if (minPen == penTop) {
111 cy = minY - radius;
112 } else {
113 cy = maxY + radius;
114 }
115}

117inline void ResolveAABBSingleTile(float& x, float& y, const float w, const float h, const int32_t tx, const int32_t ty,
118 const float ts) noexcept {
119 const float tileMinX = static_cast<float>(tx) * ts;
120 const float tileMinY = static_cast<float>(ty) * ts;
121 const float tileMaxX = tileMinX + ts;
122 const float tileMaxY = tileMinY + ts;
123
124 const float overlapX = std::min(x + w, tileMaxX) - std::max(x, tileMinX);
125 const float overlapY = std::min(y + h, tileMaxY) - std::max(y, tileMinY);
126 if (overlapX <= 0.0f || overlapY <= 0.0f) {
127 return;
128 }
129
130 const float centerX = x + w * 0.5f;
131 const float centerY = y + h * 0.5f;
132 const float tileCenterX = tileMinX + ts * 0.5f;
133 const float tileCenterY = tileMinY + ts * 0.5f;
134
135 if (overlapX < overlapY) {
136 x += (centerX < tileCenterX) ? -overlapX : overlapX;
137 } else {
138 y += (centerY < tileCenterY) ? -overlapY : overlapY;
139 }
140}

142template <typename TileGetterFn>
143inline void ResolveCirclePass(float& worldX, float& worldY, const float radius, const float ts, const int32_t minTx,
144 const int32_t maxTx, const int32_t minTy, const int32_t maxTy, const int32_t centerTx,
145 const int32_t centerTy, const bool cardinalOnly, const uint8_t collisionMask,
146 TileGetterFn& getTileByGlobal) {
147 for (int32_t ty = minTy; ty <= maxTy; ++ty) {
148 for (int32_t tx = minTx; tx <= maxTx; ++tx) {
149 const bool isCardinal = (tx == centerTx || ty == centerTy);
150 if (isCardinal != cardinalOnly) continue;
151 if (MatchesCollisionMask(getTileByGlobal(tx, ty), collisionMask)) {
152 ResolveCircleSingleTile(worldX, worldY, radius, tx, ty, ts);
153 }
154 }
155 }

161template <typename TileGetterFn>
162[[nodiscard]] TileRaycastHit RaycastTilemap(const float x0, const float y0, const float x1, const float y1,
163 const int32_t tileSize, const uint8_t collisionMask,
164 TileGetterFn&& getTileByGlobal) {
165 TileRaycastHit result{};
166 result.hitX = x1;
167 result.hitY = y1;
168 if (tileSize <= 0) {
169 return result;
170 }
171
172 const float dx = x1 - x0;
173 const float dy = y1 - y0;
174 const float maxDist = std::sqrt(dx * dx + dy * dy);
175 result.distance = maxDist;
176
177 int32_t curTx = WorldToGlobalTile(x0, tileSize);
178 int32_t curTy = WorldToGlobalTile(y0, tileSize);
179 const uint8_t startVal = getTileByGlobal(curTx, curTy);
180 if (MatchesCollisionMask(startVal, collisionMask)) {
181 result.hit = true;
182 result.hitX = x0;
183 result.hitY = y0;
184 result.distance = 0.0f;
185 result.tileValue = startVal;
186 result.tileX = curTx;
187 result.tileY = curTy;
188 return result;
189 }
190
191 if (maxDist <= 1e-5f) {
192 return result;
193 }
194
195 const float dirX = dx / maxDist;
196 const float dirY = dy / maxDist;
197 detail::DdaState dda = detail::InitDda(x0, y0, dirX, dirY, tileSize);
198
199 constexpr int kMaxSteps = 4096;
200 for (int step = 0; step < kMaxSteps; ++step) {
201 float tEnter = 0.0f;
202 float normX = 0.0f;
203 float normY = 0.0f;
204 if (dda.tMaxX < dda.tMaxY) {
205 tEnter = dda.tMaxX;
206 dda.tMaxX += dda.tDeltaX;
207 dda.tileX += dda.stepX;
208 normX = static_cast<float>(-dda.stepX);
209 } else {
210 tEnter = dda.tMaxY;
211 dda.tMaxY += dda.tDeltaY;
212 dda.tileY += dda.stepY;
213 normY = static_cast<float>(-dda.stepY);
214 }
215
216 if (tEnter > maxDist) {
217 break;
218 }
219
220 const uint8_t val = getTileByGlobal(dda.tileX, dda.tileY);
221 if (MatchesCollisionMask(val, collisionMask)) {
222 result.hit = true;
223 result.distance = std::max(0.0f, tEnter);
224 result.hitX = x0 + dirX * result.distance;
225 result.hitY = y0 + dirY * result.distance;
226 result.normalX = normX;
227 result.normalY = normY;
228 result.tileValue = val;
229 result.tileX = dda.tileX;
230 result.tileY = dda.tileY;
231 return result;
232 }
233 }
234 return result;
235}
