195// Scrambles a compiled Lua 5.4 binary chunk: swaps magic \x1bLua -> \x1bOCT,
196// permutes instructions using a 64-bit seed bijection, and masks string constants.
197inline bool ScrambleLua54Bytecode(uint8_t* data, size_t size, uint64_t seed = 0x4f63746172696e65ULL) {
198 if (data == nullptr || size < 34) return false;
199 static constexpr uint8_t kLuaMagic[] = {0x1B, 'L', 'u', 'a', 0x54};
200 if (std::memcmp(data, kLuaMagic, 5) != 0) {
201 return false;
202 }
203
204 std::array<uint8_t, 83> sbox{};
205 std::array<uint8_t, 83> invBox{};
206 GenerateOpcodeSBox(seed, sbox, invBox);
207
208 detail::BytecodeCursor cursor(data, size);
209 cursor.Skip(31); // 31-byte Lua 5.4 header
210 cursor.Skip(1); // top-level sizeupvalues byte
211 cursor.ProcessRootFunction(sbox, seed);
212
213 if (!cursor.Ok()) {
214 return false;
215 }
216
217 // Mutate header magic to \x1bOCT
218 data[1] = 'O';
219 data[2] = 'C';
220 data[3] = 'T';
221 return true;
222}
