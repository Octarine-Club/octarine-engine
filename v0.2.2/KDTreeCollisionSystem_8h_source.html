58[[nodiscard]] inline float PointSegmentDistanceSq(const glm::vec2 p, const glm::vec2 a, const glm::vec2 b) {
59 const glm::vec2 ab = b - a;
60 const float lenSq = glm::dot(ab, ab);
61 float t = 0.0f;
62 if (lenSq > 1e-12f) {
63 t = std::clamp(glm::dot(p - a, ab) / lenSq, 0.0f, 1.0f);
64 }
65 const glm::vec2 diff = p - (a + ab * t);
66 return glm::dot(diff, diff);
67}

73[[nodiscard]] inline bool SegmentsStrictlyCross(const glm::vec2 a1, const glm::vec2 a2, const glm::vec2 b1,
74 const glm::vec2 b2) {
75 const glm::vec2 da = a2 - a1;
76 const glm::vec2 db = b2 - b1;
77 const float d1 = Cross2D(db, a1 - b1);
78 const float d2 = Cross2D(db, a2 - b1);
79 const float d3 = Cross2D(da, b1 - a1);
80 const float d4 = Cross2D(da, b2 - a1);
81 return OppositeSigns(d1, d2) && OppositeSigns(d3, d4);

110[[nodiscard]] inline bool PointInTriangle(const glm::vec2 p, const glm::vec2 v0, const glm::vec2 v1,
111 const glm::vec2 v2) {
112 const float d1 = Cross2D(v1 - v0, p - v0);
113 const float d2 = Cross2D(v2 - v1, p - v1);
114 const float d3 = Cross2D(v0 - v2, p - v2);

198 // Shape narrowphase payload:
199 // - Box: (cx, cy) = center, (hx, hy) = half-extents, rot = orientation
200 // - Circle: (cx, cy) = center, (hx, hy) = center, rot.cos = radius
201 // - Capsule: (cx, cy) = start, (hx, hy) = end, rot.cos = radius
202 // - Line: (cx, cy) = start, (hx, hy) = end, rot.cos = 0
203 // - Triangle: (cx, cy) = v0, (hx, hy) = v1, (rot.cos, rot.sin) = v2
204 float cx{0.0f};
205 float cy{0.0f};
206 float hx{0.0f};
207 float hy{0.0f};

215 [[nodiscard]] bool intersectsInDimension(const Box& other, const int dim) const {
216 if (dim == 0) return !(maxX < other.minX || minX > other.maxX);

221 // SAT on the 4 face normals of two OBBs. Only invoked when at least one box is rotated;
222 // axis-aligned pairs short-circuit on the AABB check above.
223 [[nodiscard]] bool obbIntersects(const Box& other) const {
224 const float ax0 = rot.cos, ay0 = rot.sin;
225 const float ax1 = -rot.sin, ay1 = rot.cos;
226 const float bx0 = other.rot.cos, by0 = other.rot.sin;
227 const float bx1 = -other.rot.sin, by1 = other.rot.cos;
228 const float dx = other.cx - cx;
229 const float dy = other.cy - cy;
230
231 const float axes[4][2] = {{ax0, ay0}, {ax1, ay1}, {bx0, by0}, {bx1, by1}};
232 return std::ranges::all_of(axes, [&](const auto& axis) {
233 const float ux = axis[0];
234 const float uy = axis[1];
235 const float aProj = hx * std::abs(ax0 * ux + ay0 * uy) + hy * std::abs(ax1 * ux + ay1 * uy);
236 const float bProj = other.hx * std::abs(bx0 * ux + by0 * uy) + other.hy * std::abs(bx1 * ux + by1 * uy);
237 const float distProj = std::abs(dx * ux + dy * uy);
238 return distProj <= aProj + bProj;
239 });
240 }

242 [[nodiscard]] static bool CapsuleObbIntersects(const Box& cap, const Box& obb) {
243 const octarine::Rotation2D invRot{obb.rot.cos, -obb.rot.sin};
244 const glm::vec2 obbCenter{obb.cx, obb.cy};
245 const glm::vec2 localP1 = octarine::Rotate(glm::vec2{cap.cx, cap.cy} - obbCenter, invRot);
246 const float rSq = cap.rot.cos * cap.rot.cos;
247 if (cap.shape == ColliderShape::Circle) {
248 return octarine::collision::PointAabbDistanceSq(localP1, obb.hx, obb.hy) <= rSq;
249 }
250 const glm::vec2 localP2 = octarine::Rotate(glm::vec2{cap.hx, cap.hy} - obbCenter, invRot);
251 return octarine::collision::SegmentAabbDistanceSq(localP1, localP2, obb.hx, obb.hy) <= rSq;
252 }

260 [[nodiscard]] static bool CapsulePairIntersects(const Box& a, const Box& b) {
261 const float rSum = a.rot.cos + b.rot.cos;
262 if (a.shape == ColliderShape::Circle && b.shape == ColliderShape::Circle) {
263 const float dx = b.cx - a.cx;

267 return octarine::collision::SegmentSegmentDistanceSq({a.cx, a.cy}, {a.hx, a.hy}, {b.cx, b.cy}, {b.hx, b.hy}) <=
268 (rSum * rSum);
269 }

283 [[nodiscard]] bool
284 narrowphaseIntersects(const Box& other) const {
285 if (shape == ColliderShape::Box && other.shape == ColliderShape::Box) {
286 return obbIntersects(other);
287 }
288 if (IsCapsuleLike() && other.IsCapsuleLike()) {
289 return CapsulePairIntersects(*this, other);
290 }
291 if (IsCapsuleLike() && other.shape == ColliderShape::Box) return CapsuleObbIntersects(*this, other);
292 if (shape == ColliderShape::Box && other.IsCapsuleLike()) return CapsuleObbIntersects(other, *this);
293 if (IsCapsuleLike() && other.shape == ColliderShape::Triangle) return CapsuleTriangleIntersects(*this, other);
294 if (shape == ColliderShape::Triangle && other.IsCapsuleLike()) return CapsuleTriangleIntersects(other, *this);
295 if (shape == ColliderShape::Box || other.shape == ColliderShape::Box) {
296 return BoxTrianglePairIntersects(*this, other);
297 }
298 return octarine::collision::TrianglesIntersect({cx, cy}, {hx, hy}, {rot.cos, rot.sin}, {other.cx, other.cy},
299 {other.hx, other.hy}, {other.rot.cos, other.rot.sin});

302 [[nodiscard]] bool containsPoint(const glm::vec2 pt) const {
303 if (pt.x < minX || pt.x > maxX || pt.y < minY || pt.y > maxY) return false;
304 if (shape == ColliderShape::Box) {
305 if (rot.IsIdentity()) return true;
306 const glm::vec2 local = octarine::Rotate(pt - glm::vec2{cx, cy}, {rot.cos, -rot.sin});
307 return std::abs(local.x) <= hx && std::abs(local.y) <= hy;
308 }
309 if (IsCapsuleLike()) {
310 return octarine::collision::PointSegmentDistanceSq(pt, {cx, cy}, {hx, hy}) <= (rot.cos * rot.cos);
311 }
312 return octarine::collision::PointInTriangle(pt, {cx, cy}, {hx, hy}, {rot.cos, rot.sin});
313 }

549 std::vector<Box>& boxes = cachedBoxes_;
550 boxes.clear();
551 boxes.reserve(TotalColliderCount());

555 AppendBoxesSync(*triangleQuery_, boxes);

558 return a.entity.id != b.entity.id ? (a.entity.id < b.entity.id) : (a.shape < b.shape);
559 });

564 if (!boxes.empty()) {

567 if (a.first.id != b.first.id) return a.first.id < b.first.id;

613 // Returns true if entity a and entity b are currently overlapping (sustained OR just-entered).
614 // Reflects the result of the most recently completed async detection pass (one frame of lag
615 // on the very first frame, stable thereafter). Safe to call from on_update or on_collision.
616 [[nodiscard]] bool IsOverlapping(const Entity a, const Entity b) const {
617 return prevPairSet_.contains({std::min(a.id, b.id), std::max(a.id, b.id)});
618 }

621 struct PairHash {
622 size_t operator()(const std::pair<EntityID, EntityID>& p) const noexcept {
623 // Boost-style hash_combine: golden-ratio constant spreads entropy, shifts mix bits.
624 static constexpr size_t kGoldenRatio = 0x9e3779b9ULL;
625 static constexpr int kLeftShift = 6;
626 size_t h = std::hash<EntityID>{}(p.first);
627 h ^= std::hash<EntityID>{}(p.second) + kGoldenRatio + (h << kLeftShift) + (h >> 2);
628 return h;
629 }
630 };
631 using PairSet = std::unordered_set<std::pair<EntityID, EntityID>, PairHash>;
632
633 PairSet prevPairSet_;
634 std::vector<Box> cachedBoxes_;
635 std::future<CollisionResult> collisionResult_;
636 std::unique_ptr<ColliderQuery<BoxColliderComponent>> query_;
637 std::unique_ptr<ColliderQuery<CircleColliderComponent>> circleQuery_;
638 std::unique_ptr<ColliderQuery<CapsuleColliderComponent>> capsuleQuery_;
639 std::unique_ptr<ColliderQuery<TriangleColliderComponent>> triangleQuery_;
640 std::unique_ptr<ColliderQuery<LineColliderComponent>> lineQuery_;
641 bool synchronous_{false};
642
643 // Diff this frame's overlaps against the previous frame and emit the enter/exit batches.
644 // - enter: pairs overlapping now but not last frame — first-contact only, so persistent
645 // overlaps (enemy pinned against a wall, stacked entities) don't re-fire every frame.
646 // - exit: pairs that overlapped last frame but no longer do.
647 void EmitCollisionEvents(EventBus* eventBus, const std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
648 PROFILE_COUNTER_SET("Collision: Intersecting pairs", static_cast<long long>(intersectingPairs.size()));
649
650 PairSet currentSet;
651 currentSet.reserve(intersectingPairs.size());
652 std::vector<std::pair<Entity, Entity>> enteringPairs;
653 enteringPairs.reserve(intersectingPairs.size());
654 for (const auto& [a, b] : intersectingPairs) {
655 const auto key = std::make_pair(std::min(a.id, b.id), std::max(a.id, b.id));
656 if (currentSet.insert(key).second && !prevPairSet_.contains(key)) {
657 enteringPairs.emplace_back(a, b);
658 }
659 }
660
661 std::vector<std::pair<Entity, Entity>> exitingPairs;
662 for (const auto& [minId, maxId] : prevPairSet_) {
663 if (!currentSet.contains({minId, maxId})) {
664 exitingPairs.emplace_back(Entity{minId}, Entity{maxId});
665 }
666 }
667
668 prevPairSet_ = std::move(currentSet);
669
670 PROFILE_COUNTER_SET("Collision: Entering pairs", static_cast<long long>(enteringPairs.size()));
671 eventBus->EmitEvent<CollisionBatchEvent>(enteringPairs);
672 eventBus->EmitEvent<CollisionExitBatchEvent>(exitingPairs);
673 }

675 static void BruteForceHugeBoxes(const std::vector<Box>& boxes, const std::vector<Box>::iterator& hugeBoxesBegin,
676 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
677 ACCUMULATE_PROFILE_SCOPE("Huge Box Brute Force");
678 for (auto hugeIt = hugeBoxesBegin; hugeIt != boxes.end(); ++hugeIt) {
679 for (auto normalIt = boxes.begin(); normalIt != hugeBoxesBegin; ++normalIt) {
680 if (hugeIt->intersects(*normalIt)) {
681 intersectingPairs.emplace_back(hugeIt->entity, normalIt->entity);
682 }
683 }
684 for (auto otherHugeIt = hugeIt + 1; otherHugeIt != boxes.end(); ++otherHugeIt) {
685 if (hugeIt->intersects(*otherHugeIt)) {
686 intersectingPairs.emplace_back(hugeIt->entity, otherHugeIt->entity);
687 }
688 }
689 }
690 }

692 static void ComputeAsyncCollisions(std::vector<Box>& boxes,
693 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
694 AGGREGATE_PROFILE_SESSION("Async Box Creation");
695 if (boxes.empty()) return;
696
697 ACCUMULATE_PROFILE_SCOPE("Huge Box Extraction");
698
699 // Dynamically calculate the huge box threshold to be scale-invariant.
700 // Defines a "huge outlier" as a box >= 4x the average size.
701 double sum = 0.0;
702 for (const auto& b : boxes) {
703 sum += std::max(b.maxX - b.minX, b.maxY - b.minY);
704 }
705 const double mean = sum / static_cast<double>(boxes.size());
706 const auto hugeThreshold = static_cast<float>(mean * 4.0);
707 // Extract any box larger than the dynamic threshold. These are typically background bounds or map
708 // triggers that cross the entire level. Putting them in the KD-Tree destroys the median
709 // splits and breaks the Sweep Bipartite early-outs.
710 const auto hugeBoxesBegin = std::ranges::partition(boxes, [hugeThreshold](const Box& b) {
711 return (b.maxX - b.minX) <= hugeThreshold && (b.maxY - b.minY) <= hugeThreshold;
712 }).begin();
713
714 const int normalCount = static_cast<int>(std::distance(boxes.begin(), hugeBoxesBegin));
715 if (normalCount > 0) {
716 FindIntersectionsRecursive(boxes, 0, normalCount, 0, 0, intersectingPairs);
717 }
718
719 if (hugeBoxesBegin != boxes.end()) {
720 BruteForceHugeBoxes(boxes, hugeBoxesBegin, intersectingPairs);
721 }
722 }

724 [[nodiscard]] static std::future<CollisionResult> StartAsyncCollisionDetection(std::vector<Box> boxes) {
725 // Run on the persistent worker pool rather than std::async(launch::async), which spawns and
726 // tears down an OS thread per cycle. The promise is shared because ThreadPool::Submit takes a
727 // copyable std::function; the future hands back to the same valid()/wait_for/get polling below.
728 auto promise = std::make_shared<std::promise<CollisionResult>>();
729 std::future<CollisionResult> result = promise->get_future();
730 ThreadPool::Instance().Submit([boxes = std::move(boxes), promise]() mutable {
731 std::vector<std::pair<Entity, Entity>> intersectingPairs;
732 ComputeAsyncCollisions(boxes, intersectingPairs);
733 promise->set_value(CollisionResult{std::move(intersectingPairs), std::move(boxes)});
734 });
735 return result;
736 }

738 static void FindIntersectionsBruteForce(const std::vector<Box>& boxes, const int begin, const int end,
739 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
740 ACCUMULATE_PROFILE_SCOPE("Brute Force Intersection");
741
742 uint32_t entityUnion = 0;
743 uint32_t collUnion = 0;
744 for (int i = begin; i < end; ++i) {
745 entityUnion |= boxes[static_cast<size_t>(i)].entityMask;
746 collUnion |= boxes[static_cast<size_t>(i)].collisionMask;
747 }
748 if ((entityUnion & collUnion) == 0U) {
749 return;
750 }
751
752 int checks = 0;
753 for (int i = begin; i < end; ++i) {
754 for (int j = i + 1; j < end; ++j) {
755 checks++;
756 const auto& bi = boxes[static_cast<size_t>(i)];
757 const auto& bj = boxes[static_cast<size_t>(j)];
758 if (bi.intersects(bj)) {
759 intersectingPairs.emplace_back(bi.entity, bj.entity);
760 }
761 }
762 }
763#ifdef OCTARINE_PROFILING
764 PROFILE_COUNTER_ADD("Collision: Brute Force Checks", checks);
765#else
766 (void)checks;
767#endif
768 }

771 const int dimension, const float medianValue) {
772 ACCUMULATE_PROFILE_SCOPE("Partition Boxes");
773 const auto first = boxes.begin() + begin;
774 const auto last = boxes.begin() + end;
775
776 const auto leftEndIt = std::partition(first, last, [&](const Box& box) {
777 if (dimension == 0) {
778 return box.maxX < medianValue;
779 }
780
781 return box.maxY < medianValue;
782 });
783
784 const auto rightBeginIt = std::partition(leftEndIt, last, [&](const Box& box) {
785 if (dimension == 0) {
786 return box.minX <= medianValue;
787 }
788
789 return box.minY <= medianValue;
790 });
791
792 const int leftEnd = static_cast<int>(std::distance(boxes.begin(), leftEndIt));
793
794 const int rightStart = static_cast<int>(std::distance(boxes.begin(), rightBeginIt));
795
796 return Partitions{leftEnd, rightStart};

821 // KD-Tree inherently requires recursion and dense Bipartite bounds-checking logic.
822 // NOLINTNEXTLINE(misc-no-recursion,readability-function-cognitive-complexity)
823 static void FindIntersectionsSweepBipartite(std::vector<Box>& boxes, const int begin1, const int end1,
824 const int begin2, const int end2, const int dimension,
825 std::vector<std::pair<Entity, Entity>>& pairs) {
826 ACCUMULATE_PROFILE_SCOPE("Sweep Bipartite");
827
828 if (end1 - begin1 == 0 || end2 - begin2 == 0) return;
829
830 // Aggregate Mask Pruning: Skips bipartite sweep entirely if the union of entity and collision masks proves no
831 // interaction is possible.
832 uint32_t entity1 = 0;
833 uint32_t coll1 = 0;
834 for (int i = begin1; i < end1; ++i) {
835 entity1 |= boxes[static_cast<size_t>(i)].entityMask;
836 coll1 |= boxes[static_cast<size_t>(i)].collisionMask;
837 }
838
839 uint32_t entity2 = 0;
840 uint32_t coll2 = 0;
841 for (int i = begin2; i < end2; ++i) {
842 entity2 |= boxes[static_cast<size_t>(i)].entityMask;
843 coll2 |= boxes[static_cast<size_t>(i)].collisionMask;
844 }
845
846 const bool groupCanInteract = ((coll1 & entity2) | (coll2 & entity1)) != 0U;
847 if (!groupCanInteract) {
848 return;
849 }
850
851 if ((end1 - begin1) * (end2 - begin2) <= kBruteforceCutoff * kBruteforceCutoff) {
852 FindIntersectionsBruteForceBipartite(boxes, begin1, end1, begin2, end2, pairs);
853 return;
854 }
855
856 {
857 ACCUMULATE_PROFILE_SCOPE("Bipartite Sort Phase");
858 if (dimension == 0) {
859 std::sort(boxes.begin() + begin1, boxes.begin() + end1, [](const Box& a, const Box& b) {
860 const bool aFin = IsFiniteFloat(a.minX);
861 const bool bFin = IsFiniteFloat(b.minX);
862 if (!aFin || !bFin) return aFin && !bFin;
863 return a.minX < b.minX;
864 });
865 std::sort(boxes.begin() + begin2, boxes.begin() + end2, [](const Box& a, const Box& b) {
866 const bool aFin = IsFiniteFloat(a.minX);
867 const bool bFin = IsFiniteFloat(b.minX);
868 if (!aFin || !bFin) return aFin && !bFin;
869 return a.minX < b.minX;
870 });
871 } else {
872 std::sort(boxes.begin() + begin1, boxes.begin() + end1, [](const Box& a, const Box& b) {
873 const bool aFin = IsFiniteFloat(a.minY);
874 const bool bFin = IsFiniteFloat(b.minY);
875 if (!aFin || !bFin) return aFin && !bFin;
876 return a.minY < b.minY;
877 });
878 std::sort(boxes.begin() + begin2, boxes.begin() + end2, [](const Box& a, const Box& b) {
879 const bool aFin = IsFiniteFloat(a.minY);
880 const bool bFin = IsFiniteFloat(b.minY);
881 if (!aFin || !bFin) return aFin && !bFin;
882 return a.minY < b.minY;
883 });
884 }
885 }
886
887 {
888 ACCUMULATE_PROFILE_SCOPE("Bipartite Sweep Phase");
889 int sweepChecks = 0;
890 if (dimension == 0) {
891 int startJ = begin2;
892 for (int i = begin1; i < end1; ++i) {
893 const Box& a = boxes[static_cast<size_t>(i)];
894 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxX < a.minX) ++startJ;
895 for (int j = startJ; j < end2; ++j) {
896 sweepChecks++;
897 const Box& b = boxes[static_cast<size_t>(j)];
898 if (b.minX > a.maxX) break;
899 if (b.maxY < a.minY || b.minY > a.maxY) continue;
900 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
901 }
902 }
903 } else {
904 int startJ = begin2;
905 for (int i = begin1; i < end1; ++i) {
906 const Box& a = boxes[static_cast<size_t>(i)];
907 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxY < a.minY) ++startJ;
908 for (int j = startJ; j < end2; ++j) {
909 sweepChecks++;
910 const Box& b = boxes[static_cast<size_t>(j)];
911 if (b.minY > a.maxY) break;
912 if (b.maxX < a.minX || b.minX > a.maxX) continue;
913 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
914 }
915 }
916 }
917#ifdef OCTARINE_PROFILING
918 PROFILE_COUNTER_ADD("Collision: Bipartite Sweep Inner Loop Checks", sweepChecks);
919#else
920 (void)sweepChecks;
921#endif
922 }
923 }

925 static constexpr float kHalf = 0.5f;
926
927 // NOLINTNEXTLINE(misc-no-recursion) - KD-Tree cross-checking inherently requires recursion.
928 static void ProcessSpanningBipartite(std::vector<Box>& boxes, const int begin, const int end, const int leftEnd,
929 const int rightStart, const int dimension, const int nextDimension,
930 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
931 if (leftEnd >= rightStart) return;
932
933 // Find the min/max bounds of the Spanning set along the current dimension
934 float spanMin = std::numeric_limits<float>::max();
935 float spanMax = std::numeric_limits<float>::lowest();
936 for (int i = leftEnd; i < rightStart; ++i) {
937 const float minV = dimension == 0 ? boxes[static_cast<size_t>(i)].minX : boxes[static_cast<size_t>(i)].minY;
938 const float maxV = dimension == 0 ? boxes[static_cast<size_t>(i)].maxX : boxes[static_cast<size_t>(i)].maxY;
939 if (minV < spanMin) spanMin = minV;
940 if (maxV > spanMax) spanMax = maxV;
941 }
942
943 const auto activeLeftBegin = std::partition(boxes.begin() + begin, boxes.begin() + leftEnd, [=](const Box& a) {
944 const float maxV = dimension == 0 ? a.maxX : a.maxY;
945 return maxV < spanMin;
946 });
947 const int leftOverlapStart = static_cast<int>(std::distance(boxes.begin(), activeLeftBegin));
948
949 const auto activeRightEnd = std::partition(boxes.begin() + rightStart, boxes.begin() + end, [=](const Box& a) {
950 const float minV = dimension == 0 ? a.minX : a.minY;
951 return minV <= spanMax;
952 });
953 const int rightOverlapEnd = static_cast<int>(std::distance(boxes.begin(), activeRightEnd));
954
955 FindIntersectionsSweepBipartite(boxes, leftOverlapStart, leftEnd, leftEnd, rightStart, nextDimension,
956 intersectingPairs);
957 FindIntersectionsSweepBipartite(boxes, leftEnd, rightStart, rightStart, rightOverlapEnd, nextDimension,
958 intersectingPairs);
959 }
960 // KD-Tree inherently requires recursion.
961 // NOLINTNEXTLINE(misc-no-recursion)
962 static void FindIntersectionsRecursive(std::vector<Box>& boxes, const int begin, const int end, int dimension,
963 const int depth, std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
964 const int count = end - begin;
965 if (count <= 1) {
966 return;
967 }
968
969 if (count < kBruteforceCutoff || depth >= kMaxRecursionDepth) {
970 FindIntersectionsBruteForce(boxes, begin, end, intersectingPairs);
971 return;
972 }
973
974 const int mid = (begin + end) / 2;
975 float medianValue = 0.0f;
976 {
977 ACCUMULATE_PROFILE_SCOPE("Find Median");
978 std::nth_element(boxes.begin() + begin, boxes.begin() + mid, boxes.begin() + end,
979 [dimension](const Box& a, const Box& b) {
980 const float va = (dimension == 0) ? (a.maxX + a.minX) : (a.maxY + a.minY);
981 const float vb = (dimension == 0) ? (b.maxX + b.minX) : (b.maxY + b.minY);
982 const bool aFin = IsFiniteFloat(va);
983 const bool bFin = IsFiniteFloat(vb);
984 if (!aFin || !bFin) return aFin && !bFin;
985 return va < vb;
986 });
987 medianValue = (dimension == 0)
988 ? (boxes[static_cast<size_t>(mid)].minX + boxes[static_cast<size_t>(mid)].maxX) * kHalf
989 : (boxes[static_cast<size_t>(mid)].minY + boxes[static_cast<size_t>(mid)].maxY) * kHalf;
990 if (!IsFiniteFloat(medianValue)) {
991 FindIntersectionsBruteForce(boxes, begin, end, intersectingPairs);
992 return;
993 }
994 }
