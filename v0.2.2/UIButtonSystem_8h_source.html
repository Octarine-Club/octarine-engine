62 void OnMouseInput(const MouseInputEvent& event) {
63 const auto mousePos = GetTransformedMousePos(event);
64 if (!mousePos || !EnsureMouseQuery()) {
65 return;
66 }
67 lastMousePos_ = *mousePos;
68
69 const auto* cameraComp = registry_->TryGet<CameraComponent>();
70 const octarine::Rect camera = cameraComp ? cameraComp->viewport : octarine::Rect{};
71 const float zoom = (cameraComp && cameraComp->zoom > 0.001f) ? cameraComp->zoom : 1.0f;
72
73 if (event.event.type == SDL_EVENT_MOUSE_MOTION) {
74 UpdateMouseHover(camera, zoom, *mousePos);

78 ClearPressedState();

253 const glm::vec2 dir) noexcept {

261 bool NavigateDirection(const glm::vec2 dir) {
262 if (!EnsureFocusQuery()) return false;
263 const Entity current = GetFocusedEntity();
264 if (current.GetId() == static_cast<std::uint32_t>(-1)) {
265 return CycleFocusTab(false);
266 }
267
268 const Entity explicitNext = ResolveExplicitDirectionTarget(registry_->GetComponent<UIFocusComponent>(current), dir);
269 if (IsFocusableCandidate(explicitNext)) {
270 SetFocus(explicitNext);
271 return true;
272 }
273
274 if (!registry_->HasComponent<UIRectComponent>(current)) return false;
275 const glm::vec2 curCenter = registry_->GetComponent<UIRectComponent>(current).Center();
276
277 Entity best{static_cast<EcsId>(-1)};
278 float bestScore = std::numeric_limits<float>::max();
279 focusQuery_->ForEach([&](const Entity candidate, const UIFocusComponent& foc) {
280 if (candidate == current || !foc.isFocusable || foc.isDisabled || foc.isInert || !IsButtonVisible(candidate)) {
281 return;
282 }
283 if (!registry_->HasComponent<UIRectComponent>(candidate)) return;
284 const glm::vec2 delta = registry_->GetComponent<UIRectComponent>(candidate).Center() - curCenter;
285 const float proj = delta.x * dir.x + delta.y * dir.y;
286 if (proj <= 1.0f) return;
287 const float perp = std::abs(delta.x * dir.y - delta.y * dir.x);
288 const float score = proj + perp * 2.5f;
289 if (score < bestScore) {
290 bestScore = score;
291 best = candidate;
292 }
293 });
294
295 if (best.GetId() != static_cast<std::uint32_t>(-1)) {
296 SetFocus(best);
297 return true;
298 }
299 return false;
300 }

302 bool CycleFocusTab(const bool reverse) {
303 if (!EnsureFocusQuery()) return false;
304 std::vector<std::pair<std::int32_t, Entity>> candidates;
305 focusQuery_->ForEach([&](const Entity entity, const UIFocusComponent& foc) {
306 if (foc.isFocusable && !foc.isDisabled && !foc.isInert && IsButtonVisible(entity)) {
307 candidates.emplace_back(foc.tabIndex, entity);
308 }
309 });
310 if (candidates.empty()) return false;
311 std::stable_sort(candidates.begin(), candidates.end(),
312 [](const auto& a, const auto& b) { return a.first < b.first; });
313
314 const Entity current = GetFocusedEntity();
315 size_t nextIdx = 0;
316 for (size_t i = 0; i < candidates.size(); ++i) {
317 if (candidates[i].second == current) {
318 nextIdx = reverse ? ((i + candidates.size() - 1) % candidates.size()) : ((i + 1) % candidates.size());
319 break;
320 }
321 }
322 SetFocus(candidates[nextIdx].second);
323 return true;
324 }

367 [[nodiscard]] Entity FindTopmostHit(const octarine::Rect& camera, const float zoom, const glm::vec2 mousePos) {
368 const int blockingLayer = FindTopPointerBlockingLayer(mousePos);
369 Entity topEntity{static_cast<EcsId>(-1)};
370 int topLayer = std::numeric_limits<int>::min();
371 mouseQuery_->ForEach([&](const Entity entity, UIButtonComponent& button, const GlobalTransformComponent* transform,
372 const BoxColliderComponent* collider) {
373 if (!button.isActive || !IsButtonVisible(entity)) return;
374 const int layer = ComputeEntitySortLayer(entity);
375 if (layer < blockingLayer) return;
376 if (!HitTest(entity, button, transform, collider, camera, zoom, mousePos.x, mousePos.y)) return;
377 if (layer >= topLayer) {
378 topLayer = layer;
379 topEntity = entity;
380 }
381 });
382 return topEntity;
383 }

440 bool HandleFocusKeyNavigation(const SDL_Keycode keycode, const SDL_Keymod mod) {
441 if (!EnsureFocusQuery() || !HasAnyFocusable()) return false;
442 if (keycode == SDLK_TAB) return CycleFocusTab((mod & SDL_KMOD_SHIFT) != 0);
443 if (keycode == SDLK_UP) return NavigateDirection({0.0f, -1.0f});
444 if (keycode == SDLK_DOWN) return NavigateDirection({0.0f, 1.0f});
445 if (keycode == SDLK_LEFT) return NavigateDirection({-1.0f, 0.0f});
446 if (keycode == SDLK_RIGHT) return NavigateDirection({1.0f, 0.0f});
447 if (keycode == SDLK_RETURN || keycode == SDLK_SPACE) {
448 return TriggerFocusedButton(true, false);
449 }
450 return false;
451 }

510 template <typename Fn>
511 static bool ForEachCommaToken(const std::string& spec, Fn&& fn) {
512 if (spec.empty()) return false;
513 size_t start = 0;
514 while (start < spec.size()) {
515 size_t end = spec.find(',', start);
516 if (end == std::string::npos) end = spec.size();
517 std::string token = spec.substr(start, end - start);
518 const size_t first = token.find_first_not_of(" \t\r\n");
519 if (first != std::string::npos) {
520 const size_t last = token.find_last_not_of(" \t\r\n");
521 token = token.substr(first, last - first + 1);
522 if (fn(token)) return true;
523 }
524 start = end + 1;
525 }
526 return false;
527 }

529 static bool MatchesKeyString(const std::string& keySpec, const std::string& keyName, const InputSystem* inputSystem) {
530 return ForEachCommaToken(keySpec, [&](const std::string& token) {
531 if (inputSystem) {
532 return inputSystem->MatchesKeyAlias(token, keyName);
533 }
534 return InputSystem::MakeKey(token) == InputSystem::MakeKey(keyName);
535 });
536 }

563 [[nodiscard]] static bool ContainsPointInCircle(const octarine::ui::ResolvedCircle& c, const glm::vec2 pt) {
564 if (c.radiusX <= 0.0f || c.radiusY <= 0.0f) return false;
565 const float dx = (pt.x - c.center.x) / c.radiusX;
566 const float dy = (pt.y - c.center.y) / c.radiusY;
567 return (dx * dx + dy * dy) <= 1.0f;
568 }

594 const glm::vec2 mouse) {
595 if (box.width <= 0 && box.height <= 0) {

694 [[nodiscard]] bool HitTest(const Entity entity, const UIButtonComponent& button,
695 const GlobalTransformComponent* transform, const BoxColliderComponent* collider,
696 const octarine::Rect& camera, const float zoom, const float mouseX,
697 const float mouseY) const {
698 if (registry_->HasComponent<UIRectComponent>(entity)) {

700 }
701 if (!transform) return false;
702 const glm::vec2 worldMouse =
703 button.isFixed ? glm::vec2{mouseX, mouseY} : glm::vec2{mouseX / zoom + camera.x, mouseY / zoom + camera.y};

710 HitTestCollider<CapsuleColliderComponent>(entity, *transform, worldMouse) ||
711 HitTestCollider<TriangleColliderComponent>(entity, *transform, worldMouse);
712 }
713 return HitTestTransformPrimitive(entity, *transform, worldMouse);
714 }
