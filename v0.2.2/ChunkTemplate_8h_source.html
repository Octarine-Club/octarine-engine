140 [[nodiscard]] static bool IsOpenSocketTag(const std::string& s) noexcept {
141 if (s.empty() || s == "*") {
142 return true;
143 }
144 auto startsWithCaseInsensitive = [](const std::string& str, const char* prefix, const size_t len) noexcept {
145 if (str.size() < len) return false;
146 for (size_t i = 0; i < len; ++i) {
147 const char c = (str[i] >= 'A' && str[i] <= 'Z') ? static_cast<char>(str[i] + ('a' - 'A')) : str[i];
148 if (c != prefix[i]) return false;
149 }
150 return true;
151 };
152 return startsWithCaseInsensitive(s, "open", 4) || startsWithCaseInsensitive(s, "none", 4) ||
153 startsWithCaseInsensitive(s, "any", 3);
154 }

156 // Returns true when all four edge sockets are unset or tagged with an unconstrained prefix ("open*", "none*",
157 // "any*", "*"), making the template safe for neighbor-independent procedural hash placement.
158 [[nodiscard]] bool IsUnconstrainedSocketTemplate() const noexcept {
159 return IsOpenSocketTag(edgeNorth) && IsOpenSocketTag(edgeEast) && IsOpenSocketTag(edgeSouth) &&

163 [[nodiscard]] constexpr uint8_t GetCollisionLocal(const int32_t localTx, const int32_t localTy) const noexcept {
164 if (localTx < 0 || localTx >= gridWidth || localTy < 0 || localTy >= gridHeight) {
165 return kTileEmpty;
166 }
167 const auto idx = static_cast<size_t>(localTy) * static_cast<size_t>(gridWidth) + static_cast<size_t>(localTx);
168 return idx < kMaxChunkArea ? collisions[idx] : static_cast<uint8_t>(kTileEmpty);
169 }
