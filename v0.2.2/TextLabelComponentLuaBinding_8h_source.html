64 static void ApplyLuaOutlineTable(const sol::table& t, TextLabelComponent& lbl) {
65 using namespace LuaComponentHelpers;
66 bool explicitOutlineToggle = false;
67 bool outlineEnabled = false;
68 if (t["outline"].is<bool>()) {
69 explicitOutlineToggle = true;
70 outlineEnabled = t["outline"].get<bool>();
71 lbl.outlineWidth = outlineEnabled ? 1.0f : 0.0f;
72 } else if (t["outline"].is<float>()) {
73 lbl.outlineWidth = std::max(0.0f, t["outline"].get<float>());
74 } else if (t["outline"].is<sol::table>()) {
75 lbl.outlineColor = SafeGetColor(t, "outline", 0, 0, 0, 255);
76 if (lbl.outlineColor.a > 0) lbl.outlineWidth = 1.0f;
77 }
78 for (const char* widthKey : {"outline_width", "outline_thickness"}) {
79 if (t[widthKey].is<float>()) {
80 lbl.outlineWidth = std::max(0.0f, t[widthKey].get<float>());
81 break;
82 }
83 }
84 if (t["outline_color"].is<sol::table>()) {
85 lbl.outlineColor = SafeGetColor(t, "outline_color", 0, 0, 0, 255);
86 if (lbl.outlineWidth <= 0.0f && (!explicitOutlineToggle || outlineEnabled) && lbl.outlineColor.a > 0) {
87 lbl.outlineWidth = 1.0f;
88 }
89 }
