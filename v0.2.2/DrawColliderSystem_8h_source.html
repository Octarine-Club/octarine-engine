96 static void DrawBox(SDL_Renderer* renderer, const CameraComponent& cameraComp,
97 const GlobalTransformComponent& transform, const BoxColliderComponent& collider) {
98 const auto& camera = cameraComp.viewport;
99 const float zoom = EffectiveZoom(cameraComp);
100
101 float w = static_cast<float>(collider.width) * std::abs(transform.scale.x);
102 float h = static_cast<float>(collider.height) * std::abs(transform.scale.y);
103 float hx = w * 0.5f;
104 float hy = h * 0.5f;
105
106 glm::vec2 centre = {transform.position.x + collider.offset.x * transform.scale.x + hx,
107 transform.position.y + collider.offset.y * transform.scale.y + hy};
108 glm::vec2 origin = transform.position + transform.pivot;
109
110 if (!collider.isFixed) {
111 const glm::vec2 cameraOffset = {camera.x, camera.y};
112 centre = (centre - cameraOffset) * zoom;
113 origin = (origin - cameraOffset) * zoom;
114 hx *= zoom;
115 hy *= zoom;
116 w *= zoom;
117 h *= zoom;
118 }
119
120 SDL_SetRenderDrawColor(renderer, Constants::kUint8Max, 0, 0, Constants::kUint8Max);
121
122 const octarine::Rotation2D rot = octarine::Rotation2D::FromRadians(transform.rotation);
123 if (rot.IsIdentity()) {
124 const SDL_FRect rect = {centre.x - hx, centre.y - hy, w, h};
125 SDL_RenderRect(renderer, &rect);
126 return;
127 }
128
129 centre = octarine::RotateAround(centre, origin, rot);
130 DrawOrientedBox(renderer, centre, {hx, hy}, rot);
131 }

148 static void DrawCapsule(SDL_Renderer* renderer, const CameraComponent& cameraComp,
149 const GlobalTransformComponent& transform, const CapsuleColliderComponent& collider) {
150 const float zoom = EffectiveZoom(cameraComp);
151 const octarine::Rotation2D rot = octarine::Rotation2D::FromRadians(transform.rotation);
152 const glm::vec2 p1World = TransformLocalPoint(collider.start, transform, rot);
153 const glm::vec2 p2World = TransformLocalPoint(collider.end, transform, rot);
154 const float scaleMax = std::max(std::abs(transform.scale.x), std::abs(transform.scale.y));
155 const float worldR = std::max(0.0f, collider.radius) * scaleMax;
156
157 const glm::vec2 p1 = ToScreenPoint(p1World, cameraComp.viewport, zoom, collider.isFixed);
158 const glm::vec2 p2 = ToScreenPoint(p2World, cameraComp.viewport, zoom, collider.isFixed);
159 const float r = ToScreenLength(worldR, zoom, collider.isFixed);
160
161 SDL_SetRenderDrawColor(renderer, Constants::kUint8Max, 0, 0, Constants::kUint8Max);
162 const glm::vec2 spine = p2 - p1;
163 const float len = glm::length(spine);
164 if (len <= 1e-4f) {
165 DrawCircleOutline(renderer, p1, r);
166 return;
167 }
168 DrawCapsuleOutline(renderer, p1, p2, spine / len, r);
169 }

200 static void DrawOrientedBox(SDL_Renderer* renderer, const glm::vec2 centre, const glm::vec2 half,
201 const octarine::Rotation2D rot) {
202 const glm::vec2 offsets[kBoxCorners] = {{-half.x, -half.y}, {half.x, -half.y}, {half.x, half.y}, {-half.x, half.y}};
203 SDL_FPoint corners[kOutlinePoints];
204 for (int i = 0; i < kBoxCorners; ++i) {
205 const glm::vec2 p = centre + octarine::Rotate(offsets[i], rot);
206 corners[i] = {p.x, p.y};
207 }
208 corners[kBoxCorners] = corners[0];
209 SDL_RenderLines(renderer, corners, kOutlinePoints);
210 }

212 static void DrawCircleOutline(SDL_Renderer* renderer, const glm::vec2 centre, const float r) {
213 SDL_FPoint pts[kCircleSegments + 1];
214 const float step = 2.0f * std::numbers::pi_v<float> / static_cast<float>(kCircleSegments);
215 for (int i = 0; i < kCircleSegments; ++i) {
216 const float angle = static_cast<float>(i) * step;
217 pts[i] = {centre.x + r * std::cos(angle), centre.y + r * std::sin(angle)};
218 }
219 pts[kCircleSegments] = pts[0];
220 SDL_RenderLines(renderer, pts, kCircleSegments + 1);
221 }

223 static void DrawCapsuleOutline(SDL_Renderer* renderer, const glm::vec2 p1, const glm::vec2 p2, const glm::vec2 dir,
224 const float r) {
225 const glm::vec2 perp{-dir.y, dir.x};
226 constexpr int kTotalPts = (kCapArcSegments + 1) * 2 + 1;
227 SDL_FPoint pts[kTotalPts];
228 int idx = 0;
229 const float step = std::numbers::pi_v<float> / static_cast<float>(kCapArcSegments);
230 for (int i = 0; i <= kCapArcSegments; ++i) {
231 const float angle = static_cast<float>(i) * step;
232 const glm::vec2 offset = perp * std::cos(angle) + dir * std::sin(angle);
233 const glm::vec2 pt = p2 + offset * r;
234 pts[idx++] = {pt.x, pt.y};
235 }
236 for (int i = 0; i <= kCapArcSegments; ++i) {
237 const float angle = static_cast<float>(i) * step;
238 const glm::vec2 offset = -perp * std::cos(angle) - dir * std::sin(angle);
239 const glm::vec2 pt = p1 + offset * r;
240 pts[idx++] = {pt.x, pt.y};
241 }
242 pts[idx++] = pts[0];
243 SDL_RenderLines(renderer, pts, idx);
244 }
