69class RenderPrimitiveSystem : public RenderPrimitiveBase {
70 public:
71 void operator()(const SquarePrimitiveComponent& square, const GlobalTransformComponent& transform) const {
72 // square.position is a local offset from the entity's transform, so it scales with the
73 // entity the same way BoxColliderComponent::offset does in KDTreeCollisionSystem — otherwise a
74 // scaled entity's primitive grows but stays pinned at its unscaled offset.
75 const glm::vec2 scaledOffset = square.position * transform.scale;
76 const glm::vec2 origin = transform.position + scaledOffset;
77 const float w = square.width * transform.scale.x;
78 const float h = square.height * transform.scale.y;
79
80 if (TryCull(origin.x, origin.y, w, h, square.isFixed)) {
81 return;
82 }
83
84 const glm::vec2 screenOrigin = ToScreenPoint(origin, square.isFixed);
85 const float destW = ToScreenLength(w, square.isFixed);
86 const float destH = ToScreenLength(h, square.isFixed);
87 const float sortDepth = square.isFixed ? square.position.y : (origin.y + h);
88
89 auto& cmd =
90 renderQueue_->EmplaceSquare(static_cast<unsigned int>(square.layer), sortDepth, nullptr, square.blendMode);
91 cmd.destRect = {screenOrigin.x, screenOrigin.y, destW, destH};
92 cmd.color = SDL_Color{square.color.r, square.color.g, square.color.b, square.color.a};
93 cmd.rotation = transform.rotation;
94 // The quad's top-left is `origin`, but pivot is an offset from transform.position, so rebase
95 // it onto the destRect the renderer receives. Both are in scaled pixels.
96 const glm::vec2 localPivot = transform.pivot - scaledOffset;
97 cmd.pivot = square.isFixed ? SDL_FPoint{localPivot.x, localPivot.y}
98 : SDL_FPoint{localPivot.x * zoom_, localPivot.y * zoom_};
99 cmd.blendMode = octarine::ToSdlBlendMode(square.blendMode);
100 }
