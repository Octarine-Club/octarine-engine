205 static TextPlacement ResolvePlacement(Registry* registry, const Entity entity, const TextLabelComponent& text,
206 const float textWidth, const float textHeight) {
207 glm::vec2 origin = text.position;
208 float sortY = origin.y;
209 bool effectivelyFixed = text.isFixed;
210 bool visible = true;
211 int renderLayer = text.layer;
212 std::uint8_t clipId = 0;
213 if (registry->HasComponent<UIRectComponent>(entity)) {
214 const auto& rect = registry->GetComponent<UIRectComponent>(entity);
215 visible = rect.visible;
216 clipId = rect.clipId;
217 const float slackX = rect.Width() - textWidth;
218 const float slackY = rect.Height() - textHeight;
219 const auto offsetFor = [](const TextAlign align, const float slack) {
220 static constexpr float kHalf = 0.5f;
221 if (align == TextAlign::Center) return slack * kHalf;
222 if (align == TextAlign::End) return slack;
223 return 0.0f;
224 };
225 origin = glm::vec2(rect.left + offsetFor(text.hAlign, slackX), rect.top + offsetFor(text.vAlign, slackY)) +
226 text.position;
227 sortY = rect.top;
228 effectivelyFixed = true;
229 renderLayer = rect.layer;
230 } else if (registry->HasComponent<GlobalTransformComponent>(entity)) {
231 const auto& transform = registry->GetComponent<GlobalTransformComponent>(entity);
232 origin += transform.position;
233 sortY = origin.y;
234 }
235 origin.x = std::round(origin.x);
236 origin.y = std::round(origin.y);
237 return {origin, sortY, effectivelyFixed, visible, renderLayer, clipId};

260 static constexpr std::uint32_t kSweepInterval = 64;
261 if (++sweep_counter_ < kSweepInterval) return;
262 sweep_counter_ = 0;

306 // Rasterize `text` in white (255,255,255,255) to a freshly created SDL_Texture, writing its logical size
307 // into outW/outH. Per-draw colorMod in TextCommand tints the texture (and its optional shadow pass)
308 // without re-rasterizing when color or alpha changes.
309 static SDL_Texture* RasterizeLabel(const AssetManager& assetManager, SDL_Renderer* sdlRenderer, TTF_Font* font,
310 const TextLabelComponent& text, const int wrapWidth, const float rasterScale,
311 float& outW, float& outH) {
312 const SDL_Color white{255, 255, 255, 255};
313 SDL_Surface* surface = nullptr;
314 const GlyphAtlas* atlas = assetManager.GetGlyphAtlas(text.fontId);
315 const bool hasNewline = text.text.find('\n') != std::string::npos;
316 if (rasterScale <= 1.01F && wrapWidth <= 0 && !hasNewline && atlas != nullptr && atlas->IsLoaded()) {
317 surface = ComposeFromAtlas(*atlas, text.text, white);
318 }
319 if (surface == nullptr) {
320 TTF_Font* renderFont = assetManager.GetFontScaled(text.fontId, rasterScale);
321 if (renderFont == nullptr) {
322 renderFont = font;
323 }
324 surface = RenderTtfSurface(renderFont, text.text, wrapWidth, rasterScale);
325 }
326 if (surface == nullptr) {
327 Logger::Error("RenderText: surface compose failed: " + std::string(SDL_GetError()));
328 return nullptr;
329 }
330 SDL_Texture* texture = SDL_CreateTextureFromSurface(sdlRenderer, surface);
331 SDL_DestroySurface(surface);
332 if (texture == nullptr) {
333 Logger::Error("SDL_CreateTextureFromSurface failed: " + std::string(SDL_GetError()));
334 return nullptr;
335 }
336 SDL_SetTextureScaleMode(texture, SDL_SCALEMODE_LINEAR);
337 float texW = 0.0F;
338 float texH = 0.0F;
339 SDL_GetTextureSize(texture, &texW, &texH);
340 const float invScale = rasterScale > 0.0F ? (1.0F / rasterScale) : 1.0F;
341 outW = texW * invScale;
342 outH = texH * invScale;
343 return texture;
344 }

346 // Atlas compose: build an RGBA destination surface, blit each codepoint's rect from the atlas
347 // into it, color-mod the source surface to tint the white-rasterized atlas glyphs to the
348 // requested color. Returns nullptr on any missing codepoint (caller falls back to TTF). Layout
349 // is top-aligned — not pixel-perfect for descenders, but readable for ASCII at typical UI sizes;
350 // ascent-aware baseline alignment is a follow-up.
351 static SDL_Surface* ComposeFromAtlas(const GlyphAtlas& atlas, const std::string& text, SDL_Color color) {
352 float totalW = 0.0F;
353 for (const char c : text) {
354 const auto cp = static_cast<std::uint32_t>(static_cast<unsigned char>(c));
355 const auto* g = atlas.Find(cp);
356 if (g == nullptr) return nullptr;
357 totalW += g->advance;
358 }
359 const int destW = std::max(1, static_cast<int>(std::ceil(totalW)));
360 const int destH = std::max(1, atlas.LineSkip());
361 SDL_Surface* dst = SDL_CreateSurface(destW, destH, SDL_PIXELFORMAT_RGBA32);
362 if (dst == nullptr) return nullptr;
363
364 SDL_Surface* src = atlas.SourceSurface();
365 SDL_SetSurfaceColorMod(src, color.r, color.g, color.b);
366 SDL_SetSurfaceAlphaMod(src, color.a);
367 SDL_SetSurfaceBlendMode(src, SDL_BLENDMODE_BLEND);
368
369 float pen = 0.0F;
370 for (const char c : text) {
371 const auto cp = static_cast<std::uint32_t>(static_cast<unsigned char>(c));
372 const auto* g = atlas.Find(cp);
373 SDL_Rect srcR{static_cast<int>(g->x), static_cast<int>(g->y), static_cast<int>(g->w), static_cast<int>(g->h)};
374 SDL_Rect dstR{static_cast<int>(pen), 0, static_cast<int>(g->w), static_cast<int>(g->h)};
375 SDL_BlitSurface(src, &srcR, dst, &dstR);
376 pen += g->advance;
377 }
378
379 // Reset mods so the next compose (or a parallel one for a differently-colored label) starts
380 // from a clean state on the shared atlas surface.
381 SDL_SetSurfaceColorMod(src, 255, 255, 255);
382 SDL_SetSurfaceAlphaMod(src, 255);
383 return dst;
384 }
