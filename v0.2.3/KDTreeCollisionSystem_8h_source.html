59[[nodiscard]] inline float PointSegmentDistanceSq(const glm::vec2 p, const glm::vec2 a, const glm::vec2 b) {
60 const glm::vec2 ab = b - a;
61 const float lenSq = glm::dot(ab, ab);
62 float t = 0.0f;
63 if (lenSq > 1e-12f) {
64 t = std::clamp(glm::dot(p - a, ab) / lenSq, 0.0f, 1.0f);
65 }
66 const glm::vec2 diff = p - (a + ab * t);
67 return glm::dot(diff, diff);
68}

74[[nodiscard]] inline bool SegmentsStrictlyCross(const glm::vec2 a1, const glm::vec2 a2, const glm::vec2 b1,
75 const glm::vec2 b2) {
76 const glm::vec2 da = a2 - a1;
77 const glm::vec2 db = b2 - b1;

199 // Shape narrowphase payload:
200 // - Box: (cx, cy) = center, (hx, hy) = half-extents, rot = orientation
201 // - Circle: (cx, cy) = center, (hx, hy) = center, rot.cos = radius
202 // - Capsule: (cx, cy) = start, (hx, hy) = end, rot.cos = radius
203 // - Line: (cx, cy) = start, (hx, hy) = end, rot.cos = 0
204 // - Triangle: (cx, cy) = v0, (hx, hy) = v1, (rot.cos, rot.sin) = v2
205 float cx{0.0f};
206 float cy{0.0f};

222 // SAT on the 4 face normals of two OBBs. Only invoked when at least one box is rotated;
223 // axis-aligned pairs short-circuit on the AABB check above.
224 [[nodiscard]] bool obbIntersects(const Box& other) const {
225 const float ax0 = rot.cos, ay0 = rot.sin;
226 const float ax1 = -rot.sin, ay1 = rot.cos;
227 const float bx0 = other.rot.cos, by0 = other.rot.sin;
228 const float bx1 = -other.rot.sin, by1 = other.rot.cos;
229 const float dx = other.cx - cx;
230 const float dy = other.cy - cy;
231
232 const float axes[4][2] = {{ax0, ay0}, {ax1, ay1}, {bx0, by0}, {bx1, by1}};
233 return std::ranges::all_of(axes, [&](const auto& axis) {
234 const float ux = axis[0];
235 const float uy = axis[1];
236 const float aProj = hx * std::abs(ax0 * ux + ay0 * uy) + hy * std::abs(ax1 * ux + ay1 * uy);
237 const float bProj = other.hx * std::abs(bx0 * ux + by0 * uy) + other.hy * std::abs(bx1 * ux + by1 * uy);
238 const float distProj = std::abs(dx * ux + dy * uy);
239 return distProj <= aProj + bProj;
240 });
241 }

243 [[nodiscard]] static bool CapsuleObbIntersects(const Box& cap, const Box& obb) {
244 const octarine::Rotation2D invRot{obb.rot.cos, -obb.rot.sin};
245 const glm::vec2 obbCenter{obb.cx, obb.cy};
246 const glm::vec2 localP1 = octarine::Rotate(glm::vec2{cap.cx, cap.cy} - obbCenter, invRot);
247 const float rSq = cap.rot.cos * cap.rot.cos;
248 if (cap.shape == ColliderShape::Circle) {
249 return octarine::collision::PointAabbDistanceSq(localP1, obb.hx, obb.hy) <= rSq;
250 }
251 const glm::vec2 localP2 = octarine::Rotate(glm::vec2{cap.hx, cap.hy} - obbCenter, invRot);
252 return octarine::collision::SegmentAabbDistanceSq(localP1, localP2, obb.hx, obb.hy) <= rSq;
253 }

257 return octarine::collision::SegmentTriangleDistanceSq({cap.cx, cap.cy}, {cap.hx, cap.hy}, {tri.cx, tri.cy},

261 [[nodiscard]] static bool CapsulePairIntersects(const Box& a, const Box& b) {
262 const float rSum = a.rot.cos + b.rot.cos;
263 if (a.shape == ColliderShape::Circle && b.shape == ColliderShape::Circle) {
264 const float dx = b.cx - a.cx;

267 }
268 return octarine::collision::SegmentSegmentDistanceSq({a.cx, a.cy}, {a.hx, a.hy}, {b.cx, b.cy}, {b.hx, b.hy}) <=
269 (rSum * rSum);
270 }

284 [[nodiscard]] bool
285 narrowphaseIntersects(const Box& other) const {
286 if (shape == ColliderShape::Box && other.shape == ColliderShape::Box) {
287 return obbIntersects(other);
288 }
289 if (IsCapsuleLike() && other.IsCapsuleLike()) {
290 return CapsulePairIntersects(*this, other);
291 }
292 if (IsCapsuleLike() && other.shape == ColliderShape::Box) return CapsuleObbIntersects(*this, other);
293 if (shape == ColliderShape::Box && other.IsCapsuleLike()) return CapsuleObbIntersects(other, *this);
294 if (IsCapsuleLike() && other.shape == ColliderShape::Triangle) return CapsuleTriangleIntersects(*this, other);
295 if (shape == ColliderShape::Triangle && other.IsCapsuleLike()) return CapsuleTriangleIntersects(other, *this);
296 if (shape == ColliderShape::Box || other.shape == ColliderShape::Box) {
297 return BoxTrianglePairIntersects(*this, other);
298 }
299 return octarine::collision::TrianglesIntersect({cx, cy}, {hx, hy}, {rot.cos, rot.sin}, {other.cx, other.cy},
300 {other.hx, other.hy}, {other.rot.cos, other.rot.sin});
301 }

303 [[nodiscard]] bool containsPoint(const glm::vec2 pt) const {
304 if (pt.x < minX || pt.x > maxX || pt.y < minY || pt.y > maxY) return false;
305 if (shape == ColliderShape::Box) {
306 if (rot.IsIdentity()) return true;

308 return std::abs(local.x) <= hx && std::abs(local.y) <= hy;
309 }
310 if (IsCapsuleLike()) {
311 return octarine::collision::PointSegmentDistanceSq(pt, {cx, cy}, {hx, hy}) <= (rot.cos * rot.cos);
312 }
313 return octarine::collision::PointInTriangle(pt, {cx, cy}, {hx, hy}, {rot.cos, rot.sin});
314 }

317 if (entity == other.entity) return false;

606 }
607 });

633 void UpdateAsynchronous(EventBus* eventBus) {
634 if (!asyncState_) {
635 asyncState_ = std::make_unique<AsyncState>();
636 }
637
638 if (asyncState_->in_flight.load(std::memory_order_acquire)) {
639 if (!asyncState_->ready.load(std::memory_order_acquire)) {
640 return;
641 }
642 PROFILE_NAMED_SCOPE("Emit Events");
643 asyncState_->in_flight.store(false, std::memory_order_relaxed);
644 asyncState_->ready.store(false, std::memory_order_relaxed);
645 EmitCollisionEvents(eventBus, asyncState_->pairs);
646 cachedBoxes_.swap(asyncState_->boxes);
647 }
648
649 PROFILE_NAMED_SCOPE("Gather Boxes");
650 const size_t totalColliders = TotalColliderCount();
651 cachedBoxes_.resize(totalColliders);
652 std::atomic<size_t> nextIndex{0};
653
654 AppendBoxesAsync(*query_, cachedBoxes_, nextIndex);
655 AppendBoxesAsync(*circleQuery_, cachedBoxes_, nextIndex);
656 AppendBoxesAsync(*capsuleQuery_, cachedBoxes_, nextIndex);
657 AppendBoxesAsync(*triangleQuery_, cachedBoxes_, nextIndex);
658 AppendBoxesAsync(*lineQuery_, cachedBoxes_, nextIndex);
659
660 const size_t gathered = nextIndex.load(std::memory_order_relaxed);
661 cachedBoxes_.resize(std::min(gathered, totalColliders));
662
663 PROFILE_COUNTER_SET("Collision: Box count", static_cast<long long>(cachedBoxes_.size()));
664
665 if (cachedBoxes_.empty()) {
666 if (!prevPairs_.empty()) {
667 syncPairsScratch_.clear();
668 EmitCollisionEvents(eventBus, syncPairsScratch_);
669 }
670 return;
671 }
672
673 asyncState_->boxes.swap(cachedBoxes_);
674 asyncState_->ready.store(false, std::memory_order_release);
675 asyncState_->in_flight.store(true, std::memory_order_release);
676
677 ThreadPool::Instance().Submit([state = asyncState_.get()] {
678 state->pairs.clear();
679 ComputeAsyncCollisions(state->boxes, state->pairs);
680 CanonicalizeAndDeduplicatePairs(state->pairs);
681 {
682 std::lock_guard<std::mutex> lk(state->mutex);
683 state->ready.store(true, std::memory_order_release);
684 }
685 state->cv.notify_one();
686 });
687 }

689 public:
690 // Returns true if entity a and entity b are currently overlapping (sustained OR just-entered).
691 // Reflects the result of the most recently completed async detection pass (one frame of lag
692 // on the very first frame, stable thereafter). Safe to call from on_update or on_collision.
693 [[nodiscard]] bool IsOverlapping(const Entity a, const Entity b) const {
694 const auto key = a.id < b.id ? std::make_pair(a, b) : std::make_pair(b, a);
695 return std::binary_search(prevPairs_.begin(), prevPairs_.end(), key, PairLess{});
696 }

699 struct PairLess {
700 bool operator()(const std::pair<Entity, Entity>& a, const std::pair<Entity, Entity>& b) const noexcept {
701 if (a.first.id != b.first.id) return a.first.id < b.first.id;
702 return a.second.id < b.second.id;
703 }
704 };

706 static void CanonicalizeAndDeduplicatePairs(std::vector<std::pair<Entity, Entity>>& pairs) {
707 for (auto& [a, b] : pairs) {
708 if (b.id < a.id) std::swap(a, b);
709 }
710 std::sort(pairs.begin(), pairs.end(), PairLess{});
711 pairs.erase(std::unique(pairs.begin(), pairs.end()), pairs.end());
712 }
713
714 // Diff this frame's overlaps against the previous frame and emit the enter/exit batches.
715 // - enter: pairs overlapping now but not last frame — first-contact only, so persistent
716 // overlaps (entity pinned against an obstacle, stacked entities) don't re-fire every frame.
717 // - exit: pairs that overlapped last frame but no longer do.
718 void EmitCollisionEvents(EventBus* eventBus, std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
719 PROFILE_COUNTER_SET("Collision: Intersecting pairs", static_cast<long long>(intersectingPairs.size()));
720
721 enteringPairs_.clear();
722 exitingPairs_.clear();
723
724 size_t i = 0;
725 size_t j = 0;
726 const size_t nCurr = intersectingPairs.size();
727 const size_t nPrev = prevPairs_.size();
728 while (i < nCurr && j < nPrev) {
729 if (PairLess{}(intersectingPairs[i], prevPairs_[j])) {
730 enteringPairs_.push_back(intersectingPairs[i]);
731 ++i;
732 } else if (PairLess{}(prevPairs_[j], intersectingPairs[i])) {
733 exitingPairs_.push_back(prevPairs_[j]);
734 ++j;
735 } else {
736 ++i;
737 ++j;
738 }
739 }
740 while (i < nCurr) {
741 enteringPairs_.push_back(intersectingPairs[i++]);
742 }
743 while (j < nPrev) {
744 exitingPairs_.push_back(prevPairs_[j++]);
745 }
746
747 prevPairs_.swap(intersectingPairs);
748
749 PROFILE_COUNTER_SET("Collision: Entering pairs", static_cast<long long>(enteringPairs_.size()));
750 eventBus->EmitEvent<CollisionBatchEvent>(enteringPairs_);
751 eventBus->EmitEvent<CollisionExitBatchEvent>(exitingPairs_);
752 }

767 static void BruteForceHugeBoxes(const std::vector<Box>& boxes, const std::vector<Box>::iterator& hugeBoxesBegin,
768 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
769 ACCUMULATE_PROFILE_SCOPE("Huge Box Brute Force");
770 for (auto hugeIt = hugeBoxesBegin; hugeIt != boxes.end(); ++hugeIt) {
771 for (auto normalIt = boxes.begin(); normalIt != hugeBoxesBegin; ++normalIt) {
772 if (hugeIt->intersects(*normalIt)) {
773 intersectingPairs.emplace_back(hugeIt->entity, normalIt->entity);
774 }
775 }
776 for (auto otherHugeIt = hugeIt + 1; otherHugeIt != boxes.end(); ++otherHugeIt) {
777 if (hugeIt->intersects(*otherHugeIt)) {
778 intersectingPairs.emplace_back(hugeIt->entity, otherHugeIt->entity);
779 }
780 }
781 }

784 static void ComputeAsyncCollisions(std::vector<Box>& boxes,
785 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
786 AGGREGATE_PROFILE_SESSION("Async Box Creation");
787 if (boxes.empty()) return;
788
789 ACCUMULATE_PROFILE_SCOPE("Huge Box Extraction");
790
791 // Dynamically calculate the huge box threshold to be scale-invariant.
792 // Defines a "huge outlier" as a box >= 4x the average size.
793 double sum = 0.0;
794 for (const auto& b : boxes) {
795 sum += std::max(b.maxX - b.minX, b.maxY - b.minY);
796 }
797 const double mean = sum / static_cast<double>(boxes.size());
798 const auto hugeThreshold = static_cast<float>(mean * 4.0);
799 // Extract any box larger than the dynamic threshold. These are typically background bounds or map
800 // triggers that cross the entire level. Putting them in the KD-Tree destroys the median
801 // splits and breaks the Sweep Bipartite early-outs.
802 const auto hugeBoxesBegin = std::ranges::partition(boxes, [hugeThreshold](const Box& b) {
803 return (b.maxX - b.minX) <= hugeThreshold && (b.maxY - b.minY) <= hugeThreshold;
804 }).begin();
805
806 const int normalCount = static_cast<int>(std::distance(boxes.begin(), hugeBoxesBegin));
807 if (normalCount > 0) {
808 FindIntersectionsRecursive(boxes, 0, normalCount, 0, 0, intersectingPairs);
809 }
810
811 if (hugeBoxesBegin != boxes.end()) {
812 BruteForceHugeBoxes(boxes, hugeBoxesBegin, intersectingPairs);
813 }
814 }

816 static void FindIntersectionsBruteForce(const std::vector<Box>& boxes, const int begin, const int end,
817 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
818 ACCUMULATE_PROFILE_SCOPE("Brute Force Intersection");
819
820 uint32_t entityUnion = 0;
821 uint32_t collUnion = 0;
822 for (int i = begin; i < end; ++i) {
823 entityUnion |= boxes[static_cast<size_t>(i)].entityMask;
824 collUnion |= boxes[static_cast<size_t>(i)].collisionMask;
825 }
826 if ((entityUnion & collUnion) == 0U) {
827 return;
828 }
829
830 int checks = 0;
831 for (int i = begin; i < end; ++i) {
832 for (int j = i + 1; j < end; ++j) {
833 checks++;
834 const auto& bi = boxes[static_cast<size_t>(i)];
835 const auto& bj = boxes[static_cast<size_t>(j)];
836 if (bi.intersects(bj)) {
837 intersectingPairs.emplace_back(bi.entity, bj.entity);
838 }
839 }
840 }
841#ifdef OCTARINE_PROFILING
842 PROFILE_COUNTER_ADD("Collision: Brute Force Checks", checks);
843#else
844 (void)checks;
845#endif
846 }

848 [[nodiscard]] static Partitions PartitionBoxes(std::vector<Box>& boxes, const int begin, const int end,
849 const int dimension, const float medianValue) {
850 ACCUMULATE_PROFILE_SCOPE("Partition Boxes");
851 const auto first = boxes.begin() + begin;
852 const auto last = boxes.begin() + end;
853
854 const auto leftEndIt = std::partition(first, last, [&](const Box& box) {
855 if (dimension == 0) {
856 return box.maxX < medianValue;
857 }
858
859 return box.maxY < medianValue;
860 });
861
862 const auto rightBeginIt = std::partition(leftEndIt, last, [&](const Box& box) {
863 if (dimension == 0) {
864 return box.minX <= medianValue;
865 }
866
867 return box.minY <= medianValue;
868 });
869
870 const int leftEnd = static_cast<int>(std::distance(boxes.begin(), leftEndIt));
871
872 const int rightStart = static_cast<int>(std::distance(boxes.begin(), rightBeginIt));
873
874 return Partitions{leftEnd, rightStart};

899 // KD-Tree inherently requires recursion and dense Bipartite bounds-checking logic.
900 // NOLINTNEXTLINE(misc-no-recursion,readability-function-cognitive-complexity)
901 static void FindIntersectionsSweepBipartite(std::vector<Box>& boxes, const int begin1, const int end1,
902 const int begin2, const int end2, const int dimension,
903 std::vector<std::pair<Entity, Entity>>& pairs) {
904 ACCUMULATE_PROFILE_SCOPE("Sweep Bipartite");
905
906 if (end1 - begin1 == 0 || end2 - begin2 == 0) return;
907
908 // Aggregate Mask Pruning: Skips bipartite sweep entirely if the union of entity and collision masks proves no
909 // interaction is possible.
910 uint32_t entity1 = 0;
911 uint32_t coll1 = 0;
912 for (int i = begin1; i < end1; ++i) {
913 entity1 |= boxes[static_cast<size_t>(i)].entityMask;
914 coll1 |= boxes[static_cast<size_t>(i)].collisionMask;
915 }
916
917 uint32_t entity2 = 0;
918 uint32_t coll2 = 0;
919 for (int i = begin2; i < end2; ++i) {
920 entity2 |= boxes[static_cast<size_t>(i)].entityMask;
921 coll2 |= boxes[static_cast<size_t>(i)].collisionMask;
922 }
923
924 const bool groupCanInteract = ((coll1 & entity2) | (coll2 & entity1)) != 0U;
925 if (!groupCanInteract) {
926 return;
927 }
928
929 if ((end1 - begin1) * (end2 - begin2) <= kBruteforceCutoff * kBruteforceCutoff) {
930 FindIntersectionsBruteForceBipartite(boxes, begin1, end1, begin2, end2, pairs);
931 return;
932 }
933
934 {
935 ACCUMULATE_PROFILE_SCOPE("Bipartite Sort Phase");
936 if (dimension == 0) {
937 std::sort(boxes.begin() + begin1, boxes.begin() + end1, [](const Box& a, const Box& b) {
938 const bool aFin = IsFiniteFloat(a.minX);
939 const bool bFin = IsFiniteFloat(b.minX);
940 if (!aFin || !bFin) return aFin && !bFin;
941 return a.minX < b.minX;
942 });
943 std::sort(boxes.begin() + begin2, boxes.begin() + end2, [](const Box& a, const Box& b) {
944 const bool aFin = IsFiniteFloat(a.minX);
945 const bool bFin = IsFiniteFloat(b.minX);
946 if (!aFin || !bFin) return aFin && !bFin;
947 return a.minX < b.minX;
948 });
949 } else {
950 std::sort(boxes.begin() + begin1, boxes.begin() + end1, [](const Box& a, const Box& b) {
951 const bool aFin = IsFiniteFloat(a.minY);
952 const bool bFin = IsFiniteFloat(b.minY);
953 if (!aFin || !bFin) return aFin && !bFin;
954 return a.minY < b.minY;
955 });
956 std::sort(boxes.begin() + begin2, boxes.begin() + end2, [](const Box& a, const Box& b) {
957 const bool aFin = IsFiniteFloat(a.minY);
958 const bool bFin = IsFiniteFloat(b.minY);
959 if (!aFin || !bFin) return aFin && !bFin;
960 return a.minY < b.minY;
961 });
962 }
963 }
964
965 {
966 ACCUMULATE_PROFILE_SCOPE("Bipartite Sweep Phase");
967 int sweepChecks = 0;
968 if (dimension == 0) {
969 int startJ = begin2;
970 for (int i = begin1; i < end1; ++i) {
971 const Box& a = boxes[static_cast<size_t>(i)];
972 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxX < a.minX) ++startJ;
973 for (int j = startJ; j < end2; ++j) {
974 sweepChecks++;
975 const Box& b = boxes[static_cast<size_t>(j)];
976 if (b.minX > a.maxX) break;
977 if (b.maxY < a.minY || b.minY > a.maxY) continue;
978 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
979 }
980 }
981 } else {
982 int startJ = begin2;
983 for (int i = begin1; i < end1; ++i) {
984 const Box& a = boxes[static_cast<size_t>(i)];
985 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxY < a.minY) ++startJ;
986 for (int j = startJ; j < end2; ++j) {
987 sweepChecks++;
988 const Box& b = boxes[static_cast<size_t>(j)];
989 if (b.minY > a.maxY) break;
990 if (b.maxX < a.minX || b.minX > a.maxX) continue;
991 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
992 }
993 }
994 }
995#ifdef OCTARINE_PROFILING
996 PROFILE_COUNTER_ADD("Collision: Bipartite Sweep Inner Loop Checks", sweepChecks);
997#else
998 (void)sweepChecks;
999#endif
1000 }
1001 }

1005 // NOLINTNEXTLINE(misc-no-recursion) - KD-Tree cross-checking inherently requires recursion.
1006 static void ProcessSpanningBipartite(std::vector<Box>& boxes, const int begin, const int end, const int leftEnd,
1007 const int rightStart, const int dimension, const int nextDimension,
1008 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
1009 if (leftEnd >= rightStart) return;
1010
1011 // Find the min/max bounds of the Spanning set along the current dimension
1012 float spanMin = std::numeric_limits<float>::max();
1013 float spanMax = std::numeric_limits<float>::lowest();
1014 for (int i = leftEnd; i < rightStart; ++i) {
1015 const float minV = dimension == 0 ? boxes[static_cast<size_t>(i)].minX : boxes[static_cast<size_t>(i)].minY;
1016 const float maxV = dimension == 0 ? boxes[static_cast<size_t>(i)].maxX : boxes[static_cast<size_t>(i)].maxY;
1017 if (minV < spanMin) spanMin = minV;
1018 if (maxV > spanMax) spanMax = maxV;
1019 }

1040 static void FindIntersectionsRecursive(std::vector<Box>& boxes, const int begin, const int end, int dimension,
1041 const int depth, std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
1042 const int count = end - begin;
1043 if (count <= 1) {
1044 return;
1045 }
1046
1047 if (count < kBruteforceCutoff || depth >= kMaxRecursionDepth) {
1048 FindIntersectionsBruteForce(boxes, begin, end, intersectingPairs);
1049 return;
1050 }
1051
1052 const int mid = (begin + end) / 2;
1053 float medianValue = 0.0f;
1054 {
1055 ACCUMULATE_PROFILE_SCOPE("Find Median");
1056 std::nth_element(boxes.begin() + begin, boxes.begin() + mid, boxes.begin() + end,
1057 [dimension](const Box& a, const Box& b) {
1058 const float va = (dimension == 0) ? (a.maxX + a.minX) : (a.maxY + a.minY);
1059 const float vb = (dimension == 0) ? (b.maxX + b.minX) : (b.maxY + b.minY);
1060 const bool aFin = IsFiniteFloat(va);
1061 const bool bFin = IsFiniteFloat(vb);
1062 if (!aFin || !bFin) return aFin && !bFin;
1063 return va < vb;
1064 });
1065 medianValue = (dimension == 0)
1066 ? (boxes[static_cast<size_t>(mid)].minX + boxes[static_cast<size_t>(mid)].maxX) * kHalf
1067 : (boxes[static_cast<size_t>(mid)].minY + boxes[static_cast<size_t>(mid)].maxY) * kHalf;
1068 if (!IsFiniteFloat(medianValue)) {
1069 FindIntersectionsBruteForce(boxes, begin, end, intersectingPairs);
1070 return;
1071 }
1072 }
1073
1074 const auto [leftEnd, rightStart] = PartitionBoxes(boxes, begin, end, dimension, medianValue);
1075 const int nextDimension = (dimension + 1) % kMaxDimensions;
1076
1077 FindIntersectionsRecursive(boxes, begin, leftEnd, dimension, depth + 1, intersectingPairs);
1078 FindIntersectionsRecursive(boxes, rightStart, end, dimension, depth + 1, intersectingPairs);
1079 FindIntersectionsRecursive(boxes, leftEnd, rightStart, nextDimension, depth + 1, intersectingPairs);
1080
1081 ProcessSpanningBipartite(boxes, begin, end, leftEnd, rightStart, dimension, nextDimension, intersectingPairs);
