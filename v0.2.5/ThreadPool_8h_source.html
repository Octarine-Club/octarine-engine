42 // Synchronizes batch completion before destroying barrier.
43 struct Barrier {
44 size_t remaining;
45 std::mutex mutex;
46 std::condition_variable cv;
47 explicit Barrier(size_t n) : remaining(n) {}
48 void Signal() {
49 std::lock_guard<std::mutex> lk(mutex);
50 if (--remaining == 0) cv.notify_one();
51 }
52 void Wait() {
53 std::unique_lock<std::mutex> lk(mutex);
54 cv.wait(lk, [this] { return remaining == 0; });
55 }
56 } barrier(numBatches - 1);

75 // Caller runs the final batch inline rather than sitting idle, then waits for the dispatched
76 // ones. The barrier count equals the number of Submits regardless of whether any range is
77 // empty, so there is no deadlock even on a ragged final split.

86 {
