41 [[nodiscard]] static bool IsPaletteTemplateName(const std::string& identifier) noexcept {
42 if (identifier == "Chunk_Palette" || identifier == "Palette" || identifier == "__Palette") {

48 return std::tolower(static_cast<unsigned char>(c1)) == std::tolower(static_cast<unsigned char>(c2));
49 });
50 return it != identifier.end();

60 static void AppendStaticTileQuad(TileLayerMesh& mesh, const TilesetDesc& ts, const TileQuadDesc& quad) {
61 const float texW = std::max(1.0f, ts.textureWidth);
62 const float texH = std::max(1.0f, ts.textureHeight);
63 const float gridSize = std::max(1.0f, quad.size);
64 const float inset = std::min(kUvInsetPx, gridSize * 0.25f);
65
66 float u0 = (quad.srcX + inset) / texW;
67 float v0 = (quad.srcY + inset) / texH;
68 float u1 = (quad.srcX + gridSize - inset) / texW;
69 float v1 = (quad.srcY + gridSize - inset) / texH;
70
71 if (quad.flipX) {
72 std::swap(u0, u1);
73 }
74 if (quad.flipY) {
75 std::swap(v0, v1);
76 }
77
78 const float clampedAlpha = std::clamp(quad.alpha, 0.0f, 1.0f);
79 const SDL_FColor color{1.0f, 1.0f, 1.0f, clampedAlpha};
80 auto& verts = mesh.MutableVertices();
81 auto& indices = mesh.MutableIndices();
82 const auto baseIdx = static_cast<int>(verts.size());
83 verts.push_back(SDL_Vertex{{quad.localX, quad.localY}, color, {u0, v0}});
84 verts.push_back(SDL_Vertex{{quad.localX + gridSize, quad.localY}, color, {u1, v0}});
85 verts.push_back(SDL_Vertex{{quad.localX + gridSize, quad.localY + gridSize}, color, {u1, v1}});
86 verts.push_back(SDL_Vertex{{quad.localX, quad.localY + gridSize}, color, {u0, v1}});
87
88 indices.push_back(baseIdx + 0);
89 indices.push_back(baseIdx + 1);
90 indices.push_back(baseIdx + 2);
91 indices.push_back(baseIdx + 0);
92 indices.push_back(baseIdx + 2);
93 indices.push_back(baseIdx + 3);
94 }

116 // Processes Y-sorted tiles bottom-up (descending localY) so contiguous vertical stacks
117 // (e.g. multi-tile pillars, statues, or trees) share the bottom-most tile's sortLocalY.
118 static void GroupVerticalYSortTiles(std::vector<YSortTileCmd>& tiles) {
119 if (tiles.size() <= 1u) {
120 return;
121 }
122 std::vector<size_t> order(tiles.size());
123 for (size_t i = 0; i < tiles.size(); ++i) {
124 order[i] = i;
125 }
126 std::sort(order.begin(), order.end(), [&](const size_t a, const size_t b) {
127 if (std::abs(tiles[a].localY - tiles[b].localY) > 0.01f) {
128 return tiles[a].localY > tiles[b].localY;
129 }
130 return tiles[a].localX < tiles[b].localX;
131 });
132
133 for (const size_t idx : order) {
134 auto& upper = tiles[idx];
135 const float targetBelowY = upper.localY + upper.height;
136 for (const size_t otherIdx : order) {
137 const auto& lower = tiles[otherIdx];
138 if (std::abs(lower.localX - upper.localX) < 0.1f && std::abs(lower.localY - targetBelowY) < 0.1f) {
139 upper.sortLocalY = std::max(upper.sortLocalY, lower.sortLocalY);
140 break;
141 }
142 }
143 }
144 }

147 bool extracted = false;
148 for (auto it = data.templates.begin(); it != data.templates.end();) {

153 extracted = true;

184 static void ScaleSingleTemplate(ChunkTemplate& tmpl, const int32_t scaledTileSize, const float exactFactor) {
185 tmpl.tileSize = scaledTileSize;
186 for (auto& layer : tmpl.staticLayers) {

197 ytile.sortLocalY *= exactFactor;

200 prop.localX *= exactFactor;
201 prop.localY *= exactFactor;
202 prop.width *= exactFactor;

207 static void ApplyWorldTileScale(TilemapData& data, const float rawScale) {
208 const float scale = std::clamp(rawScale, 0.25f, 16.0f);
209 if (std::abs(scale - 1.0f) < 1e-4f) {
210 return;
211 }
212 const int32_t scaledTileSize =
213 std::max(1, static_cast<int32_t>(std::lround(static_cast<float>(data.tileSize) * scale)));
214 const float exactFactor = static_cast<float>(scaledTileSize) / static_cast<float>(std::max(1, data.tileSize));
215 data.tileSize = scaledTileSize;
216
217 for (auto& tmpl : data.templates) {
218 ScaleSingleTemplate(tmpl, scaledTileSize, exactFactor);
219 }
220 for (auto& pal : data.paletteTemplates) {
221 ScaleSingleTemplate(pal, scaledTileSize, exactFactor);
222 }
223 for (auto& [_, varList] : data.templateVariants) {
224 for (auto& var : varList) {
225 ScaleSingleTemplate(var, scaledTileSize, exactFactor);
226 }
227 }
228 ScaleSingleTemplate(data.borderTemplate, scaledTileSize, exactFactor);
229 }
