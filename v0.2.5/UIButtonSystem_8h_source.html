62 void OnMouseInput(const MouseInputEvent& event) {
63 const auto mousePos = GetTransformedMousePos(event);
64 if (!mousePos || !EnsureMouseQuery()) {
65 return;
66 }
67 lastMousePos_ = *mousePos;
68
69 const auto* cameraComp = registry_->TryGet<CameraComponent>();
70 const octarine::Rect camera = cameraComp ? cameraComp->viewport : octarine::Rect{};
71 const float zoom = (cameraComp && cameraComp->zoom > 0.001f) ? cameraComp->zoom : 1.0f;
72
73 if (event.event.type == SDL_EVENT_MOUSE_MOTION) {
74 UpdateMouseHover(camera, zoom, *mousePos);
75 return;
76 }
77 if (event.event.type == SDL_EVENT_MOUSE_BUTTON_UP && event.event.button == 1) {
78 ClearPressedState();
79 return;
80 }
81 if (event.event.type == SDL_EVENT_MOUSE_BUTTON_DOWN && event.event.button == 1) {
82 HandleMousePress(camera, zoom, *mousePos);
83 }

94
95 const int blockingLayer = FindTopPointerBlockingLayer(lastMousePos_);

99 if (!rect.visible || rect.layer < blockingLayer || !ContainsPointInRect(rect, lastMousePos_)) return;

109 if (scroll.scrollHorizontal) scroll.scrollX -= event.dx * scroll.wheelStep;

194 if (!EnsureFocusQuery()) return;
195 const Entity prevFocused = GetFocusedEntity();

232 [[nodiscard]] static Entity ResolveExplicitDirectionTarget(const UIFocusComponent& curFocus,
233 const glm::vec2 dir) noexcept {
234 if (dir.y < -0.5f) return curFocus.navUp;
235 if (dir.y > 0.5f) return curFocus.navDown;

241 bool NavigateDirection(const glm::vec2 dir) {
242 if (!EnsureFocusQuery()) return false;
243 const Entity current = GetFocusedEntity();
244 if (current.GetId() == static_cast<std::uint32_t>(-1)) {
245 return CycleFocusTab(false);
246 }
247
248 const Entity explicitNext = ResolveExplicitDirectionTarget(registry_->GetComponent<UIFocusComponent>(current), dir);
249 if (IsFocusableCandidate(explicitNext)) {
250 SetFocus(explicitNext);
251 return true;
252 }
253
254 if (!registry_->HasComponent<UIRectComponent>(current)) return false;
255 const glm::vec2 curCenter = registry_->GetComponent<UIRectComponent>(current).Center();
256
257 Entity best{static_cast<EcsId>(-1)};
258 float bestScore = std::numeric_limits<float>::max();
259 focusQuery_->ForEach([&](const Entity candidate, const UIFocusComponent& foc) {
260 if (candidate == current || !foc.isFocusable || foc.isDisabled || foc.isInert || !IsButtonVisible(candidate)) {
261 return;
262 }
263 if (!registry_->HasComponent<UIRectComponent>(candidate)) return;
264 const glm::vec2 delta = registry_->GetComponent<UIRectComponent>(candidate).Center() - curCenter;
265 const float proj = delta.x * dir.x + delta.y * dir.y;
266 if (proj <= 1.0f) return;
267 const float perp = std::abs(delta.x * dir.y - delta.y * dir.x);
268 const float score = proj + perp * 2.5f;
269 if (score < bestScore) {
270 bestScore = score;
271 best = candidate;
272 }
273 });
274
275 if (best.GetId() != static_cast<std::uint32_t>(-1)) {
276 SetFocus(best);
277 return true;
278 }
279 return false;
280 }

282 bool CycleFocusTab(const bool reverse) {
283 if (!EnsureFocusQuery()) return false;
284 std::vector<std::pair<std::int32_t, Entity>> candidates;
285 focusQuery_->ForEach([&](const Entity entity, const UIFocusComponent& foc) {
286 if (foc.isFocusable && !foc.isDisabled && !foc.isInert && IsButtonVisible(entity)) {
287 candidates.emplace_back(foc.tabIndex, entity);
288 }
289 });
290 if (candidates.empty()) return false;
291 std::stable_sort(candidates.begin(), candidates.end(),
292 [](const auto& a, const auto& b) { return a.first < b.first; });
293
294 const Entity current = GetFocusedEntity();
295 size_t nextIdx = 0;
296 for (size_t i = 0; i < candidates.size(); ++i) {
297 if (candidates[i].second == current) {
298 nextIdx = reverse ? ((i + candidates.size() - 1) % candidates.size()) : ((i + 1) % candidates.size());
299 break;
300 }
301 }
302 SetFocus(candidates[nextIdx].second);
303 return true;
304 }

342 [[nodiscard]] int FindTopPointerBlockingLayer(const glm::vec2 mousePos) {
343 if (!registry_) return std::numeric_limits<int>::min();
344 if (!pointerBlockQuery_) {
345 pointerBlockQuery_ = registry_->CreateQuery<UIRectComponent>();
346 }
347 if (!pointerBlockQuery_) return std::numeric_limits<int>::min();
348 pointerBlockQuery_->Update();
349 int blockingLayer = std::numeric_limits<int>::min();
350 pointerBlockQuery_->ForEach([&](const Entity /*entity*/, const UIRectComponent& rect) {
351 if (!rect.visible || !rect.blocksPointer || rect.layer <= blockingLayer) return;
352 if (!ContainsPointInRect(rect, mousePos) || !IsInsideActiveClip(rect, mousePos)) return;

358 [[nodiscard]] Entity FindTopmostHit(const octarine::Rect& camera, const float zoom, const glm::vec2 mousePos) {
359 const int blockingLayer = FindTopPointerBlockingLayer(mousePos);
360 Entity topEntity{static_cast<EcsId>(-1)};
361 int topLayer = std::numeric_limits<int>::min();
362 mouseQuery_->ForEach([&](const Entity entity, UIButtonComponent& button, const GlobalTransformComponent* transform,
363 const BoxColliderComponent* collider) {
364 if (!button.isActive || !IsButtonVisible(entity)) return;
365 const int layer = ComputeEntitySortLayer(entity);
366 if (layer < blockingLayer) return;
367 if (!HitTest(entity, button, transform, collider, camera, zoom, mousePos.x, mousePos.y)) return;
368 if (layer >= topLayer) {
369 topLayer = layer;
370 topEntity = entity;
371 }
372 });
373 return topEntity;
374 }

406 void HandleMousePress(const octarine::Rect& camera, const float zoom, const glm::vec2 mousePos) {
407 const Entity topHit = FindTopmostHit(camera, zoom, mousePos);
408 if (topHit.GetId() == static_cast<std::uint32_t>(-1) || !registry_->IsAlive(topHit)) return;
409 if (registry_->HasComponent<UIFocusComponent>(topHit)) {
410 // Suppress on-focus on mouse clicks so first-click inspection runs before focus state flips.
411 SetFocus(topHit, false);
412 }
413 if (!registry_->IsAlive(topHit) || !registry_->HasComponent<UIButtonComponent>(topHit)) return;
414 auto& button = registry_->GetComponent<UIButtonComponent>(topHit);
415 button.isHovered = true;
416 button.isPressed = true;
417 TriggerButton(topHit, button);
418 }

420 void ScrollFocusedIntoView(const Entity target) {
421 if (!registry_ || !registry_->HasComponent<UIRectComponent>(target)) return;
422 const UIRectComponent& targetRect = registry_->GetComponent<UIRectComponent>(target);
423 if (targetRect.Width() <= 0.0f || targetRect.Height() <= 0.0f) return;
424
425 for (auto anc = registry_->GetParent(target); anc.has_value(); anc = registry_->GetParent(*anc)) {
426 if (!registry_->IsAlive(*anc) || !registry_->HasComponent<UIScrollComponent>(*anc) ||
427 !registry_->HasComponent<UIRectComponent>(*anc)) {
428 continue;
429 }
430 const auto& viewRect = registry_->GetComponent<UIRectComponent>(*anc);
431 if (viewRect.Width() <= 0.0f || viewRect.Height() <= 0.0f) continue;
432 auto& scroll = registry_->GetComponent<UIScrollComponent>(*anc);
433 if (scroll.scrollVertical) {
434 if (targetRect.top < viewRect.top) scroll.scrollY -= (viewRect.top - targetRect.top);
435 if (targetRect.bottom > viewRect.bottom) scroll.scrollY += (targetRect.bottom - viewRect.bottom);
436 }
437 if (scroll.scrollHorizontal) {
438 if (targetRect.left < viewRect.left) scroll.scrollX -= (viewRect.left - targetRect.left);
439 if (targetRect.right > viewRect.right) scroll.scrollX += (targetRect.right - viewRect.right);
440 }
441 scroll.ClampToViewport(viewRect.Width(), viewRect.Height());
442 }
443 }

470 void HandleAnalogLeftStickNav(const float dt, const glm::vec2 stick) {
471 glm::ivec2 dir{0, 0};
472 if (glm::length(stick) > 0.0f) {
473 if (std::abs(stick.x) >= std::abs(stick.y)) {
474 dir.x = (stick.x > 0.0f) ? 1 : -1;
475 } else {
476 dir.y = (stick.y > 0.0f) ? 1 : -1;
477 }
478 }
479 if (dir == glm::ivec2{0, 0}) {
480 analogNavDir_ = {0, 0};
481 analogNavTimer_ = 0.0f;
482 return;
483 }
484 constexpr float kInitialRepeatDelay = 0.32f;
485 constexpr float kRepeatInterval = 0.14f;
486 if (dir != analogNavDir_) {
487 analogNavDir_ = dir;
488 analogNavTimer_ = kInitialRepeatDelay;
489 DispatchAnalogDirection(dir);
490 return;
491 }
492 analogNavTimer_ -= dt;
493 if (analogNavTimer_ <= 0.0f) {
494 analogNavTimer_ = kRepeatInterval;
495 DispatchAnalogDirection(dir);
496 }
497 }

536 void HandleAnalogRightStickScroll(const float dt, const glm::vec2 stick) {
537 if (std::abs(stick.x) <= 0.0f && std::abs(stick.y) <= 0.0f) return;

560 bool HandleFocusKeyNavigation(const SDL_Keycode keycode, const SDL_Keymod mod) {
561 if (!EnsureFocusQuery() || !HasAnyFocusable()) return false;
562 if (keycode == SDLK_TAB) return CycleFocusTab((mod & SDL_KMOD_SHIFT) != 0);
563 if (keycode == SDLK_UP) return NavigateDirection({0.0f, -1.0f});
564 if (keycode == SDLK_DOWN) return NavigateDirection({0.0f, 1.0f});
565 if (keycode == SDLK_LEFT) return AdjustFocusedSlider(-1.0f) || NavigateDirection({-1.0f, 0.0f});
566 if (keycode == SDLK_RIGHT) return AdjustFocusedSlider(1.0f) || NavigateDirection({1.0f, 0.0f});
567 if (keycode == SDLK_RETURN || keycode == SDLK_KP_ENTER) {
568 return TriggerFocusedButton();
569 }
570 return false;
571 }

573 bool HandleFocusGamepadNavigation(const SDL_GamepadButton btn) {
574 if (!EnsureFocusQuery() || !HasAnyFocusable()) return false;
575 if (btn == SDL_GAMEPAD_BUTTON_DPAD_UP) return NavigateDirection({0.0f, -1.0f});
576 if (btn == SDL_GAMEPAD_BUTTON_DPAD_DOWN) return NavigateDirection({0.0f, 1.0f});
577 if (btn == SDL_GAMEPAD_BUTTON_DPAD_LEFT) return AdjustFocusedSlider(-1.0f) || NavigateDirection({-1.0f, 0.0f});
578 if (btn == SDL_GAMEPAD_BUTTON_DPAD_RIGHT) return AdjustFocusedSlider(1.0f) || NavigateDirection({1.0f, 0.0f});
579 if (btn == SDL_GAMEPAD_BUTTON_SOUTH) {
580 return TriggerFocusedButton();
581 }

631 static bool ForEachCommaToken(const std::string& spec, Fn&& fn) {
632 if (spec.empty()) return false;
633 size_t start = 0;
634 while (start < spec.size()) {
635 size_t end = spec.find(',', start);
636 if (end == std::string::npos) end = spec.size();
637 std::string token = spec.substr(start, end - start);
638 const size_t first = token.find_first_not_of(" \t\r\n");
639 if (first != std::string::npos) {
640 const size_t last = token.find_last_not_of(" \t\r\n");
641 token = token.substr(first, last - first + 1);
642 if (fn(token)) return true;
643 }
644 start = end + 1;
645 }
646 return false;
647 }

684 if (c.radiusX <= 0.0f || c.radiusY <= 0.0f) return false;
685 const float dx = (pt.x - c.center.x) / c.radiusX;
