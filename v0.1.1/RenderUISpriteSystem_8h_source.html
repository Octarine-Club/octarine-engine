18// Renders SpriteComponent entities whose position is driven by UIRectComponent (layout-driven UI
19// sprites). Runs as a serial RegisterSystem after UILayoutSystem. World sprites without
20// UIRectComponent continue through the parallel RenderSpriteSystem path unchanged.
21//
22// Singletons are fetched per-call via the ContextFacade, the same way RenderTextSystem does: the
23// serial RegisterSystem wrapper does not invoke a Prepare() hook (only the parallel wrapper does),
24// so caching them in members would leave them null.
25class RenderUISpriteSystem {
26 public:
27 void operator()(const ContextFacade& ctx, Entity entity, const UIRectComponent& rect,
28 const SpriteComponent& sprite) const {
29 Registry* registry = ctx.GetRegistry();
30 auto& assetManager = registry->Get<AssetManager>();
31 auto& renderQueue = registry->Get<RenderQueue>();
32 auto& spriteCache = registry->Get<SpriteRenderCache>();
33
34 const SpriteRenderCache::Entry* entry = spriteCache.Lookup(entity, assetManager.TextureGeneration());
35 if (!entry || !entry->texture) {
36 // Warm the cache entry on demand (should rarely happen; the world sprite warming path
37 // covers most entities, but pure-UI sprites without GlobalTransformComponent are missed).
38 SDL_Texture* texture = assetManager.GetTexture(sprite.assetId);
39 const auto slice = assetManager.GetAtlasSlice(sprite.assetId);
40 const SDL_FRect atlasOffset = slice.has_value() ? *slice : SDL_FRect{0, 0, 0, 0};
41 spriteCache.Store(entity, texture, atlasOffset, assetManager.TextureGeneration());
42 entry = spriteCache.Lookup(entity, assetManager.TextureGeneration());
43 if (!entry) return;
44 }
45
46 SDL_Texture* texture = entry->texture;
47 const SDL_FRect atlasOffset = entry->atlasOffset;
48
49 const float destW = rect.Width();
50 const float destH = rect.Height();
51
52 auto& cmd = renderQueue.EmplaceSprite(static_cast<unsigned int>(rect.layer), rect.top, texture, sprite.blendMode);
53 cmd.destX = rect.left;
54 cmd.destY = rect.top;
55 cmd.destW = destW;
56 cmd.destH = destH;
57 cmd.srcRect.x = sprite.srcRect.x + atlasOffset.x;
58 cmd.srcRect.y = sprite.srcRect.y + atlasOffset.y;
59 cmd.srcRect.w = sprite.srcRect.w;
60 cmd.srcRect.h = sprite.srcRect.h;
61 static constexpr float kHalf = 0.5f;
62 cmd.rotation = 0.0;
63 cmd.pivot = {destW * kHalf, destH * kHalf};
64 cmd.flip = static_cast<SDL_FlipMode>(sprite.flip);
65 cmd.texture = texture;
66 cmd.colorMod = SDL_Color{sprite.colorMod.r, sprite.colorMod.g, sprite.colorMod.b, sprite.colorMod.a};
67 cmd.blendMode = octarine::ToSdlBlendMode(sprite.blendMode);
68 }
69};
