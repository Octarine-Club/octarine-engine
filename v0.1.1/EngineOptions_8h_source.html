13// Which metrics the perf overlay draws. config.ini: PerfOverlayMetrics takes a comma-separated
14// list of fps|frametime|entities|memory, or all ("both" accepted as a legacy alias).
15enum class PerfOverlayMetrics : std::uint8_t {
16 Fps = 1 << 0,
17 FrameTime = 1 << 1,
18 Entities = 1 << 2,
19 Memory = 1 << 3,
20 All = Fps | FrameTime | Entities | Memory
21};

23inline PerfOverlayMetrics operator|(PerfOverlayMetrics lhs, PerfOverlayMetrics rhs) {

25}
