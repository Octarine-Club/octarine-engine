12// Bulk system: snapshots the (first) AudioListenerComponent entity each frame into the
13// AudioListenerCache singleton on the Registry. Runs after TransformSystem so the cached
14// position reflects this frame's resolved global transform. SpatialAudioSystem reads the
15// cache once per emitter — no second archetype walk for the listener.
16class UpdateListenerTransformSystem {
17 public:
18 void operator()(const ContextFacade& ctx) {
19 auto* registry = ctx.GetRegistry();
20 auto& cache = registry->Get<AudioListenerCache>();
21 cache.valid = false;
22 cache.entity = Entity{};
23
24 if (!query_) {
25 query_ = registry->CreateQuery<GlobalTransformComponent, AudioListenerComponent>();
26 }
27 query_->Update();
28
29 bool found = false;
30 int listenerCount = 0;
31 query_->ForEach(
32 [&](const Entity entity, const GlobalTransformComponent& transform, const AudioListenerComponent& listener) {
33 ++listenerCount;
34 if (found) return;
35 cache.entity = entity;
36 cache.position = transform.position;
37 cache.velocity = registry->HasComponent<RigidBodyComponent>(entity)
38 ? registry->GetComponent<RigidBodyComponent>(entity).velocity
39 : glm::vec2(0.0f, 0.0f);
40 // 2D top-down basis. When camera rotation lands later, derive forward/right from
41 // the camera's world rotation here so the spatial math stays decoupled from any
42 // listener-entity rotation (the "Listener Dilemma" decoupling).
43 cache.forward = glm::vec2(0.0f, 1.0f);
44 cache.right = glm::vec2(1.0f, 0.0f);
45 cache.maxDistance = listener.maxDistance;
46 cache.rolloff = listener.rolloff;
47 cache.dopplerFactor = listener.dopplerFactor;
48 cache.speedOfSound = listener.speedOfSound;
49 cache.valid = true;
50 found = true;
51 });
52
53 if (listenerCount > 1 && !warnedMultipleListeners_) {
54 Logger::Warn("UpdateListenerTransformSystem: " + std::to_string(listenerCount) +
55 " AudioListenerComponent entities found; using the first one.");
56 warnedMultipleListeners_ = true;
57 }
58 }
