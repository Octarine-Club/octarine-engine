65 // One parent paired with the contiguous span of its children inside children_. Grouping by
66 // parent is what makes the descend pass cheap: the parent's global is resolved once and reused

98 // pays for itself near the measured serial/parallel crossover of ~16k entities. Half that,
99 // to stay parallel where the win is real and serial where dispatch dominates.
100 static constexpr size_t kFlatSerialBelowEntities = 8192;

102 // Each entity's global is its local (identity for missing slots), in a single parallel pass.
103 // Final for everything outside a hierarchy, and the seed value for hierarchy roots.
104 void UpdateFlat() {
105 PROFILE_NAMED_SCOPE("TransformSystem: Flat");
106 optionalQuery_->ParallelForEach(
107 [](GlobalTransformComponent& global, const PositionComponent* p, const ScaleComponent* s,
108 const RotationComponent* r, const PivotComponent* pivot, const SpriteComponent* sprite,
109 const SquarePrimitiveComponent* square, const BoxColliderComponent* collider) {
110 const glm::vec2 scale = s ? s->value : glm::vec2(1.0f, 1.0f);
111 const glm::vec2 anchor = ResolveAnchor(pivot, sprite, square, collider);
112 global.position = (p ? p->value : glm::vec2(0.0f, 0.0f)) + AnchorScaleShift(anchor, scale);
113 global.scale = scale;
114 global.rotation = r ? r->value : 0.0f;
115 global.pivot = anchor * scale;
116 },
117 kFlatSerialBelowEntities);
118 }

120 // A normalized pivot resolved against the entity's own size. No pivot means the centre, which
121 // is what the default (0.5, 0.5) resolves to anyway, so the two cases share one expression.
122 static glm::vec2 ResolveAnchor(const PivotComponent* pivot, const SpriteComponent* sprite,
123 const SquarePrimitiveComponent* square, const BoxColliderComponent* collider) {
124 const glm::vec2 normalized = pivot ? pivot->value : glm::vec2(PivotComponent::kDefaultX, PivotComponent::kDefaultY);
125 return octarine::LocalSize(sprite, square, collider) * normalized;
126 }
127
128 // Scaling about the anchor is the same as sliding the origin by this, then scaling about the
129 // origin. Every consumer reads `position` as the top-left and multiplies its own offsets by
130 // `scale`, so folding it in here fixes the sprite quad, primitive quad, and collider at once.
131 static glm::vec2 AnchorScaleShift(const glm::vec2 anchor, const glm::vec2 scale) {
132 return anchor * (glm::vec2(1.0f, 1.0f) - scale);
133 }

135 // Hierarchy members are re-composed in breadth-first depth order. Every entity at depth N is
136 // written before any entity at depth N+1 is read, so a parent's global is always final by the
137 // time its children consume it — no recursion, no per-entity stack, and no revisiting.
138 void UpdateHierarchical(const Registry* registry) {
139 PROFILE_NAMED_SCOPE("TransformSystem: Hierarchical");
140 RebuildLevelsIfStale(registry);
141 Descend(registry);
142 }
143
144 // The depth buckets depend only on hierarchy topology, which the registry versions for us, so
145 // they survive across frames and are rebuilt only when a SetParent / detach / blam changes the
146 // shape. Entity locations are deliberately not cached — an archetype transition moves an
147 // entity without touching the topology, so those are resolved fresh each frame.
148 //
149 // Holding entity handles across frames is safe here on two counts. Every entity that can land
150 // in a bucket either has a parent or has children, so destroying one always sets the
151 // registry's hierarchyMutated flag and bumps the generation. And destruction is deferred to
152 // the end of Update, after every system has run, so a bucket can never go stale mid-pass —
153 // which matters because GetEntityLocation keys on the 32-bit id and would happily resolve a
154 // recycled handle to whatever entity now occupies that slot.
155 void RebuildLevelsIfStale(const Registry* registry) {
156 const uint64_t generation = registry->HierarchyGeneration();
157 if (levelsBuilt_ && generation == cachedHierarchyGeneration_) return;
158
159 PROFILE_NAMED_SCOPE("TransformSystem: Rebuild levels");
160 levels_.clear();
161 children_.clear();
162 visited_.clear();
163
164 std::vector<FrontierNode> frontier;
165 registry->ForEachHierarchyRoot([&](const Entity root) {
166 if (visited_.insert(root.id).second) frontier.push_back({root, kNoParentSlot});
167 });
168
169 while (!frontier.empty()) {
170 frontier = BuildLevel(registry, frontier);
171 }
172
173 // Written at depth N before any depth N+1 group reads it, so it never needs clearing.
174 childGlobals_.resize(children_.size());
175
176 cachedHierarchyGeneration_ = generation;
177 levelsBuilt_ = true;
178 }

180 // Append one depth bucket for the given parents and return the next frontier (their children).
181 std::vector<FrontierNode> BuildLevel(const Registry* registry, const std::vector<FrontierNode>& parents) {
182 std::vector<ParentGroup> level;
183 std::vector<FrontierNode> next;
184
185 for (const FrontierNode& parent : parents) {
186 const auto first = static_cast<uint32_t>(children_.size());
187 registry->ForEachChild(parent.entity, [&](const Entity child) {
188 // A SetParent cycle would otherwise loop forever here. visited_ also keeps an entity
189 // reachable by two routes from being composed twice.
190 if (!visited_.insert(child.id).second) return;
191 next.push_back({child, static_cast<uint32_t>(children_.size())});
192 children_.push_back(child);
193 });
194 if (const auto count = static_cast<uint32_t>(children_.size()) - first; count > 0) {
195 level.push_back({parent.entity, parent.slot, first, count});
196 }
197 }
198
199 if (!level.empty()) levels_.push_back(std::move(level));
200 return next;
201 }

203 // Globals thread down through childGlobals_ rather than being read back out of the components,
204 // so a grouping node with no GlobalTransformComponent still passes composition to its children.
205 // Only the write-out is gated on the component existing.
206 void Descend(const Registry* registry) {
207 PROFILE_NAMED_SCOPE("TransformSystem: Descend");
208 for (const auto& level : levels_) {
209 for (const ParentGroup& group : level) {
210 const GlobalTransform parentGlobal = group.parentSlot == kNoParentSlot ? LoadRootGlobal(registry, group.parent)
211 : childGlobals_[group.parentSlot];
212 // Every child in the group shares this one parent, so its cos/sin pair is resolved once
213 // per family rather than once per child. This is the payoff for grouping by parent: a
214 // hundred-child emitter costs one trig pair instead of a hundred.
215 const octarine::Rotation2D parentRotation = octarine::Rotation2D::FromRadians(parentGlobal.rotation);
216 for (uint32_t i = 0; i < group.childCount; ++i) {
217 const uint32_t slot = group.firstChild + i;
218 childGlobals_[slot] = ComposeChild(registry, children_[slot], parentGlobal, parentRotation);
219 }
220 }
221 }
222 }

224 GlobalTransform ComposeChild(const Registry* registry, const Entity child, const GlobalTransform& parentGlobal,
225 const octarine::Rotation2D parentRotation) const {
226 const auto [archetype, chunkIdx, indexInChunk] = registry->GetEntityLocation(child);
227 if (!archetype) return {};
228 const GlobalTransform global = Compose(parentGlobal, parentRotation, LoadLocal(archetype, chunkIdx, indexInChunk));
229 WriteGlobal(archetype, chunkIdx, indexInChunk, global);
230 return global;
231 }

233 // Whatever the flat pass finalised. A root with no GlobalTransformComponent was never in that
234 // query, so its local stands in — otherwise a grouping node's own transform would be dropped.
235 GlobalTransform LoadRootGlobal(const Registry* registry, const Entity root) const {
236 const auto [archetype, chunkIdx, indexInChunk] = registry->GetEntityLocation(root);
237 if (!archetype) return {};
238 if (!archetype->HasComponent(globalEntity_.GetId())) {
239 return FromLocal(LoadLocal(archetype, chunkIdx, indexInChunk));
240 }
241 const auto* gArr = archetype->GetComponentArray<GlobalTransformComponent>(chunkIdx, globalEntity_.GetId());
242 const auto& g = gArr[indexInChunk];
243 return {g.position, g.scale, g.rotation, g.pivot};
244 }

247 static GlobalTransform FromLocal(const LocalTransform& local) {
248 return {local.position + AnchorScaleShift(local.anchor, local.scale), local.scale, local.rotation,

294 // Compose parent x local into world space. The parent's scale and rotation act about the
295 // parent's own pivot — the same point the renderer spins its quad around — so a child orbits
296 // where the parent visibly turns. A zero pivot collapses this to a plain parent-origin orbit,
297 // which is what the pre-pivot engine did.
298 //
299 // Known limit: decomposed TRS is not closed under non-uniform scale combined with rotation.
300 // A parent with scale (2, 1) rotated 45 degrees should shear its children; storing scale as a
301 // vector cannot express that, so the shear is dropped. Representing it would require a full
302 // affine matrix, which SDL_RenderTextureRotated cannot consume anyway.
303 static GlobalTransform Compose(const GlobalTransform& parent, const octarine::Rotation2D parentRotation,
304 const LocalTransform& local) {
305 const glm::vec2 scale = parent.scale * local.scale;
306 const glm::vec2 pivotWorld = parent.position + parent.pivot;
307 // Place the anchor, not the corner: the anchor offset is a vector in the child's own frame,
308 // so it has to travel through the parent's scale and rotation like everything else. Composing
309 // the corner and adding an unrotated shift afterwards breaks at any parent angle or scale.
310 const glm::vec2 anchorWorld = octarine::RotateAround(
311 parent.position + ((local.position + local.anchor) * parent.scale), pivotWorld, parentRotation);
312 return {
313 // Backing off by the scaled anchor puts the renderer's `position + pivot` on the anchor.
314 .position = anchorWorld - (local.anchor * scale),
315 .scale = scale,
316 .rotation = parent.rotation + local.rotation,
317 // Pivot is per-entity, not inherited: a child spins about its own anchor.
318 .pivot = local.anchor * scale,
319 };
320 }

322 void WriteGlobal(Archetype* archetype, size_t chunkIdx, size_t indexInChunk, const GlobalTransform& g) const {
323 // Hierarchy members are visited via the topology regardless of components, so a parented
324 // entity without a GlobalTransformComponent (e.g. a UI node) reaches here. Guard before the
325 // array fetch — GetComponentArray asserts HasComponent, it does not return null.
326 if (!archetype->HasComponent(globalEntity_.GetId())) return;
327 auto* gArray = archetype->GetComponentArray<GlobalTransformComponent>(chunkIdx, globalEntity_.GetId());
328 if (!gArray) return;
329 auto& global = gArray[indexInChunk];
330 global.position = g.position;
331 global.scale = g.scale;
332 global.rotation = g.rotation;
333 global.pivot = g.pivot;
334 }
