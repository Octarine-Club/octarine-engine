31// One resolved, loadable asset. Produced by AssetCatalog::Build from a discovered file plus its
32// (optional) sidecar metadata. Holds no SDL/MIX handle — actually loading the asset is
33// AssetManager's job (it consults these entries via Acquire).
34struct CatalogEntry {
35 AssetType type{};
36 std::string id; // resolved unique id (filename stem, or meta.id; fonts append "-<size>")
37 std::string fullPath; // absolute path to the source file on disk
38 // Texture
39 std::optional<ScaleMode> scaleMode;
40 std::optional<std::string> atlas; // sidecar `meta.atlas = <group>` (raw group name)
41 bool noAtlas{false}; // sidecar `meta.no_atlas = true` (escape hatch, see AssetMetadata.h)
42 // Resolved atlas membership — populated by TextureAtlasBaker after the scan. When set, this entry
43 // is rendered as a slice of `atlasId`'s texture instead of its own. Both fields move
44 // through asset_manifest.lua so shipped builds skip the bake.
45 std::optional<std::string> atlasId; // catalog id of the packed atlas texture (e.g. "__atlas_main")
46 std::optional<SDL_FRect> atlasSlice; // pixel-rect of this asset inside the atlas
47 // Font — one entry per size
48 float fontSize{0.0F};
49 // Audio
50 bool stream{false};
51 bool normalize{false};
52};

54// Directory-discovery + sidecar-parsing catalog: maps asset id -> CatalogEntry. Loads nothing;
55// it is the engine's index of what assets exist and how each should be loaded.
56class AssetCatalog {
57 public:
58 // Populate the id -> entry map for the project at `basePath`. When `allowManifest` is true and a
59 // baked `asset_manifest.lua` is present at the root, load it and skip the filesystem walk — assets
60 // resolve with no scan, which is what makes read-only APK/.app bundles work. Otherwise (dev build,
61 // or no manifest) recursively scan the convention folders and parse `<file>.meta` sidecars via the
62 // borrowed `lua` state. `allowManifest` is the dev-vs-shipped gate (OCTARINE_SHIPPED / the
63 // `--use-manifest` dev override): when false the manifest probe is skipped entirely, so a stray
64 // manifest beside dev assets can never hijack the scan. Returns false if any id collision (scan)
65 // or malformed entry (manifest) was found. The GameConfig overload pulls the project default scale
66 // mode from config; the explicit overload takes it directly (used by tests, which avoids
67 // GameConfig's SDL-backed load path).
68 bool Build(const std::string& basePath, sol::state& lua, const GameConfig& gameConfig, bool allowManifest = true);
69 bool Build(const std::string& basePath, sol::state& lua, std::optional<ScaleMode> defaultScaleMode,
70 bool allowManifest = true);
71
72 // Force a live filesystem scan, ignoring any baked manifest at the root. `Build` delegates here
73 // when no manifest is present; the bake step calls it directly so it always re-derives the
74 // catalog from the real files (rather than round-tripping a stale manifest). Returns false on an
75 // id collision. Sets IsFromManifest() to false.
76 bool ScanFilesystem(const std::string& basePath, sol::state& lua, const GameConfig& gameConfig);
77 bool ScanFilesystem(const std::string& basePath, sol::state& lua, std::optional<ScaleMode> defaultScaleMode);

79 // Load a baked manifest directly (the branch `Build` takes when `asset_manifest.lua` exists).
80 // Each entry's relative `file` is re-rooted against `basePath` so the stored `fullPath` resolves
81 // the same way the directory scan's does. Exposed for tests (manifest-vs-scan parity); the
82 // runtime reaches it through `Build`. Returns false on a missing/unparseable manifest or a
