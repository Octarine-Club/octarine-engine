17// Per-emitter system: gates spatial sources by listener radius. Inside the radius the
18// emitter carries AudioActiveTag and plays normally; outside, the system stashes the
19// MIX_Track's playback frame onto AudioSourceComponent, halts the track, and removes the
20// AudioSinkComponent so the track returns to the pool for some other emitter to use.
21// AudioSystem's "resumingFromCull" path consumes the stored offset on re-entry, so a
22// looping clip picks up mid-loop without an audible glitch.
23//
24// Registered BEFORE SpatialAudioSystem so sinks halted this frame don't make a wasted
25// pass through the spatial / Doppler systems. The query is (GlobalTransform, AudioSource)
26// — sink is optional because a brand-new spatial source may enter the radius before
27// AudioSystem has had a chance to spawn its sink.
28class AudioCullingSystem {
29 public:
30 void operator()(const ContextFacade& ctx, Entity entity, const GlobalTransformComponent& transform,
31 AudioSourceComponent& source) {
32 if (!source.spatial) return;
33
34 auto* registry = ctx.GetRegistry();
35 // Both singleton caches are Set once at bootstrap and never reseated; resolve them together on
36 // the first tick so the pointers stay stable for the registry's lifetime.
37 if (!cache_) {
38 cache_ = &registry->Get<AudioListenerCache>();
39 track_cache_ = &registry->Get<AudioTrackCache>();
40 }
41 if (!cache_->valid) return;
42
43 const auto& engineOptions = registry->Get<GameConfig>().GetEngineOptions();
44 const float radius = engineOptions.audioListenerRadius;
45 const float radiusSq = radius * radius;
46
47 const glm::vec2 rel = transform.position - cache_->position;
48 const float distSq = glm::dot(rel, rel);
49 const bool inRange = distSq <= radiusSq;
50 const bool isActive = registry->HasTag<AudioActiveTag>(entity);
51
52 if (inRange && !isActive) {
53 cmd_buffer_.AddTag<AudioActiveTag>(entity);
54 return;
55 }
56
57 if (!inRange && isActive) {
58 if (registry->HasComponent<AudioSinkComponent>(entity)) {
59 const auto& sink = registry->GetComponent<AudioSinkComponent>(entity);
60 if (MIX_Track* track = track_cache_->Track(entity); track && !sink.finished) {
61 // Capture position BEFORE Stop — MIX_GetTrackPlaybackPosition on a stopped
62 // track may return 0 or undefined.
63 source.playbackOffsetFrames = MIX_GetTrackPlaybackPosition(track);
64 MIX_StopTrack(track, 0);
65 }
66 // Drop the cache entry alongside the sink so the released track returns to the pool and a
67 // later re-entry (AudioSystem re-spawns the sink) starts from a clean slot.
68 track_cache_->Forget(entity);
69 cmd_buffer_.RemoveComponent<AudioSinkComponent>(entity);
70 }
71 cmd_buffer_.RemoveTag<AudioActiveTag>(entity);
72 }
73 }
74
75 CommandBuffer& GetCommandBuffer() { return cmd_buffer_; }
76
77 private:
78 const AudioListenerCache* cache_ = nullptr;
79 AudioTrackCache* track_cache_ = nullptr;
80 CommandBuffer cmd_buffer_;
81};
