135 [[nodiscard]] CellPlacement ComputeCellPlacement(const float cellX, const float cellY, const float cellW,
136 const float cellH, const GridVisibleRegion& reg,
137 const bool isBounded, const bool isFixed) const {
138 float x = cellX;
139 float y = cellY;
140 float w = cellW;
141 float h = cellH;
142 if (isBounded) {
143 const float x2 = std::min(x + w, reg.viewR);
144 const float y2 = std::min(y + h, reg.viewB);
145 x = std::max(x, reg.viewL);
146 y = std::max(y, reg.viewT);
147 w = x2 - x;
148 h = y2 - y;
149 if (w <= 0.0f || h <= 0.0f) {

153 const float destX = isFixed ? x : ((x - camera_.x) * zoom_);
154 const float destY = isFixed ? y : ((y - camera_.y) * zoom_);
155 const float destW = isFixed ? w : (w * zoom_);
156 const float destH = isFixed ? h : (h * zoom_);
