25 void Prepare(Registry* registry) {
26 const auto& gameConfig = registry->Get<GameConfig>();
27 const auto& cameraComp = registry->Get<CameraComponent>();
28 camera_ = cameraComp.viewport;
29 zoom_ = cameraComp.zoom > 0.001f ? cameraComp.zoom : 1.0f;
30 assetManager_ = &registry->Get<AssetManager>();
31 renderQueue_ = &registry->Get<RenderQueue>();
32 spriteCache_ = &registry->Get<SpriteRenderCache>();
33 windowWidth_ = static_cast<float>(gameConfig.windowWidth);
34 windowHeight_ = static_cast<float>(gameConfig.windowHeight);
35
36 // Single-threaded cache warm: walk every sprite entity once and resolve the SDL_Texture* +
37 // atlas slice now. operator() (which runs in parallel via ParallelForEach) then does a
38 // race-free read-only Lookup. SpriteRenderCache is an std::unordered_map and cannot tolerate
39 // concurrent inserts.
40 if (!warmingQuery_) {
41 warmingQuery_ = registry->CreateQuery<SpriteComponent>();
42 }
43 warmingQuery_->Update();
44 const auto assetGen = assetManager_->TextureGeneration();
45 warmingQuery_->ForEach([this, assetGen](Entity entity, const SpriteComponent& sprite) {
46 if (spriteCache_->Lookup(entity, assetGen) == nullptr) {
47 SDL_Texture* texture = assetManager_->GetTexture(sprite.assetId);
48 const auto slice = assetManager_->GetAtlasSlice(sprite.assetId);
49 const SDL_FRect atlasOffset = slice.has_value() ? *slice : SDL_FRect{0, 0, 0, 0};
50 spriteCache_->Store(entity, texture, atlasOffset, assetGen);
51 }
52 });
53
54#ifdef OCTARINE_PROFILING
55 if (!culledCounter_) culledCounter_ = PROFILE_COUNTER_HANDLE("RenderSprite: Culled");
56 if (!emplacedCounter_) emplacedCounter_ = PROFILE_COUNTER_HANDLE("RenderSprite: Emplaced");
57#endif
58 }

60 void operator()(Entity entity, const GlobalTransformComponent& transform, const SpriteComponent& sprite) const {
61 const bool isOutsideCamera = IsRenderableOutsideViewport(
62 transform.position.x, transform.position.y, sprite.width * transform.scale.x, sprite.height * transform.scale.y,
63 sprite.isFixed, camera_, windowWidth_, windowHeight_);
64
65 if (isOutsideCamera) {
66 PROFILE_COUNTER_INC(culledCounter_);
67 return;
68 }
69
70 PROFILE_COUNTER_INC(emplacedCounter_);
71
72 const SpriteRenderCache::Entry* entry = spriteCache_->Lookup(entity, assetManager_->TextureGeneration());
73 SDL_Texture* texture = entry ? entry->texture : nullptr;
74 const SDL_FRect atlasOffset = entry ? entry->atlasOffset : SDL_FRect{0, 0, 0, 0};
75
76 const float x = sprite.isFixed ? transform.position.x : (transform.position.x - camera_.x) * zoom_;
77 const float y = sprite.isFixed ? transform.position.y : (transform.position.y - camera_.y) * zoom_;
78 const float destW = sprite.isFixed ? sprite.width * transform.scale.x : sprite.width * transform.scale.x * zoom_;
79 const float destH = sprite.isFixed ? sprite.height * transform.scale.y : sprite.height * transform.scale.y * zoom_;
80
81 auto& cmd = renderQueue_->EmplaceSprite(static_cast<unsigned int>(sprite.layer), transform.position.y, texture,
82 sprite.blendMode);
83 cmd.destX = x;
84 cmd.destY = y;
85 cmd.destW = destW;
86 cmd.destH = destH;
87 // Atlas-aware src_rect compose: a loose texture leaves atlasOffset at zero (no-op add); an
88 // atlas member adds the slice origin so the sprite's logical (sx, sy) selects inside the
89 // packed atlas instead of the standalone source PNG.
90 cmd.srcRect.x = sprite.srcRect.x + atlasOffset.x;
91 cmd.srcRect.y = sprite.srcRect.y + atlasOffset.y;
92 cmd.srcRect.w = sprite.srcRect.w;
93 cmd.srcRect.h = sprite.srcRect.h;
94 cmd.rotation = transform.rotation;
95 // transform.pivot is already the scaled offset from the sprite's top-left, which is exactly
96 // the origin SDL_RenderTextureRotated rotates about.
97 cmd.pivot = sprite.isFixed ? SDL_FPoint{transform.pivot.x, transform.pivot.y}
98 : SDL_FPoint{transform.pivot.x * zoom_, transform.pivot.y * zoom_};
99 cmd.flip = static_cast<SDL_FlipMode>(sprite.flip);
100 cmd.texture = texture;
101 cmd.colorMod = SDL_Color{sprite.colorMod.r, sprite.colorMod.g, sprite.colorMod.b, sprite.colorMod.a};
102 cmd.blendMode = octarine::ToSdlBlendMode(sprite.blendMode);
103 }
