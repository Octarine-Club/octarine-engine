5// DevListenServer — editor dev-iteration TCP listener (compile-time stripped from shipped builds
6// via #ifndef OCTARINE_SHIPPED). When enabled, owns a background listener thread bound to
7// 127.0.0.1:<port> and serves a tiny length-prefixed
8// binary protocol:
9//
10// each message: uint32 LE length | uint32 LE op | body[length-4]
11// hello reply: op=Hello, body = "OCTARINE_DEV_LISTEN/v1\n"
12// ping reply: op=Pong, body = client nonce echoed back
13//
14// Pure-I/O ops (Hello/Ping) reply directly on the listener thread. The engine-mutating ops
15// (EvalLua / ReloadScene / PushScript / PushAsset) cannot touch the Registry / sol::state /
16// AssetManager off the main thread, so the listener parks each one on a queue and blocks until
17// the main thread drains it in Pump(); Pump runs the command and hands the reply back so the
18// listener can write it and close. See `kPush*` body layout on each opcode below.
19

27enum class OpCode : std::uint32_t {
28 Hello = 1, // body: empty -> reply Hello with banner
29 Ping = 2, // body: u64 LE nonce -> reply Pong with same nonce
30 // Engine-mutating ops, parked by the listener and dispatched to the main thread via Pump():
31 EvalLua = 10, // body: lua source -> reply { u8 ok, utf8 result/error }
32 ReloadScene = 11, // body: empty -> re-runs the active scene; reply { u8 ok }
33 PushScript = 12, // body: u32 path_len | path | bytes -> writes script + forces reload; reply { u8 ok }
34 PushAsset = 13, // body: u32 path_len | path | bytes -> writes asset + reloads it; reply { u8 ok }
35 Step = 20, // body: [u32 ticks][float dt] or empty -> advances simulation; reply { u8 ok }
36 Pause = 21, // body: [u8 paused] or empty to query -> sets/queries pause; reply { u8 ok, u8 is_paused }
37 Inspect = 22, // body: [query_mode] or empty -> returns state JSON; reply { u8 ok, utf8 json }
38};

40// Handler the engine installs to apply an engine-mutating op on the main thread. Invoked from
41// Pump() for each parked command; receives the op + raw body and fills reply_op/reply_body (the
42// frame written back to the client). Keeping this a std::function is what lets the transport layer
43// carry zero engine dependencies — the Registry/sol/AssetManager-touching logic lives in
44// DevListenDispatch (engine layer), installed via SetCommandHandler.
45using CommandHandler =
46 std::function<void(OpCode op, const std::vector<char>& body, std::uint32_t& reply_op, std::string& reply_body)>;
