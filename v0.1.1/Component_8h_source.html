174 // Swaps entity and component data between two slots in the chunk.
175 void Swap(const size_t i, const size_t j, const std::vector<size_t>& componentOffsets,
176 const std::vector<ComponentInfo>& componentInfos) const {
177 assert(componentOffsets.size() == componentInfos.size());
178 assert(i < header_.entity_count && j < header_.entity_count);
179 if (i == j) return;
180 auto* entity_array = reinterpret_cast<Entity*>(buffer_);
181 std::swap(entity_array[i], entity_array[j]);
182 for (size_t c = 0; c < componentOffsets.size(); ++c) {
183 const auto& info = componentInfos[c];
184 if (info.size == 0 || !info.swap) continue;
185 unsigned char* a = buffer_ + componentOffsets[c] + i * info.size;
186 unsigned char* b = buffer_ + componentOffsets[c] + j * info.size;
187 info.swap(a, b);
188 }
189 }

191 // Removes entity and relocates boundary entities to preserve dense storage without heap allocation.
192 ChunkLocalSwaps RemoveEntity(const size_t index, const std::vector<size_t>& componentOffsets,
193 const std::vector<ComponentInfo>& componentInfos) {
194 assert(componentOffsets.size() == componentInfos.size());
195 ChunkLocalSwaps swaps;
196 if (index >= header_.entity_count) return swaps;
197
198 const auto* entityArray = reinterpret_cast<Entity*>(buffer_);
199 const bool wasActive = index < header_.active_count;
200
201 DestroyComponentsAt(index, componentOffsets, componentInfos);
202
203 // Active-region collapse: pull the last active entity into the freed slot so the active
204 // prefix stays dense. Skip when the removed entity was already at the boundary.
205 if (wasActive && index + 1 < header_.active_count) {
206 const size_t srcIdx = header_.active_count - 1;
207 MoveEntitySlot(index, srcIdx, componentOffsets, componentInfos);
208 swaps.Push(entityArray[index], index);
209 }
210
211 if (wasActive) --header_.active_count;
212
213 // The freed slot after the active collapse is either `index` (inactive removal) or the
214 // post-decrement active_count (active removal — the slot that was the active boundary).
215 const size_t freedSlot = wasActive ? header_.active_count : index;
216 const size_t lastSlot = header_.entity_count - 1;
217
218 // Standard swap-with-end: pull the last entity (always inactive at this point) into the
219 // freed slot. Skip when there's nothing to pull.
220 if (freedSlot != lastSlot) {
221 MoveEntitySlot(freedSlot, lastSlot, componentOffsets, componentInfos);
222 swaps.Push(entityArray[freedSlot], freedSlot);
223 }
224
225 --header_.entity_count;
226 assert(header_.active_count <= header_.entity_count);
227 return swaps;
228 }

231 void DestroyComponentsAt(const size_t slot, const std::vector<size_t>& componentOffsets,
232 const std::vector<ComponentInfo>& componentInfos) const {
233 assert(slot < header_.entity_count);
234 assert(componentOffsets.size() == componentInfos.size());
235 for (size_t c = 0; c < componentOffsets.size(); ++c) {
236 const auto& info = componentInfos[c];
237 if (!info.destroy) continue;
238 unsigned char* slot_ptr = buffer_ + componentOffsets[c] + slot * info.size;
239 info.destroy(slot_ptr);
240 }
241 }

243 void MoveEntitySlot(const size_t destSlot, const size_t srcSlot, const std::vector<size_t>& componentOffsets,
244 const std::vector<ComponentInfo>& componentInfos) const {
245 assert(destSlot < header_.entity_count);
246 assert(srcSlot < header_.entity_count);
247 assert(destSlot != srcSlot);
248 assert(componentOffsets.size() == componentInfos.size());
249 auto* entityArray = reinterpret_cast<Entity*>(buffer_);
250 for (size_t c = 0; c < componentOffsets.size(); ++c) {
251 const auto& info = componentInfos[c];
252 if (info.size == 0) continue;
253 unsigned char* dest_ptr = buffer_ + componentOffsets[c] + destSlot * info.size;
254 unsigned char* source_ptr = buffer_ + componentOffsets[c] + srcSlot * info.size;
255 if (info.move_construct) {
256 info.move_construct(dest_ptr, source_ptr);
257 } else {
258 std::memcpy(dest_ptr, source_ptr, info.size);
259 }
260 if (info.destroy) info.destroy(source_ptr);
261 }
262 entityArray[destSlot] = entityArray[srcSlot];
263 }

350 // Allocation-free entity removal invoking onRelocate(Entity, chunkIndex, indexInChunk) for displaced entities.
351 template <typename RelocateFn>
352 void RemoveEntity(const EntityLocation& location, RelocateFn&& onRelocate) {
353 AssertLocation(location);
354 const size_t targetChunk = location.chunkIndex;
355 const auto localSwaps =
356 chunks_[targetChunk].RemoveEntity(location.indexInChunk, component_offsets_, component_infos_);
357
358 for (uint8_t i = 0; i < localSwaps.count; ++i) {
359 const auto& swap = localSwaps.items[i];
360 assert(swap.indexInChunk < chunks_[targetChunk].GetEntityCount());
361 onRelocate(swap.entity, targetChunk, swap.indexInChunk);
362 }
363
364 if (chunks_[targetChunk].GetEntityCount() == 0) {
365 PruneEmptyChunkAt(targetChunk, std::forward<RelocateFn>(onRelocate));
366 } else if (targetChunk < first_non_full_chunk_) {
367 first_non_full_chunk_ = targetChunk;
368 }
369 assert(retain_first_chunk_ || targetChunk >= chunks_.size() || chunks_[targetChunk].GetEntityCount() > 0);
370 }

514 size_t ComputeLayoutForCapacity(const size_t capacity) {
515 constexpr size_t kCacheLineAlignment = 64;
516 size_t current_offset = capacity * sizeof(Entity);
517 for (size_t i = 0; i < component_infos_.size(); ++i) {
518 const auto& info = component_infos_[i];
519 if (info.size == 0) {
520 component_offsets_[i] = current_offset;
521 continue;
522 }
523 const size_t align = std::max(info.alignment, kCacheLineAlignment);
524 const size_t padding = (align - (current_offset % align)) % align;
525 current_offset += padding;
526 component_offsets_[i] = current_offset;
527 current_offset += capacity * info.size;
528 }
529 return current_offset;
530 }

532 void CalculateLayout() {
533 size_t entityComponentSize = sizeof(Entity);
534 for (const auto& info : component_infos_) {
535 entityComponentSize += info.size;
536 }
537
538 if (entityComponentSize == 0) {
539 chunk_capacity_ = 0;
540 return;
541 }
542
543 chunk_capacity_ = kUsableSpace / entityComponentSize;
544 component_offsets_.resize(component_infos_.size());
545 size_t current_offset = ComputeLayoutForCapacity(chunk_capacity_);
546
547 while (chunk_capacity_ > 0 && current_offset > kUsableSpace) {
548 const size_t overflow = current_offset - kUsableSpace;
549 const size_t step = std::max<size_t>(1, overflow / entityComponentSize);
550 chunk_capacity_ = chunk_capacity_ > step ? chunk_capacity_ - step : 0;
551 current_offset = ComputeLayoutForCapacity(chunk_capacity_);
552 }
553 assert(chunk_capacity_ > 0);
554 }

562 template <typename RelocateFn>
563 void PruneEmptyChunkAt(const size_t targetChunk, RelocateFn&& onRelocate) {
564 assert(!chunks_.empty());
565 assert(targetChunk < chunks_.size());
566 assert(chunks_[targetChunk].GetEntityCount() == 0);
567 if (retain_first_chunk_ && chunks_.size() == 1) {
568 first_non_full_chunk_ = 0;
569 return;
570 }
571 const size_t lastChunkIdx = chunks_.size() - 1;
572 if (targetChunk == lastChunkIdx) {
573 chunks_.pop_back();
574 } else {
575 chunks_[targetChunk] = std::move(chunks_.back());
576 chunks_.pop_back();
577 assert(chunks_[targetChunk].GetEntityCount() > 0);
578 const auto* entityArray = chunks_[targetChunk].GetEntityArray();
579 const size_t count = chunks_[targetChunk].GetEntityCount();
580 for (size_t i = 0; i < count; ++i) {
581 onRelocate(entityArray[i], targetChunk, i);
582 }
583 }
584
585 if (first_non_full_chunk_ > chunks_.size()) {
586 first_non_full_chunk_ = chunks_.size();
587 } else if (targetChunk < first_non_full_chunk_) {
588 first_non_full_chunk_ = targetChunk;
589 }
590 assert(first_non_full_chunk_ <= chunks_.size());
591 }
