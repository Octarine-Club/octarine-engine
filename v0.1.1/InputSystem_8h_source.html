22// Per-frame input state machine: held/pressed/released sets for keys + mouse buttons, accumulated
23// mouse wheel delta, action map (named action → list of keys), and the script-installed
24// callback vectors that fire on each edge. EventBus subscriptions feed OnKeyInput / OnMouseInput
25// / OnMouseWheel; the Lua `input.*` surface (is_key_*, on_key_*, bind, etc.) lives in
26// `Lua/Bindings/InputSystemLuaBinding.cpp` — InputSystem just exposes the public accessors +
27// mutators that binding's lambdas call into.

41 static std::string MakeKey(const std::string& key) {
42 std::string lower = key;
43 std::transform(lower.begin(), lower.end(), lower.begin(),
44 [](const unsigned char c) { return static_cast<char>(std::tolower(c)); });
45 return lower;
46 }

48 void SubscribeToEvents(const std::unique_ptr<EventBus>& eventBus, Registry* registry) {

157 if (fn == sol::lua_nil) continue;

169 template <typename Pred>
170 bool AnyBoundKey(const std::string& action, Pred&& pred) const {
171 const auto it = actions_.find(MakeKey(action));
172 if (it == actions_.end()) return false;
173 for (const auto& k : it->second) {
174 if (pred(k)) return true;
175 }
176 return false;
177 }
