7// Per-file asset metadata, parsed from an optional `<file>.meta` Lua sidecar. A missing sidecar
8// means "all defaults". Each struct owns an `applyDefaults(...)` that fills any unset field from
9// project-wide defaults (GameConfig) and the constants below, so downstream code always sees a
10// fully-resolved struct. These types depend on nothing engine-specific (no SDL) so they can be
11// unit-tested in isolation.
12
13// Texture scale filtering. Mirrors SDL_ScaleMode but kept engine-side so the metadata layer does
14// not pull in SDL headers. Converted to the engine's string form via toScaleModeString.
15enum class ScaleMode { Nearest, Linear };
16
17// Glyph size baked when a font sidecar omits `sizes`.
18inline constexpr float kDefaultFontSize = 16.0F;
19
20inline const char* toScaleModeString(ScaleMode mode) { return mode == ScaleMode::Linear ? "linear" : "nearest"; }

22struct TextureMeta {
23 std::optional<std::string> id; // override the derived id (filename stem)
24 std::optional<ScaleMode> scaleMode; // nearest/linear; falls back to the project default
25 std::optional<std::string> atlas; // atlas group name ; nullopt = loose texture
26 bool noAtlas{false}; // meta.no_atlas = true ; never folded into an atlas even if
27 // a sibling pulls the group via atlas=. For pixel-art that
28 // hates packing bleed at sub-pixel src_rect math.
29
30 // `defaultScaleMode` is the GameConfig DefaultScalingMode (nullopt -> nearest). An explicit
31 // sidecar value always wins.
32 void applyDefaults(std::optional<ScaleMode> defaultScaleMode) {
33 if (!scaleMode.has_value()) {
34 scaleMode = defaultScaleMode.value_or(ScaleMode::Nearest);
35 }
36 }
37};
