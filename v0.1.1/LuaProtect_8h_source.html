7// XOR decryption for Lua scripts protected at package time by OctarinePackage.cmake's
8// OCTARINE_LUA_XOR_KEY step. Only active when the key is compiled in — shipped builds with
9// OCTARINE_PROTECT_SCRIPTS=ON and a non-empty key. No-op in dev builds (the define is absent).
10//
11// Encrypted payloads are prefixed at package time with a 4-byte magic sentinel (0x1B 'O' 'C' 'T').
12// DecryptLuaBytes() only transforms files carrying that sentinel: it strips the prefix and reverses
13// the rolling XOR (key at position i is (base_key + i) % 256). Files WITHOUT the sentinel — plain
14// Lua source, stock \x1bLua bytecode, or any future un-encrypted payload such as user mods — pass
15// through untouched, so decryption stays robust even when only part of the script surface is
16// encrypted. base_key is recoverable from the known Lua header, so this is obfuscation against
17// automated tooling, not cryptographic protection.
18inline void DecryptLuaBytes(std::string& bytes) {
19#if defined(OCTARINE_LUA_XOR_KEY)
20 static constexpr unsigned char kMagic[] = {0x1B, 'O', 'C', 'T'};
21 constexpr std::size_t kMagicLen = sizeof(kMagic);
22 if (bytes.size() < kMagicLen || std::memcmp(bytes.data(), kMagic, kMagicLen) != 0) {
23 return; // not an Octarine-encrypted payload (plain source / stock bytecode / mods) — leave as-is
24 }
25 bytes.erase(0, kMagicLen);
26 constexpr auto key = static_cast<uint8_t>(OCTARINE_LUA_XOR_KEY);
27 for (std::size_t i = 0; i < bytes.size(); ++i) bytes[i] ^= static_cast<uint8_t>((key + i) & 0xFF);
28#else
29 (void)bytes;
30#endif
31}
