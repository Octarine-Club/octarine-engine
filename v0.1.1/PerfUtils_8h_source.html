24// Returns the value at the given percentile (0..1) of `samples`, partitioning in-place.
25inline float GetPercentileInPlace(std::vector<float>& samples, const float percentile) {
26 assert(percentile >= 0 && percentile <= 1);
27 if (samples.empty()) {
28 return 0.0f;
29 }
30 const auto elementNum =
31 std::min(static_cast<size_t>(static_cast<float>(samples.size()) * percentile), samples.size() - 1);
32 std::ranges::nth_element(samples, samples.begin() + static_cast<std::ptrdiff_t>(elementNum));
33 return samples[elementNum];
34}

37inline float GetPercentile(const std::vector<float>& samples, const float percentile) {
38 if (samples.empty()) {
39 return 0.0f;
40 }
41 std::vector<float> scratchpad(samples);
42 return GetPercentileInPlace(scratchpad, percentile);
43}

150// Frame-scoped integer counters, separate from duration-based profiling.
151// Counters support atomic accumulation (Add) or single-writer snapshots (Set).
152// ResetValues() clears values per frame while preserving allocations, allowing
153// callers to cache counter pointers across frames.
154class PerfCounters {
155 public:
156 static std::atomic<long long>* GetOrCreate(const std::string& name) {
157 {
158 std::lock_guard lock(s_mutex);
159 if (const auto it = s_counters.find(name); it != s_counters.end()) {
160 return it->second.get();
161 }
162 auto slot = std::make_unique<std::atomic<long long>>(0);
163 auto* raw = slot.get();
164 s_counters.emplace(name, std::move(slot));
165 return raw;
166 }
167 }
168
169 static void Add(const std::string& name, const long long delta = 1) {
170 GetOrCreate(name)->fetch_add(delta, std::memory_order_relaxed);
171 }
172
173 static void Set(const std::string& name, const long long value) {
174 GetOrCreate(name)->store(value, std::memory_order_relaxed);
175 }
176
177 // Zero every registered counter without destroying the slots. Called once per frame so
178 // accumulated values are per-frame, while pointer handles cached by hot-path callers stay
179 // valid across frames.
180 static void ResetValues() {
181 std::lock_guard lock(s_mutex);
182 for (const auto& [name, counter] : s_counters) {
183 counter->store(0, std::memory_order_relaxed);
184 }
185 }
186
187 static void Clear() {
188 std::lock_guard lock(s_mutex);
189 s_counters.clear();
190 }
191
192 static std::map<std::string, long long> GetCounters() {
193 std::lock_guard lock(s_mutex);
194 std::map<std::string, long long> result;
195 for (const auto& [name, counter] : s_counters) {
196 result[name] = counter->load(std::memory_order_relaxed);
197 }
198 return result;
199 }
200
201 // Emit each counter as `COUNTER: name: value` on the logger. Intended to be called once per
202 // frame so bench output gets the same per-frame granularity as `TIMER:` lines.
203 static void Report() {
204 std::lock_guard lock(s_mutex);
205 for (const auto& [name, counter] : s_counters) {
206 Logger::Info("COUNTER: " + name + ": " + std::to_string(counter->load(std::memory_order_relaxed)));
207 }
208 }
209
210 private:
211 inline static std::mutex s_mutex;
212 inline static std::map<std::string, std::unique_ptr<std::atomic<long long>>> s_counters;
213};
