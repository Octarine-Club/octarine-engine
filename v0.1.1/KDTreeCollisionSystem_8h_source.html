31struct Box {
32 Entity entity{};
33 EntityMask entityMask{};
34 EntityMask collisionMask{};
35 // AABB enclosing the (possibly rotated) OBB — used by the median-cut broadphase.
36 float minX{0.0f};
37 float minY{0.0f};
38 float maxX{0.0f};
39 float maxY{0.0f};
40 // OBB narrowphase state.
41 float cx{0.0f};
42 float cy{0.0f};
43 float hx{0.0f};
44 float hy{0.0f};
45 octarine::Rotation2D rot{};
46
47 [[nodiscard]] bool intersectsInDimension(const Box& other, const int dim) const {
48 if (dim == 0) return !(maxX < other.minX || minX > other.maxX);
49 if (dim == 1) return !(maxY < other.minY || minY > other.maxY);
50 return false;
51 }
52
53 // SAT on the 4 face normals of two OBBs. Only invoked when at least one box is rotated;
54 // axis-aligned pairs short-circuit on the AABB check above.
55 [[nodiscard]] bool obbIntersects(const Box& other) const {
56 const float ax0 = rot.cos, ay0 = rot.sin;
57 const float ax1 = -rot.sin, ay1 = rot.cos;
58 const float bx0 = other.rot.cos, by0 = other.rot.sin;
59 const float bx1 = -other.rot.sin, by1 = other.rot.cos;
60 const float dx = other.cx - cx;
61 const float dy = other.cy - cy;
62
63 const float axes[4][2] = {{ax0, ay0}, {ax1, ay1}, {bx0, by0}, {bx1, by1}};
64 return std::ranges::all_of(axes, [&](const auto& axis) {
65 const float ux = axis[0];
66 const float uy = axis[1];
67 const float aProj = hx * std::abs(ax0 * ux + ay0 * uy) + hy * std::abs(ax1 * ux + ay1 * uy);
68 const float bProj = other.hx * std::abs(bx0 * ux + by0 * uy) + other.hy * std::abs(bx1 * ux + by1 * uy);
69 const float distProj = std::abs(dx * ux + dy * uy);
70 return distProj <= aProj + bProj;
71 });
72 }
73
74 [[nodiscard]] bool intersects(const Box& other) const {
75 const bool canInteract = !(collisionMask & other.entityMask).none() || !(other.collisionMask & entityMask).none();
76 if (!canInteract) {
77 return false;
78 }
79 if (!intersectsInDimension(other, 0) || !intersectsInDimension(other, 1)) return false;
80 if (rot.IsIdentity() && other.rot.IsIdentity()) return true;
81 return obbIntersects(other);
82 }
83};

102 static Box BuildBox(Entity entity, const GlobalTransformComponent& transform, const BoxColliderComponent& collider,
103 const EntityMaskComponent& entityMask) {
104 const float w = static_cast<float>(collider.width) * transform.scale.x;
105 const float h = static_cast<float>(collider.height) * transform.scale.y;
106 const float hx = w * 0.5f;
107 const float hy = h * 0.5f;
108
109 // transform.position is top-left. apply collider offset (scaled).
110 const float boxCx = transform.position.x + collider.offset.x * transform.scale.x + hx;
111 const float boxCy = transform.position.y + collider.offset.y * transform.scale.y + hy;
112
113 // FromRadians short-circuits the unrotated majority, and RotateAround collapses to a
114 // copy for identity — so a world of axis-aligned colliders pays no trig at all here.
115 const octarine::Rotation2D rot = octarine::Rotation2D::FromRadians(transform.rotation);
116 const glm::vec2 centre = octarine::RotateAround({boxCx, boxCy}, transform.position + transform.pivot, rot);
117 const glm::vec2 aabbHalf = octarine::RotatedHalfExtents({hx, hy}, rot);
118
119 return {.entity = entity,
120 .entityMask = entityMask.mask,
121 .collisionMask = collider.collisionMask,
122 .minX = centre.x - aabbHalf.x,
123 .minY = centre.y - aabbHalf.y,
124 .maxX = centre.x + aabbHalf.x,
125 .maxY = centre.y + aabbHalf.y,
126 .cx = centre.x,
127 .cy = centre.y,
128 .hx = hx,
129 .hy = hy,
130 .rot = rot};
131 }

220 struct PairHash {
221 size_t operator()(const std::pair<EntityID, EntityID>& p) const noexcept {
222 // Boost-style hash_combine: golden-ratio constant spreads entropy, shifts mix bits.
223 static constexpr size_t kGoldenRatio = 0x9e3779b9ULL;
224 static constexpr int kLeftShift = 6;
225 size_t h = std::hash<EntityID>{}(p.first);
226 h ^= std::hash<EntityID>{}(p.second) + kGoldenRatio + (h << kLeftShift) + (h >> 2);
227 return h;
228 }
229 };
230 using PairSet = std::unordered_set<std::pair<EntityID, EntityID>, PairHash>;

232 PairSet prevPairSet_;
233 std::vector<Box> cachedBoxes_;
234 std::future<CollisionResult> collisionResult_;

238 // Diff this frame's overlaps against the previous frame and emit the enter/exit batches.
239 // - enter: pairs overlapping now but not last frame — first-contact only, so persistent
240 // overlaps (enemy pinned against a wall, stacked entities) don't re-fire every frame.
241 // - exit: pairs that overlapped last frame but no longer do.
242 void EmitCollisionEvents(EventBus* eventBus, const std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
243 PROFILE_COUNTER_SET("Collision: Intersecting pairs", static_cast<long long>(intersectingPairs.size()));
244
245 PairSet currentSet;
246 currentSet.reserve(intersectingPairs.size());
247 for (const auto& [a, b] : intersectingPairs) {
248 currentSet.emplace(std::min(a.id, b.id), std::max(a.id, b.id));
249 }
250
251 std::vector<std::pair<Entity, Entity>> enteringPairs;
252 enteringPairs.reserve(intersectingPairs.size());
253 for (const auto& [a, b] : intersectingPairs) {
254 if (!prevPairSet_.contains({std::min(a.id, b.id), std::max(a.id, b.id)})) {
255 enteringPairs.emplace_back(a, b);
256 }
257 }
258
259 std::vector<std::pair<Entity, Entity>> exitingPairs;
260 for (const auto& [minId, maxId] : prevPairSet_) {
261 if (!currentSet.contains({minId, maxId})) {
262 exitingPairs.emplace_back(Entity{minId}, Entity{maxId});
263 }
264 }
265
266 prevPairSet_ = std::move(currentSet);
267
268 PROFILE_COUNTER_SET("Collision: Entering pairs", static_cast<long long>(enteringPairs.size()));
269 eventBus->EmitEvent<CollisionBatchEvent>(enteringPairs);
270 eventBus->EmitEvent<CollisionExitBatchEvent>(exitingPairs);
271 }

273 static void BruteForceHugeBoxes(const std::vector<Box>& boxes, const std::vector<Box>::iterator& hugeBoxesBegin,
274 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
275 ACCUMULATE_PROFILE_SCOPE("Huge Box Brute Force");
276 for (auto hugeIt = hugeBoxesBegin; hugeIt != boxes.end(); ++hugeIt) {
277 for (auto normalIt = boxes.begin(); normalIt != hugeBoxesBegin; ++normalIt) {
278 if (hugeIt->intersects(*normalIt)) {
279 intersectingPairs.emplace_back(hugeIt->entity, normalIt->entity);
280 }
281 }
282 for (auto otherHugeIt = hugeIt + 1; otherHugeIt != boxes.end(); ++otherHugeIt) {
283 if (hugeIt->intersects(*otherHugeIt)) {
284 intersectingPairs.emplace_back(hugeIt->entity, otherHugeIt->entity);
285 }
286 }
287 }

290 static void ComputeAsyncCollisions(std::vector<Box>& boxes,
291 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
292 AGGREGATE_PROFILE_SESSION("Async Box Creation");
293 if (boxes.empty()) return;
294
295 ACCUMULATE_PROFILE_SCOPE("Huge Box Extraction");
296
297 // Dynamically calculate the huge box threshold to be scale-invariant.
298 // Defines a "huge outlier" as a box >= 4x the average size.
299 double sum = 0.0;
300 for (const auto& b : boxes) {
301 sum += std::max(b.maxX - b.minX, b.maxY - b.minY);
302 }
303 const double mean = sum / static_cast<double>(boxes.size());
304 const auto hugeThreshold = static_cast<float>(mean * 4.0);
305 // Extract any box larger than the dynamic threshold. These are typically background bounds or map
306 // triggers that cross the entire level. Putting them in the KD-Tree destroys the median
307 // splits and breaks the Sweep Bipartite early-outs.
308 const auto hugeBoxesBegin = std::ranges::partition(boxes, [hugeThreshold](const Box& b) {
309 return (b.maxX - b.minX) <= hugeThreshold && (b.maxY - b.minY) <= hugeThreshold;
310 }).begin();
311
312 const int normalCount = static_cast<int>(std::distance(boxes.begin(), hugeBoxesBegin));
313 if (normalCount > 0) {
314 FindIntersectionsRecursive(boxes, 0, normalCount, 0, 0, intersectingPairs);
315 }
316
317 if (hugeBoxesBegin != boxes.end()) {
318 BruteForceHugeBoxes(boxes, hugeBoxesBegin, intersectingPairs);
319 }
320 }

322 [[nodiscard]] static std::future<CollisionResult> StartAsyncCollisionDetection(std::vector<Box> boxes) {
323 // Run on the persistent worker pool rather than std::async(launch::async), which spawns and
324 // tears down an OS thread per cycle. The promise is shared because ThreadPool::Submit takes a
325 // copyable std::function; the future hands back to the same valid()/wait_for/get polling below.
326 auto promise = std::make_shared<std::promise<CollisionResult>>();
327 std::future<CollisionResult> result = promise->get_future();
328 ThreadPool::Instance().Submit([boxes = std::move(boxes), promise]() mutable {
329 std::vector<std::pair<Entity, Entity>> intersectingPairs;
330 ComputeAsyncCollisions(boxes, intersectingPairs);
331 promise->set_value(CollisionResult{std::move(intersectingPairs), std::move(boxes)});
332 });
333 return result;
334 }

336 static void FindIntersectionsBruteForce(const std::vector<Box>& boxes, const int begin, const int end,
337 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
338 ACCUMULATE_PROFILE_SCOPE("Brute Force Intersection");
339
340 EntityMask entityUnion = 0;
341 EntityMask collUnion = 0;
342 for (int i = begin; i < end; ++i) {
343 entityUnion |= boxes[static_cast<size_t>(i)].entityMask;
344 collUnion |= boxes[static_cast<size_t>(i)].collisionMask;
345 }
346 if ((entityUnion & collUnion).none()) {
347 return;
348 }
349
350 int checks = 0;
351 for (int i = begin; i < end; ++i) {
352 for (int j = i + 1; j < end; ++j) {
353 checks++;
354 const auto& bi = boxes[static_cast<size_t>(i)];
355 const auto& bj = boxes[static_cast<size_t>(j)];
356 if (bi.intersects(bj)) {
357 intersectingPairs.emplace_back(bi.entity, bj.entity);
358 }
359 }
360 }
361#ifdef OCTARINE_PROFILING
362 PROFILE_COUNTER_ADD("Collision: Brute Force Checks", checks);
363#else
364 (void)checks;
365#endif
366 }

368 [[nodiscard]] static Partitions PartitionBoxes(std::vector<Box>& boxes, const int begin, const int end,
369 const int dimension, const float medianValue) {
370 ACCUMULATE_PROFILE_SCOPE("Partition Boxes");
371 const auto first = boxes.begin() + begin;
372 const auto last = boxes.begin() + end;
373
374 const auto leftEndIt = std::partition(first, last, [&](const Box& box) {
375 if (dimension == 0) {
376 return box.maxX < medianValue;
377 }
378
379 return box.maxY < medianValue;
380 });
381
382 const auto rightBeginIt = std::partition(leftEndIt, last, [&](const Box& box) {
383 if (dimension == 0) {
384 return box.minX <= medianValue;
385 }
386
387 return box.minY <= medianValue;
388 });
389
390 const int leftEnd = static_cast<int>(std::distance(boxes.begin(), leftEndIt));
391
392 const int rightStart = static_cast<int>(std::distance(boxes.begin(), rightBeginIt));
393
394 return Partitions{leftEnd, rightStart};
395 }

419 // KD-Tree inherently requires recursion and dense Bipartite bounds-checking logic.
420 // NOLINTNEXTLINE(misc-no-recursion,readability-function-cognitive-complexity)
421 static void FindIntersectionsSweepBipartite(std::vector<Box>& boxes, const int begin1, const int end1,
422 const int begin2, const int end2, const int dimension,
423 std::vector<std::pair<Entity, Entity>>& pairs) {
424 ACCUMULATE_PROFILE_SCOPE("Sweep Bipartite");
425
426 if (end1 - begin1 == 0 || end2 - begin2 == 0) return;
427
428 // Aggregate Mask Pruning: Skips bipartite sweep entirely if the union of entity and collision masks proves no
429 // interaction is possible.
430 EntityMask entity1 = 0;
431 EntityMask coll1 = 0;
432 for (int i = begin1; i < end1; ++i) {
433 entity1 |= boxes[static_cast<size_t>(i)].entityMask;
434 coll1 |= boxes[static_cast<size_t>(i)].collisionMask;
435 }
436
437 EntityMask entity2 = 0;
438 EntityMask coll2 = 0;
439 for (int i = begin2; i < end2; ++i) {
440 entity2 |= boxes[static_cast<size_t>(i)].entityMask;
441 coll2 |= boxes[static_cast<size_t>(i)].collisionMask;
442 }
443
444 const bool groupCanInteract = !(coll1 & entity2).none() || !(coll2 & entity1).none();
445 if (!groupCanInteract) {
446 return;
447 }
448
449 if ((end1 - begin1) * (end2 - begin2) <= kBruteforceCutoff * kBruteforceCutoff) {
450 FindIntersectionsBruteForceBipartite(boxes, begin1, end1, begin2, end2, pairs);
451 return;
452 }
453
454 {
455 ACCUMULATE_PROFILE_SCOPE("Bipartite Sort Phase");
456 if (dimension == 0) {
457 std::sort(boxes.begin() + begin1, boxes.begin() + end1,
458 [](const Box& a, const Box& b) { return a.minX < b.minX; });
459 std::sort(boxes.begin() + begin2, boxes.begin() + end2,
460 [](const Box& a, const Box& b) { return a.minX < b.minX; });
461 } else {
462 std::sort(boxes.begin() + begin1, boxes.begin() + end1,
463 [](const Box& a, const Box& b) { return a.minY < b.minY; });
464 std::sort(boxes.begin() + begin2, boxes.begin() + end2,
465 [](const Box& a, const Box& b) { return a.minY < b.minY; });
466 }
467 }
468
469 {
470 ACCUMULATE_PROFILE_SCOPE("Bipartite Sweep Phase");
471 int sweepChecks = 0;
472 if (dimension == 0) {
473 int startJ = begin2;
474 for (int i = begin1; i < end1; ++i) {
475 const Box& a = boxes[static_cast<size_t>(i)];
476 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxX < a.minX) ++startJ;
477 for (int j = startJ; j < end2; ++j) {
478 sweepChecks++;
479 const Box& b = boxes[static_cast<size_t>(j)];
480 if (b.minX > a.maxX) break;
481 if (b.maxY < a.minY || b.minY > a.maxY) continue;
482 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
483 }
484 }
485 } else {
486 int startJ = begin2;
487 for (int i = begin1; i < end1; ++i) {
488 const Box& a = boxes[static_cast<size_t>(i)];
489 while (startJ < end2 && boxes[static_cast<size_t>(startJ)].maxY < a.minY) ++startJ;
490 for (int j = startJ; j < end2; ++j) {
491 sweepChecks++;
492 const Box& b = boxes[static_cast<size_t>(j)];
493 if (b.minY > a.maxY) break;
494 if (b.maxX < a.minX || b.minX > a.maxX) continue;
495 if (a.intersects(b)) pairs.emplace_back(a.entity, b.entity);
496 }
497 }
498 }
499#ifdef OCTARINE_PROFILING
500 PROFILE_COUNTER_ADD("Collision: Bipartite Sweep Inner Loop Checks", sweepChecks);
501#else
502 (void)sweepChecks;
503#endif
504 }
505 }

509 // NOLINTNEXTLINE(misc-no-recursion) - KD-Tree cross-checking inherently requires recursion.
510 static void ProcessSpanningBipartite(std::vector<Box>& boxes, const int begin, const int end, const int leftEnd,
511 const int rightStart, const int dimension, const int nextDimension,
512 std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
513 if (leftEnd >= rightStart) return;
514
515 // Find the min/max bounds of the Spanning set along the current dimension
516 float spanMin = std::numeric_limits<float>::max();
517 float spanMax = std::numeric_limits<float>::lowest();
518 for (int i = leftEnd; i < rightStart; ++i) {
519 const float minV = dimension == 0 ? boxes[static_cast<size_t>(i)].minX : boxes[static_cast<size_t>(i)].minY;
520 const float maxV = dimension == 0 ? boxes[static_cast<size_t>(i)].maxX : boxes[static_cast<size_t>(i)].maxY;
521 if (minV < spanMin) spanMin = minV;
522 if (maxV > spanMax) spanMax = maxV;
523 }
524
525 const auto activeLeftBegin = std::partition(boxes.begin() + begin, boxes.begin() + leftEnd, [=](const Box& a) {
526 const float maxV = dimension == 0 ? a.maxX : a.maxY;
527 return maxV < spanMin;
528 });
529 const int leftOverlapStart = static_cast<int>(std::distance(boxes.begin(), activeLeftBegin));
530
531 const auto activeRightEnd = std::partition(boxes.begin() + rightStart, boxes.begin() + end, [=](const Box& a) {
532 const float minV = dimension == 0 ? a.minX : a.minY;
533 return minV <= spanMax;
534 });
535 const int rightOverlapEnd = static_cast<int>(std::distance(boxes.begin(), activeRightEnd));
536
537 FindIntersectionsSweepBipartite(boxes, leftOverlapStart, leftEnd, leftEnd, rightStart, nextDimension,
538 intersectingPairs);
539 FindIntersectionsSweepBipartite(boxes, leftEnd, rightStart, rightStart, rightOverlapEnd, nextDimension,
540 intersectingPairs);
541 }
542 // KD-Tree inherently requires recursion.
543 // NOLINTNEXTLINE(misc-no-recursion)
544 static void FindIntersectionsRecursive(std::vector<Box>& boxes, const int begin, const int end, int dimension,
545 const int depth, std::vector<std::pair<Entity, Entity>>& intersectingPairs) {
546 const int count = end - begin;
547 if (count <= 1) {
548 return;
549 }
550
551 if (count < kBruteforceCutoff || depth >= kMaxRecursionDepth) {
552 FindIntersectionsBruteForce(boxes, begin, end, intersectingPairs);
553 return;
554 }
555
556 const int mid = (begin + end) / 2;
557 float medianValue = 0.0f;
558 {
559 ACCUMULATE_PROFILE_SCOPE("Find Median");
560 std::nth_element(boxes.begin() + begin, boxes.begin() + mid, boxes.begin() + end,
561 [dimension](const Box& a, const Box& b) {
562 if (dimension == 0) return a.maxX + a.minX < b.maxX + b.minX;
563 return a.maxY + a.minY < b.maxY + b.minY;
564 });
565 medianValue = (dimension == 0)
566 ? (boxes[static_cast<size_t>(mid)].minX + boxes[static_cast<size_t>(mid)].maxX) * kHalf
