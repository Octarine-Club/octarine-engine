14// Lock-free, multi-producer render queue. Pre-sized to a fixed capacity; parallel
15// systems atomically claim a slot via fetch_add and write directly. No per-system
16// channel + playback merge — that intermediate copy was the bulk of the overhead in
17// the previous CommandBuffer-driven design.
18//
19// Sort() uses an LSB radix sort on the precomputed RenderKey::sortKey. With ~32K+ keys
20// this is dramatically faster than std::sort on the full 80-byte RenderKey because the
21// radix passes only touch 16-byte (key, srcIdx) entries. The full 80-byte permutation is
22// applied via a gather into a scratch buffer at the end — sequential writes are
23// prefetch-friendly and the source data fits in L3 cache at typical entry counts.
24//
25// Capacity must comfortably exceed the max keys-per-frame for the scene; an overflow
26// asserts in debug. Bump Constants::kInitialRenderQueueCapacity if you hit it.
27class RenderQueue {

77 void Sort() {
78 const size_t n = std::min(count_.load(std::memory_order_relaxed), render_keys_.size());
79 if (n <= 1) return;
80
81 // Build (sortKey, srcIdx) entries — 16 bytes each. Sorting these instead of the
82 // 80-byte RenderKeys is the main speedup; we only gather the heavy keys once at
83 // the end via a sequential-write scatter into a scratch buffer.
84 sort_entries_.resize(n);
85 for (size_t i = 0; i < n; ++i) {
86 sort_entries_[i].key = render_keys_[i].sortKey;
87 sort_entries_[i].srcIdx = static_cast<std::uint32_t>(i);
88 }
89
90 RadixSort();
91
92 // Gather: write the sorted RenderKeys sequentially into permute_scratch_ by
93 // reading from random source positions. Sequential writes are prefetch-friendly;
94 // the random reads hit ~2.5 MB of source data which fits comfortably in L3 cache.
95 // The scratch vector is kept alive across frames so this doesn't allocate after
96 // the first frame.
97 // Resize to full capacity — after the swap this becomes render_keys_, and ClaimSlot()
98 // indexes into it up to the pre-sized capacity.
99 permute_scratch_.resize(render_keys_.size());
100 const auto* perm = sort_entries_.data();
101 const auto* src = render_keys_.data();
102 auto* dst = permute_scratch_.data();
103 for (size_t i = 0; i < n; ++i) {
104 dst[i] = src[perm[i].srcIdx];
105 }
106 std::swap(render_keys_, permute_scratch_);
107 }

136 // LSB radix sort on the 64-bit key with two optimizations over the naive 8-pass version:
137 //
138 // 1. Single-pass histogram: build all 8 byte-histograms in one scan of the data instead
139 // of re-reading the array for each pass. At 32k entries × 16 bytes this avoids ~3.5 MB
140 // of redundant reads and the associated cache-miss overhead.
141 //
142 // 2. Skip-uniform-byte: if every entry has the same value for a given byte (common — e.g.
143 // all sprites on layer 0 share the top 2 bytes), that pass cannot reorder anything.
144 // Skipping it saves both the scatter pass AND its prefix-sum. On a typical stress scene
145 // this cuts 8 passes to ~4.
146 //
147 // Pass-parity bookkeeping ensures the final sorted result lives in sort_entries_ regardless
148 // of how many passes actually executed.
149 void RadixSort() {
150 const size_t n = sort_entries_.size();
151 if (n <= 1) return;
152 radix_scratch_.resize(n);
153
154 // --- Phase 1: build all 8 histograms in one scan ---
155 std::array<std::array<size_t, 256>, 8> histograms{};
156 for (size_t i = 0; i < n; ++i) {
157 const std::uint64_t k = sort_entries_[i].key;
158 histograms[0][k & 0xFFu]++;
159 histograms[1][(k >> 8) & 0xFFu]++;
160 histograms[2][(k >> 16) & 0xFFu]++;
161 histograms[3][(k >> 24) & 0xFFu]++;
162 histograms[4][(k >> 32) & 0xFFu]++;
163 histograms[5][(k >> 40) & 0xFFu]++;
164 histograms[6][(k >> 48) & 0xFFu]++;
165 histograms[7][(k >> 56) & 0xFFu]++;
166 }
167
168 // --- Phase 2: determine which passes are needed ---
169 // A pass is skippable when exactly one bucket holds all n entries (all bytes identical).
170 std::array<bool, 8> skip{};
171 for (size_t pass = 0; pass < 8; ++pass) {
172 for (size_t bucket = 0; bucket < 256; ++bucket) {
173 if (histograms[pass][bucket] == n) {
174 skip[pass] = true;
175 break;
176 }
177 }
178 }
179
180 // --- Phase 3: scatter passes (LSB → MSB), skipping uniform bytes ---
181 std::vector<SortEntry>* in = &sort_entries_;
182 std::vector<SortEntry>* out = &radix_scratch_;
183
184 for (size_t pass = 0; pass < 8; ++pass) {
185 if (skip[pass]) continue;
186
187 const size_t shift = pass * 8;
188
189 // Convert the histogram to exclusive prefix sums (offsets) for scatter.
190 auto& counts = histograms[pass];
191 size_t sum = 0;
192 for (auto& c : counts) {
193 const size_t saved = c;
194 c = sum;
195 sum += saved;
196 }
197
198 // Scatter
199 for (size_t i = 0; i < n; ++i) {
200 const auto bucket = static_cast<std::uint8_t>(((*in)[i].key >> shift) & 0xFFu);
201 (*out)[counts[bucket]++] = (*in)[i];
202 }
203 std::swap(in, out);
204 }
205
206 // Ensure the sorted result lives in sort_entries_.
207 if (in != &sort_entries_) {
208 std::swap(sort_entries_, radix_scratch_);
209 }
210 }

213 std::atomic<size_t> count_{0};
214
215 std::vector<SortEntry> sort_entries_;
216 std::vector<SortEntry> radix_scratch_;
217 std::vector<RenderKey> permute_scratch_; // Gather output buffer, kept alive to avoid re-alloc.
218};
