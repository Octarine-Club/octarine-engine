65 // Entity-keyed cache (mirrors Renderer/SpriteRenderCache): one rasterized texture per text
66 // entity, re-rasterized only when its (fontId, text, color) changes. Steady-state lookups do
67 // no heap allocation, and a label that updates every frame (score/timer) frees its old texture
68 // instead of leaking a fresh one per frame.

119 static TextPlacement ResolvePlacement(Registry* registry, const Entity entity, const TextLabelComponent& text,
120 const float textWidth, const float textHeight) {
121 glm::vec2 origin = text.position;
122 bool effectivelyFixed = text.isFixed;
123 int renderLayer = text.layer;
124 if (registry->HasComponent<UIRectComponent>(entity)) {
125 const auto& rect = registry->GetComponent<UIRectComponent>(entity);
126 const float slackX = rect.Width() - textWidth;
127 const float slackY = rect.Height() - textHeight;
128 const auto offsetFor = [](const TextAlign align, const float slack) {
129 static constexpr float kHalf = 0.5f;
130 if (align == TextAlign::Center) return slack * kHalf;
131 if (align == TextAlign::End) return slack;
132 return 0.0f;
133 };
134 origin = glm::vec2(rect.left + offsetFor(text.hAlign, slackX), rect.top + offsetFor(text.vAlign, slackY)) +
135 text.position;
136 effectivelyFixed = true;
137 renderLayer = rect.layer;
138 } else if (registry->HasComponent<GlobalTransformComponent>(entity)) {
139 const auto& transform = registry->GetComponent<GlobalTransformComponent>(entity);
140 origin += transform.position;
141 }
142 return {origin, effectivelyFixed, renderLayer};
143 }

154 // Entity → last-rasterized label. Re-rasterized only when (fontId, text, color) changes, so a
155 // label that updates every frame holds a single texture rather than leaking one per frame. Two
156 // entities with identical text no longer share a texture (entity-keyed, not content-keyed) —
157 // fine for typical UI label counts. Entries for despawned text entities persist until the system
158 // is destroyed (its destructor frees every cached texture).
159 mutable std::unordered_map<EcsId, TextCacheEntry> text_cache_;
160
161 // Store the freshly rasterized texture as this entity's cache entry, freeing any texture it was
162 // previously holding, and return the iterator to the live entry. `it` is this entity's current
163 // slot (text_cache_.end() if it had none).
164 std::unordered_map<EcsId, TextCacheEntry>::iterator StoreEntry(std::unordered_map<EcsId, TextCacheEntry>::iterator it,
165 Entity entity, const TextLabelComponent& text,
166 Uint32 color, SDL_Texture* texture, float width,
167 float height) const {
168 if (it != text_cache_.end()) {
169 if (it->second.texture != nullptr) SDL_DestroyTexture(it->second.texture);
170 it->second = {text.fontId, text.text, color, texture, width, height};
171 return it;
172 }
173 return text_cache_.emplace(entity.GetId(), TextCacheEntry{text.fontId, text.text, color, texture, width, height})
174 .first;
175 }

177 // Rasterize `text` to a freshly created SDL_Texture, writing its size into outW/outH.
178 // The atlas fast path composes the surface by blitting resident glyphs from the font's GlyphAtlas
179 // and falls back to TTF_RenderText_Blended when no atlas is baked or the string hits an uncovered
180 // glyph. Returns nullptr on failure; caller owns the returned texture.
181 static SDL_Texture* RasterizeLabel(const AssetManager& assetManager, SDL_Renderer* sdlRenderer, TTF_Font* font,
182 const TextLabelComponent& text, float& outW, float& outH) {
183 const SDL_Color sdlColor{text.color.r, text.color.g, text.color.b, text.color.a};
184 SDL_Surface* surface = nullptr;
185 const GlyphAtlas* atlas = assetManager.GetGlyphAtlas(text.fontId);
186 if (atlas != nullptr && atlas->IsLoaded()) {
187 surface = ComposeFromAtlas(*atlas, text.text, sdlColor);
188 }
189 if (surface == nullptr) {
190 surface = TTF_RenderText_Blended(font, text.text.c_str(), 0, sdlColor);
191 }
192 if (surface == nullptr) {
193 Logger::Error("RenderText: surface compose failed: " + std::string(SDL_GetError()));
194 return nullptr;
195 }
196 SDL_Texture* texture = SDL_CreateTextureFromSurface(sdlRenderer, surface);
197 SDL_DestroySurface(surface);
198 if (texture == nullptr) {
199 Logger::Error("SDL_CreateTextureFromSurface failed: " + std::string(SDL_GetError()));
200 return nullptr;
201 }
202 outW = 0;
203 outH = 0;
204 SDL_GetTextureSize(texture, &outW, &outH);
205 return texture;
206 }

208 // Atlas compose: build an RGBA destination surface, blit each codepoint's rect from the atlas
209 // into it, color-mod the source surface to tint the white-rasterized atlas glyphs to the
210 // requested color. Returns nullptr on any missing codepoint (caller falls back to TTF). Layout
211 // is top-aligned — not pixel-perfect for descenders, but readable for ASCII at typical UI sizes;
212 // ascent-aware baseline alignment is a follow-up.
213 static SDL_Surface* ComposeFromAtlas(const GlyphAtlas& atlas, const std::string& text, SDL_Color color) {
214 float totalW = 0.0F;
215 for (const char c : text) {
216 const auto cp = static_cast<std::uint32_t>(static_cast<unsigned char>(c));
217 const auto* g = atlas.Find(cp);
218 if (g == nullptr) return nullptr;
219 totalW += g->advance;
220 }
221 const int destW = std::max(1, static_cast<int>(std::ceil(totalW)));
222 const int destH = std::max(1, atlas.LineSkip());
223 SDL_Surface* dst = SDL_CreateSurface(destW, destH, SDL_PIXELFORMAT_RGBA32);
224 if (dst == nullptr) return nullptr;
225
226 SDL_Surface* src = atlas.SourceSurface();
227 SDL_SetSurfaceColorMod(src, color.r, color.g, color.b);
228 SDL_SetSurfaceAlphaMod(src, color.a);
229 SDL_SetSurfaceBlendMode(src, SDL_BLENDMODE_BLEND);
230
231 float pen = 0.0F;
232 for (const char c : text) {
233 const auto cp = static_cast<std::uint32_t>(static_cast<unsigned char>(c));
234 const auto* g = atlas.Find(cp);
235 SDL_Rect srcR{static_cast<int>(g->x), static_cast<int>(g->y), static_cast<int>(g->w), static_cast<int>(g->h)};
236 SDL_Rect dstR{static_cast<int>(pen), 0, static_cast<int>(g->w), static_cast<int>(g->h)};
237 SDL_BlitSurface(src, &srcR, dst, &dstR);
238 pen += g->advance;
239 }
240
241 // Reset mods so the next compose (or a parallel one for a differently-colored label) starts
242 // from a clean state on the shared atlas surface.
243 SDL_SetSurfaceColorMod(src, 255, 255, 255);
244 SDL_SetSurfaceAlphaMod(src, 255);
245 return dst;
246 }
