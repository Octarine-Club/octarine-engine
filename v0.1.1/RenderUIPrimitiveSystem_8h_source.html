11// Renders SquarePrimitiveComponent entities whose rectangle is driven by UIRectComponent
12// (layout-driven colored panels). Registered serially after UILayoutSystem. World squares without
13// UIRectComponent continue through the parallel RenderPrimitiveSystem path unchanged; the two
14// queries are disjoint because UI nodes carry no GlobalTransformComponent.
15//
16// The RenderQueue is fetched per-call via the ContextFacade, the same way RenderTextSystem does:
17// the serial RegisterSystem wrapper does not invoke a Prepare() hook (only the parallel wrapper
18// does), so caching it in a member would leave it null.
19class RenderUIPrimitiveSystem {
20 public:
21 void operator()(const ContextFacade& ctx, const UIRectComponent& rect, const SquarePrimitiveComponent& square) const {
22 // Position and size come from the resolved rect; color and blend come from the square. Layer
23 // is the rect's computed layer (canvas base + accumulated z), matching RenderUISpriteSystem.
24 auto& renderQueue = ctx.GetRegistry()->Get<RenderQueue>();
25 auto& cmd = renderQueue.EmplaceSquare(static_cast<unsigned int>(rect.layer), rect.top, nullptr, square.blendMode);
26 cmd.destRect = {rect.left, rect.top, rect.Width(), rect.Height()};
27 cmd.color = SDL_Color{square.color.r, square.color.g, square.color.b, square.color.a};
28 cmd.rotation = 0.0;
29 cmd.blendMode = octarine::ToSdlBlendMode(square.blendMode);
30 }
31};
