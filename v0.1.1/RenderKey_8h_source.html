11// Render entry — slim header (precomputed sortKey + type) plus one of three POD payloads
12// in a union, selected by `type`. Inlining the payload here means a sorted iteration is
13// fully sequential reads through one vector instead of a random gather into per-type
14// command arrays.
15//
16// `sortKey` packs all of (layer, depthBand, type, batchKey-hash) into a 64-bit integer
17// so RenderQueue::Sort can use a comparison-free radix sort. Layout, MSB → LSB:
18// bits 63-48 : layer (16 bits, unsigned — 64K layers covers kDebugUIBaseLayer headroom)
19// bits 47-28 : depthBand + bias (20 bits — ±16M world pixels at 32px bands)
20// bits 27-24 : type (4 bits)
21// bits 23-21 : blend mode (3 bits) — clusters same-blend runs within a band so SDL3
22// doesn't break batches on blend-state changes.
23// bits 20- 0 : batchKey hash (21 bits) — clusters same-texture runs within a
24// band+blend so SDL3's internal renderer can coalesce them. 2M slots makes
25// collisions vanishingly rare for any realistic texture count; collisions
26// only degrade batching, never produce wrong pixels.
27//
28// type is also stored separately because Renderer::Render dispatches on it to choose the
29// payload union member.
30struct RenderKey {
31 std::uint64_t sortKey{};
32 RenderableType type{};
33
34 union Payload {
35 SpriteCommand sprite;
36 SquareCommand square;
37 TextCommand text;
38
39 // Leaves union storage uninitialized — the producing system always overwrites the
40 // active member chosen by `type`. Skipping zero-init saves ~56 bytes of writes per
41 // emplace, which is meaningful at 100K+ keys per frame.
42 Payload() {}
43 } payload;
44
45 RenderKey() = default;

47 static std::uint64_t ComputeSortKey(unsigned int layer, float depth, RenderableType type, const void* batchKey,
48 std::uint8_t blendBits = 0) {
49 std::int32_t band = 0;
50 if constexpr (Constants::kRenderBatchYBandPx > 0.0f) {
51 band = static_cast<std::int32_t>(std::floor(depth / Constants::kRenderBatchYBandPx));
52 }
53
54 // Bias depthBand to non-negative 20-bit range. Layer is unsigned, masked to 16 bits
55 // (kDebugUIBaseLayer is ~1000, so 64K headroom is plenty).
56 constexpr std::int32_t kBandBias = 1 << 19;
57 const std::uint64_t layer16 = static_cast<std::uint64_t>(layer & 0xFFFFu);
58 const std::uint64_t band20 = static_cast<std::uint64_t>(static_cast<std::uint32_t>(band + kBandBias) & 0xFFFFFu);
59 const std::uint64_t type4 = static_cast<std::uint64_t>(static_cast<std::uint8_t>(type)) & 0xFu;
60
61 // 21-bit hash. Shift off allocator alignment noise then XOR-fold; for ~20 textures
62 // mapped into 2M slots the collision probability is effectively zero. nullptr (used
63 // by primitives) hashes to 0, which keeps all primitives clustered. Blend mode sits
64 // above the hash so same-blend runs cluster before same-texture runs.
65 const auto ptr = reinterpret_cast<std::uintptr_t>(batchKey);
66 const std::uint64_t hash21 = static_cast<std::uint64_t>((ptr >> 4) ^ (ptr >> 25)) & 0x1FFFFFu;
67 const std::uint64_t blend3 = static_cast<std::uint64_t>(blendBits) & 0x7u;
68
69 // Field offsets of the packed layout documented at the top of this header.
70 constexpr int kLayerShift = 48;
71 constexpr int kBandShift = 28;
72 constexpr int kTypeShift = 24;
73 constexpr int kBlendShift = 21;
74 return (layer16 << kLayerShift) | (band20 << kBandShift) | (type4 << kTypeShift) | (blend3 << kBlendShift) | hash21;
75 }
