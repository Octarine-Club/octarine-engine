52// Specialize for each component to expose it to Lua. Each specialization MUST provide:
53// static constexpr const char* kLuaKey; // entity-table key, e.g. "health"
54// static constexpr const char* kUsertypeName; // sol2 usertype name, e.g. "health_component"
55// static T fromLua(const sol::object& data); // sol::object (not table) so bare-string/int edge cases fit
56// static void bindUsertype(sol::state& lua); // call lua.new_usertype<T>(kUsertypeName, ...);
57//
58// bindUsertype may include both fields and member functions in the new_usertype call,
59// e.g. "damage", &HealthComponent::Damage alongside "max_health", &T::max_health.
60//
61// Component-method hard rule: a component method may read/write the component's own fields
62// ONLY. No Registry&, no EventBus*, no cross-entity reads/writes, no archetype mutation, no
63// allocation, no event emission. Anything crossing those lines belongs in a system. The
64// moment a method needs side effects beyond its own fields, move it out.
65//
66// For derived read-only state (IsDead, Fraction), bind with sol::property so scripts get
67// field-style access: "is_dead", sol::property(&T::IsDead).
68// For fields with invariants (current_health clamped to [0, max_health]), bind with
69// sol::property + a clamping setter so Lua can't bypass via direct field assignment.
70template <typename T>
71struct LuaBinding;
