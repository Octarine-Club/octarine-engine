71 void Update() {
72 ACCUMULATE_PROFILE_SCOPE("Query::Update");
73 const uint64_t current_gen = registry_->ArchetypeGeneration();
74 if (current_gen == cached_generation_) {
75 return;
76 }
77 ACCUMULATE_PROFILE_SCOPE("Query::Update (rematch)");
78
79 if (cached_generation_ == UINT64_MAX) {
80 matched_ = registry_->GetMatchingArchetypes(sorted_type_);
81 if (!excluded_.empty()) {
82 std::erase_if(matched_, [&](const Archetype* arch) { return IsExcluded(*arch); });
83 }
84 archetype_query_ = ArchetypeQuery<TComponents...>(type_, matched_, include_inactive_);
85 } else if (!sorted_type_.empty()) {
86 // Incrementally test archetypes created since last update.
87 const auto& log = registry_->ArchetypeLog();
88 for (uint64_t gen = cached_generation_; gen < current_gen; ++gen) {
89 Archetype* arch = log[gen];
90 if (Registry::MatchesType(*arch, sorted_type_) && !IsExcluded(*arch)) {
91 matched_.push_back(arch);
92 archetype_query_.AddMatchingArchetype(arch);
93 }
94 }
95 }
96
97 cached_generation_ = current_gen;
98 }
