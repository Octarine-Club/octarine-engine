65 static ComponentMaskAndCollider ScanComponentsForMaskAndCollider(const sol::table& currentData) {
66 ComponentMaskAndCollider result;
67 const sol::optional<sol::table> componentsOpt = currentData["components"];
68 if (!componentsOpt || !componentsOpt.value().valid()) {
69 return result;
70 }
71
72 const sol::table& components = componentsOpt.value();
73 result.mask = components.get<sol::object>("entity_mask");
74 result.hasMask = result.mask.valid() && result.mask.is<int>();
75
76 const sol::object boxCollider = components.get<sol::object>("box_collider");
77 result.hasCollider = boxCollider.valid() && boxCollider.is<sol::table>();
78
79 if (result.hasCollider && result.hasMask) {
80 return result;
81 }
82
83 for (const auto& [name, data] : components) {
84 if (data.is<sol::table>() && (name.is<int>() || name.is<size_t>())) {
85 ScanInnerComponentTable(data.as<sol::table>(), result);
86 if (result.hasCollider && result.hasMask) {
87 break;
88 }
89 }
90 }
91 return result;

94 // Reads top-level `mask` int from the entity table and attaches an EntityMaskComponent.
95 // Also attaches a default EntityMaskComponent when the entity declares a `box_collider`
96 // (KDTreeCollisionSystem queries this component, so collider entities must have one).
97 static void ApplyEntityMask(const sol::table& currentData, Registry* registry, const Entity& entity) {
98 const sol::object topLevelMask = currentData.get<sol::object>("entity_mask");
99 const bool hasTopLevelMask = topLevelMask.valid() && topLevelMask.is<int>();
100
101 const auto [componentMask, hasComponentMask, hasCollider] = ScanComponentsForMaskAndCollider(currentData);
102
103 if (hasTopLevelMask && hasComponentMask) {
104 Logger::Error("LoadEntityFromLua: entity_mask set at both top-level and components - pick one. Skipping mask.");
105 return;
106 }
107
108 if (!hasTopLevelMask && !hasComponentMask && !hasCollider) return;
109
110 const sol::object& maskValue = hasTopLevelMask ? topLevelMask : componentMask;
111 const int raw = maskValue.is<int>() ? maskValue.as<int>() : Constants::kDefaultEntityMask;
112 registry->AddComponent(entity, EntityMaskComponent(EntityMask(static_cast<unsigned long long>(raw))));
113 }

159 static void LoadEntityComponents(const sol::table& currentData, Registry* registry, const Entity& entity) {
160 const std::string componentsKey = "components";
161 sol::optional<sol::table> componentsTableOpt = currentData[componentsKey];
162 if (!componentsTableOpt || !componentsTableOpt.value().valid()) {
163 Logger::Info("LoadEntityFromLua: Entity has no 'components' table. Skipping.");
164 return;
165 }
166
167 const sol::table& componentsTable = componentsTableOpt.value();
168 for (const auto& [name, data] : componentsTable) {
169 if (data.is<sol::table>() && (name.is<int>() || name.is<size_t>())) {
170 for (const auto& [innerName, innerData] : data.as<sol::table>()) {
171 if (innerName.is<std::string>()) {
172 ProcessComponentEntry(registry, entity, innerName.as<std::string>(), innerData);
173 }
174 }
175 } else if (name.is<std::string>()) {
176 ProcessComponentEntry(registry, entity, name.as<std::string>(), data);
177 }
178 }

180 MaybeAttachGlobalTransform(registry, entity);

193 static std::optional<Entity> LoadEntityFromLua(Registry* registry, const sol::table& entityData) {
194 if (!entityData.valid()) {
195 Logger::Error("LoadEntityFromLua: Invalid root entity data table.");
196 return std::nullopt;
197 }
198
199 std::optional<Entity> rootEntity;
200 std::stack<std::pair<sol::table, std::optional<Entity>>> nodesToProcess;
201 nodesToProcess.emplace(entityData, std::nullopt);
202
203 while (!nodesToProcess.empty()) {
204 auto [currentData, parentEntity] = nodesToProcess.top();
205 nodesToProcess.pop();
206
207 // Create the entity and establish parentage
208 const Entity entity = registry->CreateEntity();
209 if (parentEntity) {
210 registry->SetParent(entity, *parentEntity);
211 } else {
212 rootEntity = entity;
213 }
214
215 ApplyName(currentData, registry, entity);
216 TagAndGroupEntity(currentData, registry, entity);
217 ApplyEntityMask(currentData, registry, entity);
218 LoadEntityComponents(currentData, registry, entity);
219
220 // Add any child entities to the stack to be processed.
221 const std::string entitiesKey = "entities";
222 sol::optional<sol::table> childEntitiesOpt = currentData[entitiesKey];
223 if (childEntitiesOpt && childEntitiesOpt.value().valid()) {
224 const sol::table& childEntitiesTable = childEntitiesOpt.value();
225
226 std::vector<sol::table> children;
227 children.reserve(childEntitiesTable.size());
228 for (const auto& [key, value] : childEntitiesTable) {
229 children.push_back(value.as<sol::table>());
230 }
231
232 for (auto it = children.rbegin(); it != children.rend(); ++it) {
233 nodesToProcess.emplace(*it, entity);
234 }
235 }
236 }
237
238 return rootEntity;
239 }

242 // Add a GlobalTransformComponent iff (a) any local transform component is present and
243 // (b) at least one world-space consumer system (sprite/primitive/collider/UI button/text label)
244 // is attached. Keeps script-only or camera-anchor entities lean.
245 static void MaybeAttachGlobalTransform(Registry* registry, const Entity& entity) {
246 const bool isGrid = registry->HasComponent<ColorGridComponent>(entity);
247 if (isGrid && !registry->HasComponent<PositionComponent>(entity)) {
248 registry->AddComponent(entity, PositionComponent{});
249 }
250
251 const bool hasAnyLocal = registry->HasComponent<PositionComponent>(entity) ||
252 registry->HasComponent<ScaleComponent>(entity) ||
253 registry->HasComponent<RotationComponent>(entity);
254 if (!hasAnyLocal) return;
255
256 const bool needsGlobal =
257 registry->HasComponent<SpriteComponent>(entity) || registry->HasComponent<SquarePrimitiveComponent>(entity) ||
258 registry->HasComponent<ColorGridComponent>(entity) || registry->HasComponent<BoxColliderComponent>(entity) ||
259 registry->HasComponent<UIButtonComponent>(entity) || registry->HasComponent<TextLabelComponent>(entity);
260 if (!needsGlobal) return;
261
262 if (registry->HasComponent<GlobalTransformComponent>(entity)) return;
263 registry->AddComponent(entity, GlobalTransformComponent{});
264 }

267 const sol::object value = currentData.get<sol::object>(key);
