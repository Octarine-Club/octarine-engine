188 template <typename Fn>
189 static bool ForEachCommaToken(const std::string& spec, Fn&& fn) {
190 if (spec.empty()) return false;
191 size_t start = 0;
192 while (start < spec.size()) {
193 size_t end = spec.find(',', start);
194 if (end == std::string::npos) end = spec.size();
195 std::string token = spec.substr(start, end - start);
196 const size_t first = token.find_first_not_of(" \t\r\n");
197 if (first != std::string::npos) {
198 const size_t last = token.find_last_not_of(" \t\r\n");
199 token = token.substr(first, last - first + 1);
200 if (fn(token)) return true;
201 }
202 start = end + 1;
203 }
204 return false;
205 }

207 static bool MatchesKeyString(const std::string& keySpec, const std::string& keyName, const InputSystem* inputSystem) {
208 return ForEachCommaToken(keySpec, [&](const std::string& token) {
209 if (inputSystem) {
210 return inputSystem->MatchesKeyAlias(token, keyName);
211 }
212 return InputSystem::MakeKey(token) == InputSystem::MakeKey(keyName);
213 });
214 }

241 [[nodiscard]] bool HitTest(const Entity entity, const UIButtonComponent& button,
242 const GlobalTransformComponent* transform, const BoxColliderComponent* collider,
243 const octarine::Rect& camera, const float zoom, const float mouseX,
244 const float mouseY) const {
245 // Layout-driven UI button: the resolved rect already accounts for anchoring, so no collider or
246 // transform is needed — this is the preferred path for `ui_button` + `ui_anchor` entities.
247 if (registry_->HasComponent<UIRectComponent>(entity)) {
248 const auto& rect = registry_->GetComponent<UIRectComponent>(entity);
249 return mouseX >= rect.left && mouseX <= rect.right && mouseY >= rect.top && mouseY <= rect.bottom;
250 }
251 // World-space button: position the collider in transform space.
252 if (collider && transform) {
253 float x = transform->position.x + collider->offset.x * transform->scale.x;
254 float y = transform->position.y + collider->offset.y * transform->scale.y;
255 float w = static_cast<float>(collider->width) * transform->scale.x;
256 float h = static_cast<float>(collider->height) * transform->scale.y;
257 if (!button.isFixed) {
258 x = (x - camera.x) * zoom;
259 y = (y - camera.y) * zoom;
260 w *= zoom;
261 h *= zoom;
262 }
263 return mouseX >= x && mouseX <= x + w && mouseY >= y && mouseY <= y + h;
264 }
265 return false;
266 }
