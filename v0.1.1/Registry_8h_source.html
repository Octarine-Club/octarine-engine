30 template <typename T>
31 void RegisterComponent(const Entity& componentEntity) {
32 const ComponentID id = componentEntity.GetId();
33 ComponentInfo info{id, typeid(T).name(), sizeof(T), alignof(T)};
34 if constexpr (!std::is_trivially_copyable_v<T>) {
35 if constexpr (std::is_move_constructible_v<T>) {
36 info.move_construct = [](void* dst, void* src) { ::new (dst) T(std::move(*static_cast<T*>(src))); };
37 } else {
38 info.move_construct = [](void* dst, void* src) { std::memcpy(dst, src, sizeof(T)); };
39 }
40 }
41 if constexpr (!std::is_trivially_destructible_v<T>) {
42 info.destroy = [](void* ptr) { static_cast<T*>(ptr)->~T(); };
43 }
44 if constexpr (std::is_swappable_v<T>) {
45 info.swap = [](void* a, void* b) {
46 using std::swap;
47 swap(*static_cast<T*>(a), *static_cast<T*>(b));
48 };
49 }
50 auto [it, inserted] = component_infos_.try_emplace(id, std::move(info));
51 if (!inserted && it->second.name != typeid(T).name()) {
52 throw std::runtime_error("Component ID collision: " + it->second.name + " and " + typeid(T).name());
53 }
54 }

60 if (!inserted && it->second.name != name) {
61 throw std::runtime_error("Tag ID collision: " + it->second.name + " and " + name);
62 }

251 template <typename T>
252 T& GetComponent(const Entity entity) const {
253 const std::uint32_t id = entity.GetId();
254 if (id >= entity_locations_.size() || !entity_manager_->IsValid(entity) || !entity_locations_[id].archetype) {
255 throw std::runtime_error("Failed to get required component " + std::string(typeid(T).name()) + " for entity " +
256 std::to_string(entity.id));
257 }
258 const auto componentEntity = Component<T>();
259 const auto [archetype, chunkIndex, indexInChunk] = entity_locations_[id];
260
261 const auto compIdx = archetype->GetComponentIndex(componentEntity.GetId());
262 if (compIdx == Archetype::kInvalidComponentIndex) {
263 throw std::runtime_error("Failed to get required component " + std::string(typeid(T).name()) + " for entity " +
264 std::to_string(entity.id));
265 }
266 const auto componentArray = archetype->GetComponentArrayByIndex<T>(chunkIndex, compIdx);
267
268 if (componentArray == nullptr) {
269 throw std::runtime_error("Failed to get required component " + std::string(typeid(T).name()) + " for entity " +
270 std::to_string(entity.id));
271 }
272
273 return componentArray[indexInChunk];
274 }

299 SystemHandle<std::decay_t<Func>> RegisterSystem(Func&& func) {
300 using StoredFunc = std::decay_t<Func>;
301 struct SystemState {
302 StoredFunc func;
303 std::unique_ptr<ComponentQuery<TArgs...>> query;
304 };
305 auto state = std::make_shared<SystemState>(SystemState{std::forward<Func>(func), CreateQuery<TArgs...>()});
306
307 RegisteredSystem sys;
308 sys.name = PrettifyTypeName(typeid(StoredFunc).name());
309 sys.update = [state](const Registry& registry) {
310 state->query->Update();
311 // Pass by reference so captured state in the system lambda persists across invocations.
312 state->query->ForEach(state->func);
313
314 if constexpr (requires { state->func.GetCommandBuffer(); }) {
315 state->func.GetCommandBuffer().Playback(const_cast<Registry*>(&registry));
316 }
317 };
318
319 const SystemId id = systems_.size();
320 systems_.push_back(std::move(sys));
321 return SystemHandle<StoredFunc>(id, &state->func);
322 }

324 // Registers a system running per-entity callbacks in parallel across chunks on ThreadPool.
325 template <typename... TArgs, typename Func>
326 SystemHandle<std::decay_t<Func>> RegisterParallelSystem(Func&& func) {
327 using StoredFunc = std::decay_t<Func>;
328 static_assert(std::is_invocable_v<StoredFunc, Entity, float, TArgs&...> ||
329 std::is_invocable_v<StoredFunc, Entity, TArgs&...> ||
330 std::is_invocable_v<StoredFunc, float, TArgs&...> || std::is_invocable_v<StoredFunc, TArgs&...>,
331 "RegisterParallelSystem func must match void(Entity, float, TArgs&...), void(Entity, TArgs&...), "
332 "void(float, TArgs&...), or void(TArgs&...). "
333 "ContextFacade signatures are not supported on the parallel path.");
334
335 struct SystemState {
336 StoredFunc func;
337 std::unique_ptr<ComponentQuery<TArgs...>> query;
338 };
339 auto state = std::make_shared<SystemState>(SystemState{std::forward<Func>(func), CreateQuery<TArgs...>()});
340
341 RegisteredSystem sys;
342 sys.name = PrettifyTypeName(typeid(StoredFunc).name());
343 sys.update = [state](const Registry& registry) {
344 state->query->Update();
345 const float dt = registry.DeltaTime();
346 auto* regPtr = const_cast<Registry*>(&registry);
347
348 if constexpr (requires {
349 { state->func.Prepare(regPtr) } -> std::same_as<bool>;
350 }) {
351 if (!state->func.Prepare(regPtr)) return;
352 } else if constexpr (requires { state->func.Prepare(regPtr); }) {
353 state->func.Prepare(regPtr);
354 }
355
356 if constexpr (std::is_invocable_v<StoredFunc, Entity, float, TArgs&...>) {
357 state->query->ParallelForEach([&](Entity entity, TArgs&... args) { state->func(entity, dt, args...); });
358 } else if constexpr (std::is_invocable_v<StoredFunc, Entity, TArgs&...>) {
359 state->query->ParallelForEach([&](Entity entity, TArgs&... args) { state->func(entity, args...); });
360 } else if constexpr (std::is_invocable_v<StoredFunc, float, TArgs&...>) {
361 state->query->ParallelForEach([&](TArgs&... args) { state->func(dt, args...); });
362 } else if constexpr (std::is_invocable_v<StoredFunc, TArgs&...>) {
363 state->query->ParallelForEach([&](TArgs&... args) { state->func(args...); });
364 } else {
365 static_assert(!std::is_same_v<StoredFunc, StoredFunc>, "RegisterParallelSystem func signature mismatch.");
366 }
367
368 if constexpr (requires { state->func.GetCommandBuffer(); }) {
369 state->func.GetCommandBuffer().Playback(regPtr);
370 }
371 };
372
373 const SystemId id = systems_.size();
374 systems_.push_back(std::move(sys));
375 return SystemHandle<StoredFunc>(id, &state->func);
376 }
