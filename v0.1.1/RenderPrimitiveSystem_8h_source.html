31 void operator()(const SquarePrimitiveComponent& square, const GlobalTransformComponent& transform) const {
32 // square.position is a local offset from the entity's transform, so it scales with the
33 // entity the same way BoxColliderComponent::offset does in KDTreeCollisionSystem — otherwise a
34 // scaled entity's primitive grows but stays pinned at its unscaled offset.
35 const glm::vec2 scaledOffset = square.position * transform.scale;
36 const glm::vec2 origin = transform.position + scaledOffset;
37 const float w = square.width * transform.scale.x;
38 const float h = square.height * transform.scale.y;
39
40 const bool isOutsideCamera =
41 IsRenderableOutsideViewport(origin.x, origin.y, w, h, square.isFixed, camera_, windowWidth_, windowHeight_);
42
43 if (isOutsideCamera) {
44 PROFILE_COUNTER_INC(culledCounter_);
45 return;
46 }
47
48 PROFILE_COUNTER_INC(emplacedCounter_);
49
50 const float x = square.isFixed ? origin.x : (origin.x - camera_.x) * zoom_;
51 const float y = square.isFixed ? origin.y : (origin.y - camera_.y) * zoom_;
52 const float destW = square.isFixed ? w : w * zoom_;
53 const float destH = square.isFixed ? h : h * zoom_;
54
55 auto& cmd = renderQueue_->EmplaceSquare(static_cast<unsigned int>(square.layer), square.position.y, nullptr,
56 square.blendMode);
57 cmd.destRect = {x, y, destW, destH};
58 cmd.color = SDL_Color{square.color.r, square.color.g, square.color.b, square.color.a};
59 cmd.rotation = transform.rotation;
60 // The quad's top-left is `origin`, but pivot is an offset from transform.position, so rebase
61 // it onto the destRect the renderer receives. Both are in scaled pixels.
62 const glm::vec2 localPivot = transform.pivot - scaledOffset;
63 cmd.pivot = square.isFixed ? SDL_FPoint{localPivot.x, localPivot.y}
64 : SDL_FPoint{localPivot.x * zoom_, localPivot.y * zoom_};
65 cmd.blendMode = octarine::ToSdlBlendMode(square.blendMode);
66 }
