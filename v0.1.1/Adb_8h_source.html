19// One row of `adb devices -l`. `state` is adb's own word ("device", "offline", "unauthorized",
20// "no permissions", ...); only "device" means usable, which IsReady() reflects.
21struct AdbDevice {
22 std::string serial; // "emulator-5554" or "192.168.1.5:5555"
23 std::string state; // adb connection state
24 std::string model; // model:<...> from -l (underscores kept), empty when absent
25 std::string product; // product:<...> from -l, empty when absent
26
27 [[nodiscard]] bool IsReady() const { return state == "device"; }
28};
